\subsection{Vector Algebra}
\begin{definition}[Scalar]
    A scalar is a quantity with just a magnitude.
\end{definition}
\begin{definition}[Vector]
    A vector is a quantity with a magnitude and a direction, i.e. $\vec{v} = v_x\hat{x} + v_y\hat{y} + v_z\hat{z}$.
\end{definition}
\begin{definition}[Cross Product]
    The cross product is a vector perpendicular to both, set by the right hand rule, and it vanishes for parallel vectors:
    \begin{equation}
        \vec{v}\times\vec{w} = |\vec{v}||\vec{w}|\sin\theta\,\hat{n}
        = \begin{vmatrix} \hat{x} & \hat{y} & \hat{z} \\ v_x & v_y & v_z \\ w_x & w_y & w_z \end{vmatrix}
        = \hat{x}(v_yw_z - w_yv_z) - \hat{y}(v_xw_z - w_xv_z) + \hat{z}(v_xw_y - w_xv_y).
    \end{equation}
\end{definition}
\begin{definition}[Dot Product]
    The dot product measures how much of $\vec{v}$ points along $\vec{w}$, so it is zero for perpendicular vectors:
    \begin{equation}
        \vec{v}\cdot\vec{w} = v_xw_x + v_yw_y + v_zw_z = |\vec{v}||\vec{w}|\cos\theta.
    \end{equation}
\end{definition}
\begin{remark}
    Note $\vec{v}\times\vec{w} = -\vec{w}\times\vec{v}$, and $|\vec{v}| = \sqrt{\vec{v}\cdot\vec{v}}$.
\end{remark}
\begin{definition}[Field]
    A field is something with a value at every point in space. A scalar field assigns a number to each point. A vector field assigns a vector to each point.
\end{definition}

\subsection{Differential Calculus}
\begin{definition}[Del operator]
    The del operator
    \[\vec{\nabla} = \left(\frac{\partial}{\partial x}, \frac{\partial}{\partial y}, \frac{\partial}{\partial z}\right).\]
\end{definition}
\begin{definition}[Gradient]
    The gradient turns a scalar into a vector, pointing in the direction of steepest increase of $f$:
    \begin{equation}
        \vec{\nabla}f = \frac{\partial f}{\partial x}\hat{x} + \frac{\partial f}{\partial y}\hat{y} + \frac{\partial f}{\partial z}\hat{z}.
    \end{equation}
\end{definition}
\begin{remark}
    The gradient points in the steepest uphill direction, and its length grows where the hill is steeper. It is always perpendicular to the contours of constant $f$.
\end{remark}
\begin{definition}[Divergence]
    Divergence turns a vector into a scalar, measuring sources and sinks. For a small box, non-zero divergence means the inflow and outflow through its faces do not balance: arrows spreading from a point (positive) mean fluid is created there, arrows converging (negative) mean it is destroyed.
    \begin{equation}
        \vec{\nabla}\cdot\vec{F} = \frac{\partial F_x}{\partial x} + \frac{\partial F_y}{\partial y} + \frac{\partial F_z}{\partial z}.
    \end{equation}
\end{definition}
\begin{definition}[Curl]
    Curl turns a vector into a vector, measuring local rotation. Picture a small paddle wheel: if the flow spins it, the curl is non-zero there. Uniform flow does not spin it, but shear flow (parallel arrows, longer on one side) does, so straight field lines do not imply zero curl.
    \begin{equation}
        \vec{\nabla}\times\vec{F} = \begin{vmatrix} \hat{x} & \hat{y} & \hat{z} \\[2pt] \frac{\partial}{\partial x} & \frac{\partial}{\partial y} & \frac{\partial}{\partial z} \\[2pt] F_x & F_y & F_z \end{vmatrix} = \left(\frac{\partial}{\partial y}F_z - \frac{\partial}{\partial z}F_y, -(\frac{\partial}{\partial x}F_z - \frac{\partial}{\partial z}F_x), \frac{\partial}{\partial x}F_y - \frac{\partial}{\partial y}F_x\right).
    \end{equation}
\end{definition}
\begin{remark}
    The direction of the curl follows the right-hand rule: curl your fingers along the paddle wheel's rotation and your thumb points along $\vec{\nabla}\times\vec{F}$.
\end{remark}
\begin{remark}
    Divergence and curl are themselves fields: they can vanish in some places and not others, and a general field has both.
\end{remark}

\subsection{Integral Calculus}
\begin{definition}[Line Integral]
    A line integral sums $\vec{F}$ along a curve, picking up only the component tangent to the path:
    \[\int_C \vec{F}\cdot d\vec{l}.\]
\end{definition}
\begin{example}
    Take $\vec{F} = y\hat{x}$ along a counterclockwise quarter circle of radius $R$.

    On the circle $d\vec{l} = \frac{d\vec{r}}{d\phi}d\phi = R(-\sin\phi\,\hat{x} + \cos\phi\,\hat{y})d\phi$ and $y = R\sin\phi$, so
    \[\int_C \vec{F}\cdot d\vec{l} = \int_{0}^{\pi/2}\begin{pmatrix} R\sin\phi \\ 0 \end{pmatrix}\cdot R\begin{pmatrix}-\sin\phi\\ \cos\phi\end{pmatrix}d\phi = -R^2\int_0^{\pi/2}\sin^2\phi\,d\phi = -\frac{R^2\pi}{4}.\]
    It is negative because $\vec{F}$ points along $+\hat{x}$ while we travel in the $-\hat{x}$ direction over most of the arc.
\end{example}
\begin{definition}[Flux]
    Flux measures how much of the field pokes through a surface:
    \[\int_S \vec{F}\cdot d\vec{a} = \int_S \vec{F}\cdot\hat{n}\,da.\]
\end{definition}
\begin{remark}
    Flux is the total amount of a vector field flowing through a surface. Divergence is the flux per unit volume at a specific point.
\end{remark}
\begin{definition}[Volume Integral]
    A volume integral sums a scalar over a region:
    \[\int_V f\,d^3\vec{r} = \iiint_V f\,dx\,dy\,dz.\]
\end{definition}
\begin{remark}
    We write $\oint$ when the curve is a closed loop or the surface is closed.
\end{remark}

\subsubsection{The Fundamental Theorems}
\begin{theorem}[The Fundamental Theorem of Calculus]
    The boundary of an interval is two points:
    \[\int_a^b f'(x)\,dx = f(b) - f(a).\]
\end{theorem}
\begin{theorem}[The Fundamental Theorem for Gradients]
    \[\int_a^b \vec{\nabla}T \cdot d\vec{l} = T(b) - T(a).\]
    The boundary of a curve is its two endpoints. To find the Eiffel Tower's height, you can sum the rise of every step (left side) or subtract altimeter readings at the top and bottom (right side).
\end{theorem}
\begin{theorem}[The Fundamental Theorem for Curls]
    Another name for this theorem is Stokes' theorem.
    \begin{equation}
        \label{Stokes}
        \iint_S (\vec{\nabla}\times\vec{F})\cdot d\vec{S} = \oint_C \vec{F}\cdot d\vec{l}.
    \end{equation}
    Geometric meaning: to get the total swirl, I can work out the rotation at the boundary, or add up all the little whirlpools inside. Neighbouring whirlpools cancel along their shared edges, leaving only the boundary.
\end{theorem}
\begin{theorem}[The Fundamental Theorem for Divergence]
    \label{FTD}
    It has 2 important names: Gauss's theorem (or the divergence theorem) and Green's theorem. Respectively:
    \[\oiint_S (\vec{F}\cdot\hat{n})dS = \iiint_V(\vec{\nabla}\cdot\vec{F})dV,\]
    \[\oint_C Pdx + Qdy = \iint_R (\frac{\partial Q}{\partial x}-\frac{\partial P}{\partial y}) dxdy.\]
\end{theorem}

\subsection{Spherical Coordinates}
\begin{definition}[Spherical coordinates]
    $r$ is the distance from the origin, $\theta\in[0,\pi]$ the angle down from the $+z$ axis, and $\phi\in[0,2\pi)$ the angle around it. The unit vectors $\hat{r},\hat{\theta},\hat{\phi}$ change direction from point to point, so they cannot be pulled out of an integral; convert to $\hat{x},\hat{y},\hat{z}$ first. The elements are
    \[d\vec{l} = dr\,\hat{r} + r\,d\theta\,\hat{\theta} + r\sin\theta\,d\phi\,\hat{\phi}, \qquad d\tau = r^2\sin\theta\,dr\,d\theta\,d\phi, \qquad d\vec{a}_{\text{sphere}} = R^2\sin\theta\,d\theta\,d\phi\,\hat{r}.\]
\end{definition}
\begin{proposition}[Derivatives in spherical coordinates]
    \begin{align*}
        \vec{\nabla}f &= \frac{\partial f}{\partial r}\hat{r} + \frac{1}{r}\frac{\partial f}{\partial \theta}\hat{\theta} + \frac{1}{r\sin\theta}\frac{\partial f}{\partial \phi}\hat{\phi},\\
        \vec{\nabla}\cdot\vec{F} &= \frac{1}{r^2}\frac{\partial}{\partial r}\left(r^2F_r\right) + \frac{1}{r\sin\theta}\frac{\partial}{\partial\theta}\left(\sin\theta\,F_\theta\right) + \frac{1}{r\sin\theta}\frac{\partial F_\phi}{\partial\phi},\\
        \vec{\nabla}\times\vec{F} &= \frac{1}{r\sin\theta}\left[\frac{\partial}{\partial\theta}(\sin\theta\,F_\phi) - \frac{\partial F_\theta}{\partial\phi}\right]\hat{r} + \frac{1}{r}\left[\frac{1}{\sin\theta}\frac{\partial F_r}{\partial\phi} - \frac{\partial}{\partial r}(rF_\phi)\right]\hat{\theta} + \frac{1}{r}\left[\frac{\partial}{\partial r}(rF_\theta) - \frac{\partial F_r}{\partial\theta}\right]\hat{\phi},\\
        \nabla^2 f &= \frac{1}{r^2}\frac{\partial}{\partial r}\left(r^2\frac{\partial f}{\partial r}\right) + \frac{1}{r^2\sin\theta}\frac{\partial}{\partial\theta}\left(\sin\theta\frac{\partial f}{\partial\theta}\right) + \frac{1}{r^2\sin^2\theta}\frac{\partial^2 f}{\partial\phi^2}.
    \end{align*}
\end{proposition}
\begin{remark}
    For a purely radial field $\vec{F} = F_r(r)\hat{r}$, only the first term of the divergence survives, and the curl vanishes. So every central field $f(r)\hat{r}$ is curl-free.
\end{remark}

\subsection{Index Notation}
\begin{remark}
    A matrix carries two indices, $A_{ij}$, where $i$ says which row and $j$ says which column.
\end{remark}
\begin{definition}[Transpose]
    The transpose swaps the indices, $(A^T)_{ij} = A_{ji}$.
\end{definition}
\begin{definition}[Symmetry]
    A symmetric matrix has $A_{ij} = A_{ji}$, and an antisymmetric one has $B_{ij} = -B_{ji}$.
\end{definition}
\begin{definition}[Kronecker delta]
    The Kronecker delta equals $+1$ when its condition is fulfilled and $0$ otherwise, so it could be the identity matrix in index notation:
    \[\delta_{ij} = \begin{cases} +1 & i = j \\ 0 & i \neq j \end{cases}.\]
\end{definition}
\begin{example}
    \[\sum_j \delta_{ij}v_j = v_i.\]
\end{example}
\begin{proposition}[Matrix Multiplication]
    \[\vec{w} = A\vec{v} \implies w_i = \sum_j A_{ij}v_j.\]
\end{proposition}
\begin{proposition}[Dot Product]
    \[\vec{v}\cdot\vec{w} = \sum_i v_iw_i; \vec{\nabla}\cdot\vec{F} = \sum_i \partial_i F_i.\]
\end{proposition}
\begin{proposition}[Cross Product]
    \[(\vec{v}\times\vec{w})_i = \sum_{j,k}\varepsilon_{ijk}v_jw_k; (\vec{\nabla}\times\vec{F})_i = \sum_{j,k}\varepsilon_{ijk}\partial_j F_k.\]
\end{proposition}
\begin{definition}[Levi-Civita symbol]
    $\varepsilon_{ijk}$ is defined by
    \begin{itemize}
        \item $\varepsilon_{123} = \varepsilon_{231} = \varepsilon_{312} = +1$.
        \item $\varepsilon_{321} = \varepsilon_{213} = \varepsilon_{132} = -1$.
        \item If any two indices are equal, $\varepsilon_{ijk} = 0$.
    \end{itemize}
\end{definition}
\begin{example}
    For $i = 1$, only the terms with $(j,k) = (2,3)$ and $(3,2)$ survive:
    \[(\vec{v}\times\vec{w})_1 = \varepsilon_{123}v_2w_3 + \varepsilon_{132}v_3w_2 = v_2w_3 - v_3w_2.\]
\end{example}
\begin{proposition}[Epsilon-delta identity]
    \label{Epsilon-delta-id}
    Contracting two Levi-Civita symbols on a shared index gives
    \[\sum_i \varepsilon_{ijk}\varepsilon_{imn} = \delta_{jm}\delta_{kn} - \delta_{jn}\delta_{km}.\]
\end{proposition}

\subsection{Second Derivatives Identities}
\begin{definition}[Laplacian]
    Taking the divergence of a gradient gives
    \begin{equation}
        \vec{\nabla}\cdot(\vec{\nabla}f) = \frac{\partial^2 f}{\partial x^2} + \frac{\partial^2 f}{\partial y^2} + \frac{\partial^2 f}{\partial z^2} \equiv \nabla^2 f,
    \end{equation}
    where $\nabla^2$ is the Laplacian\footnote{Mathematicians often write $\Delta$, but in physics we use $\nabla^2$.}.
\end{definition}
\begin{remark}
    $\nabla^2$ also acts on vector fields, component by component: $(\nabla^2\vec{F})_i = \nabla^2 F_i$.
\end{remark}
The full list of second derivative identities:

\subsection{The Dirac Delta Function}
\begin{definition}[Dirac delta]
    $\delta(x)$ is zero everywhere except at $x = 0$, where it is an infinitely tall spike of unit area:
    \[\int_{-\infty}^{\infty}\delta(x)\,dx = 1.\]
\end{definition}
\begin{proposition}[Properties]
    \begin{enumerate}[label=(\roman*)]
        \item Sifting: $\int_a^b f(x)\,\delta(x - x_0)\,dx = f(x_0)$ if $a < x_0 < b$, and $0$ if $x_0$ is outside $[a,b]$.
        \item Scaling: $\delta(kx) = \frac{1}{|k|}\delta(x)$.
        \item In 3D: $\int_V f(\vec{r})\,\delta^3(\vec{r} - \vec{r}_0)\,d\tau = f(\vec{r}_0)$ if $\vec{r}_0 \in V$.
    \end{enumerate}
\end{proposition}
\begin{example}
    $\int_0^3 x^3\delta(x+1)\,dx = 0$ since $x = -1$ is outside the range, while
    \[\int_{-1}^{1}9x^2\,\delta(3x+1)\,dx = \frac{1}{3}\int_{-1}^{1}9x^2\,\delta\!\left(x+\tfrac{1}{3}\right)dx = \frac{1}{3}\cdot 9\cdot\frac{1}{9} = \frac{1}{3}.\]
\end{example}
\begin{theorem}[Divergence of $\hat{r}/r^2$]
    \label{div-r2}
    By the spherical divergence, $\vec{\nabla}\cdot\left(\hat{r}/r^2\right) = \frac{1}{r^2}\partial_r(1) = 0$ for $r \neq 0$, yet its flux through any sphere is $\oint\frac{1}{R^2}R^2\sin\theta\,d\theta\,d\phi = 4\pi$. By the divergence theorem, everything sits at the origin:
    \[\vec{\nabla}\cdot\left(\frac{\hat{r}}{r^2}\right) = 4\pi\,\delta^3(\vec{r}).\]
    So for a point charge, $\vec{\nabla}\cdot\vec{E} = \frac{q}{\epsilon_0}\delta^3(\vec{r}) = \rho/\epsilon_0$, as Gauss' law requires.
\end{theorem}

\subsection{The Theory of Vector Fields}
\begin{remark}
    The condition at infinity is needed: a field can be both curl-free and divergence-free without being zero, e.g.\ $\vec{F} = yz\,\hat{x} + zx\,\hat{y} + xy\,\hat{z} = \vec{\nabla}(xyz)$. Adding it to any field changes neither $\vec{\nabla}\cdot\vec{F}$ nor $\vec{\nabla}\times\vec{F}$.
\end{remark}
\begin{theorem}[Irrotational fields]
    \label{irrotational}
    Let $\vec{F}$ be a $C^1$ vector field on a simply connected domain (e.g.\ all of $\mathbb{R}^3$). The following are equivalent:
    \begin{enumerate}[label=(\roman*)]
        \item $\vec{\nabla}\times\vec{F} = 0$ everywhere.
        \item $\int_a^b \vec{F}\cdot d\vec{l}$ is independent of path, for any given end points.
        \item $\oint_C \vec{F}\cdot d\vec{l} = 0$ for every closed loop $C$.
        \item $\vec{F} = -\vec{\nabla}\psi$ for some scalar field $\psi$.
    \end{enumerate}
\end{theorem}
\begin{theorem}[Solenoidal fields]
    Let $\vec{F}$ be a $C^1$ vector field on a domain with no enclosed voids (e.g.\ all of $\mathbb{R}^3$). The following are equivalent:
    \begin{enumerate}[label=(\roman*)]
        \item $\vec{\nabla}\cdot\vec{F} = 0$ everywhere.
        \item $\int_S \vec{F}\cdot d\vec{a}$  is independent of surface, for any given boundary line.
        \item $\oint_S \vec{F}\cdot d\vec{a} = 0$ for any closed surface $S$.
        \item $\vec{F} = \vec{\nabla}\times\vec{A}$ for some vector field $\vec{A}$.
    \end{enumerate}
\end{theorem}
\begin{remark}
    In electrostatics we will find $\vec{\nabla}\times\vec{E} = 0$, which licenses $\vec{E} = -\vec{\nabla}V$ and the electric potential. In magnetostatics we will find $\vec{\nabla}\cdot\vec{B} = 0$, which licenses $\vec{B} = \vec{\nabla}\times\vec{A}$, the vector potential.
\end{remark}
\begin{theorem}[Helmholtz]
    If $\vec{\nabla}\cdot\vec{F}$ and $\vec{\nabla}\times\vec{F}$ are specified everywhere and $\vec{F}\to 0$ at infinity, then $\vec{F}$ is uniquely determined. It can be written as $\vec{F} = \vec{\nabla}\psi + \vec{\nabla}\times\vec{A}$, an irrotational part plus a solenoidal part.
\end{theorem}


\section{Electrostatics}
\marginpar{September 11-16 \& 21-25, 2026}
\begin{definition}[Electrostatics]
    Electrostatics is the study of stationary electric charges.
\end{definition}

\subsection{Electric Field}
Goal of electrodynamics: given source charges with known positions, find the force on a test charge $Q$.
\begin{proposition}[Principle of Superposition]
    \label{Superposition}
    The force between two charges is unaffected by other charges, so
    \[\vec{F} = \sum_i \vec{F}_i.\]
\end{proposition}
\begin{law}[Coulomb's law]
    The force on a test charge $Q$ due to a point charge $q$ is
    \begin{equation}
        \vec{F} = \frac{1}{4\pi\epsilon_0}\frac{qQ}{\mathfrak{r}^2}\,\hat{\mathfrak{r}} \left[N\right],
    \end{equation}
    where $\vec{\mathfrak{r}} = \vec{r} - \vec{r'}$ is the separation vector from the location of $q$ ($\vec{r'}$) to $Q$ ($\vec{r}$).
\end{law}
\begin{remark}
    Coulomb's law has the same shape as Newton's gravity. The reason is geometric rather than a coincidence, and Gauss' law makes that explicit: in $d$ spatial dimensions the flux spreads over a $(d-1)$-dimensional sphere, so $F\propto 1/\mathfrak{r}^{d-1}$, and the $4\pi$ is the area of the unit sphere in $d=3$.
\end{remark}
\begin{definition}[Permittivity of free space]
    The permittivity of free space measures how easily an electric field can form in a classical vacuum:
    \[\epsilon_0 = 8.85 \times 10^{-12}\ \left[\frac{C^2}{N\cdot m^2}\right].\]
\end{definition}
By the principle of superposition, the total force on a test charge $Q$ due to several point charges is
\[\vec{F} = \frac{Q}{4\pi\epsilon_0}\sum_{i=1}^{n}\frac{q_i}{\mathfrak{r}_i^2}\,\hat{\mathfrak{r}}_i.\]
Which leads us to the definition of the electric field
\[\vec{F} = Q\vec{E}.\]
\begin{definition}[Electric field]
    The electric field is the force per unit test charge.
    \begin{equation}
        \vec{E}(\vec{r}) = \frac{1}{4\pi\epsilon_0}\sum_{i=1}^{n}\frac{q_i}{\mathfrak{r}_i^2}\,\hat{\mathfrak{r}}_i.
    \end{equation}
    $\vec{\mathfrak{r}}_i \equiv \vec{r} - \vec{r}_i$, where $\vec{r}$ is the location of the field point, where we want to know what force a charge would feel, and $\vec{r}_i$ is the location of the source charge $q_i$.
\end{definition}

\subsubsection{Continuous Charge Distributions}
\begin{definition}[Continuous distributions]
    For a charge along a line, over a surface, or through a volume, replace $q_i$ by $dq = \lambda\,dl' = \sigma\,da' = \rho\,d\tau'$ and the sum becomes an integral:
    \begin{equation}
        \vec{E}(\vec{r}) = \frac{1}{4\pi\epsilon_0}\int\frac{dq}{\mathfrak{r}^2}\,\hat{\mathfrak{r}},
        \qquad \vec{\mathfrak{r}} = \vec{r} - \vec{r}',
    \end{equation}
    where $\vec{r}$ is the field point and $\vec{r}'$ the source point.
\end{definition}
\begin{example}[Line of charge]
    \label{ex:line-charge}
    A straight segment of length $2L$, centred at the origin on the $x$ axis, carries a uniform line charge density $\lambda$. Find $\vec{E}$ at the point $P = (0,0,z)$, a height $z$ above its midpoint.
        \begin{figure}[H]
        \centering
        \begin{tikzpicture}[scale=1.1, >=stealth]
            % axes
            \draw[->] (-3.2,0) -- (3.4,0) node[right] {$\hat{x}$};
            \draw[->] (0,-0.3) -- (0,3.2) node[above] {$\hat{z}$};
            % line of charge

    \end{figure}
    With $\vec{r} = z\hat{z}$ and $\vec{r}' = x'\hat{x}$,
    \[\vec{\mathfrak{r}} = z\hat{z} - x'\hat{x}, \qquad\mathfrak{r} = \sqrt{z^2 + x'^2},\]
    so
    \[\vec{E} = \frac{\lambda}{4\pi\epsilon_0}\int_{-L}^{L}\frac{z\hat{z} - x'\hat{x}}{(z^2 + x'^2)^{3/2}}\,dx' = \frac{\lambda}{4\pi\epsilon_0}\left[\hat{z}\int_{-L}^{L}\frac{z}{(z^2+x'^2)^{3/2}}dx' - \hat{x}\int_{-L}^{L}\frac{x'}{(z^2+x'^2)^{3/2}}dx'\right].\]
    The second integral vanishes by symmetry: its integrand is odd in $x'$, so the pieces at $x'$ and $-x'$ cancel. The first is done with the substitution $x' = z\tan\theta$, giving
    \[\vec{E} = \frac{\lambda}{4\pi\epsilon_0}\,\frac{1}{z}\left[\frac{x'}{\sqrt{x'^2+z^2}}\right]_{-L}^{L}\hat{z} = \frac{\lambda}{4\pi\epsilon_0}\,\frac{2L}{z\sqrt{L^2+z^2}}\,\hat{z} = \frac{\lambda}{2\pi\epsilon_0 z}\cdot\frac{L}{\sqrt{L^2 + z^2}}\,\hat{z}.\]
    This has two limits:
    \begin{itemize}
        \item Close to the line ($z \ll L$), it looks like an infinite line: $\vec{E} \to \dfrac{\lambda}{2\pi\epsilon_0 z}\hat{z}$.
        \item Far from the line ($z \gg L$), it looks like a point charge: $\vec{E} \to \dfrac{Q}{4\pi\epsilon_0 z^2}\hat{z}$ with $Q = 2\lambda L$.
    \end{itemize}
\end{example}

\subsection{Divergence and Curl of Electrostatic Fields}

\subsubsection{Gauss's Law}
\begin{law}[Gauss' law]
    The flux of $\vec{E}$ through any closed surface is set by the charge inside it:
    \begin{equation}
        \oiint_S \vec{E}\cdot d\vec{a} = \frac{Q_{enc}}{\epsilon_0}.
    \end{equation}
    Here $Q_{enc}$ is the total charge of the object inside the gaussian surface $S$.
\end{law}

\begin{proposition}[Differential form]
    Apply the divergence theorem (Theorem~\ref{FTD}) to the left side and write $Q_{enc} = \iiint_V\rho\,d\tau$:
    \[\iiint_V(\vec{\nabla}\cdot\vec{E})\,d\tau = \frac{1}{\epsilon_0}\iiint_V \rho\,d\tau,\]
    where $\rho$ is the charge density. Since this holds for every volume $V$, the integrands match:
    \begin{equation}
        \vec{\nabla}\cdot\vec{E} = \frac{\rho}{\epsilon_0}.
    \end{equation}
\end{proposition}
\begin{definition}[Gaussian Surface]
    The Gaussian surface just means an imaginary surface (nothing is physically there).
\end{definition}
\begin{remark}
    If there is no charge inside the Gaussian surface, the net flux is zero: every field line that enters on one side leaves on another.
\end{remark}

\subsubsection{Applications of Gauss's Law}
Gauss's law always holds but is only useful with symmetry: we need $\vec{E}$ parallel to $d\vec{a}$ and $|\vec{E}|$ constant over the Gaussian surface to pull $|\vec{E}|$ out of the integral. Only three symmetries work: spherical, cylindrical, and planar.
\paragraph{Spherical Symmetry.}
\begin{example}[Uniform solid sphere]
    \label{ex:solid-sphere}
    Let a uniform solid sphere of radius $R$ and total charge $q$. Imagine a gaussian sphere of radius $r$, so
    \[\vec{E}\oiint d\vec{a} = \frac{Q_{enc}}{\epsilon_0},\]
    let $Q_{enc} = q\,(r^3/R^3)$ for $r < R$ and $Q_{enc} = q$ for $r > R$. The flux is $\oiint d\vec{a} = 4\pi r^2$, so
    \[\vec{E} = \begin{cases} \dfrac{q\,r}{4\pi\epsilon_0 R^3}\hat{r} & r < R \\[8pt] \frac{q}{4\pi\epsilon_0 R^2}\hat{r} & r = R \\[8pt] \dfrac{q}{4\pi\epsilon_0 r^2}\hat{r} & r > R \end{cases}.\]
\end{example}
\begin{example}
    A thin shell instead gives
    \[\vec{E} = \begin{cases} 0 & r < R \\[8pt] \frac{\sigma}{\epsilon_0}\hat{r} & r = R \\[8pt] \dfrac{q}{4\pi\epsilon_0 r^2}\hat{r} & r > R \end{cases}.\]
\end{example}
\begin{remark}
    A uniformly charged shell that pulses in and out, $R(t) = R_0 + A\sin\omega t$, keeps the constant field $\frac{Q}{4\pi\epsilon_0 r^2}\hat{r}$ outside: accelerating charges, but nothing escapes to infinity, so no radiation.
\end{remark}
\paragraph{Cylindrical Symmetry.}
\begin{example}[Infinite line of charge]
    For a line with charge $\lambda$ per unit length, take the Gaussian surface to be a coaxial cylinder of radius $z$ (the distance from the line to the field point, not the radius of the line) and length $d$. $\vec{E}$ points radially outward, so $\vec{E}\cdot d\vec{a} = 0$ on the end caps and only the curved side contributes:
    \[E\,(2\pi z d) = \frac{\lambda d}{\epsilon_0} \implies E = \frac{\lambda}{2\pi\epsilon_0 z}.\]
\end{example}
\begin{example}[Finite versus infinite line]
    Near its midpoint, a finite line has a weaker field than an infinite one: the same enclosed charge, but some flux leaks through the caps.
\end{example}
\begin{problem}[Midterm 2025, Q2]
    An infinite non-conducting tube along the $x$ axis has $\rho(s) = A\,s/a$ for $s < a$ and $0$ elsewhere, with $s$ the distance from the axis. Find $\vec{E}$ everywhere.
\end{problem}
\paragraph{Planar Symmetry.}
\begin{example}[Infinite plane of charge]
    For a plane with surface charge density $\sigma$, take the Gaussian surface to be a pillbox straddling the plane, with caps of area $A$. $\vec{E}$ points away from the plane on both sides, so there is no flux through the side wall and $EA$ through each cap:
    \[2EA = \frac{\sigma A}{\epsilon_0} \implies E = \frac{\sigma}{2\epsilon_0}.\]
    \begin{figure}[H]
        \centering
        \begin{tikzpicture}[scale=1, >=stealth]
            \fill[gray!25] (-3,-0.07) rectangle (3,0.07);
            \node[right] at (3,0) {plane, $\sigma$};
            \draw[thick] (-0.8,-0.8) rectangle (0.8,0.8);
            \draw[->, red, thick] (0,0.8) -- (0,1.6) node[above] {$\vec{E}$};
            \draw[->, red, thick] (0,-0.8) -- (0,-1.6) node[below] {$\vec{E}$};
            \node[left] at (-0.8,0.8) {cap, area $A$};
            \node[left] at (-0.8,-0.8) {cap, area $A$};
            \node[right] at (0.8,0.45) {side wall};
        \end{tikzpicture}
        \caption{Side view of the pillbox straddling the plane.}
    \end{figure}
\end{example}
\begin{example}[Two planes]
    Superpose two sheets a distance $d$ apart, each giving $\sigma/2\epsilon_0$, pointing away from a positive sheet and toward a negative one.
    \begin{itemize}
        \item $+\sigma$ and $-\sigma$ (parallel plates):
            \[E = \begin{cases}\frac{\sigma}{\epsilon_0} & \text{between the plates}, \\ 0 & \text{outside the plates}.\end{cases}\]
        \item $+\sigma$ and $+\sigma$: 
            \[E = \begin{cases}0 & \text{between the plates}, \\ \frac{\sigma}{\epsilon_0} & \text{outside the plates}.\end{cases}\]
    \end{itemize}
\end{example}
\begin{remark}[Fall-off]
    The fall-off is how fast $E$ decreases with the distance $r$ from the source: $E \propto 1/r^2$ for a point charge, $1/r$ for an infinite line, and $r^0$ (uniform) for an infinite plane. To see why, picture a narrow cone opening from the field point toward the source. Its cross section at distance $r$ grows like $r^2$, which alone gives the $1/r^2$ of a point charge, but for extended sources the charge inside the cone grows too: it intercepts a length $\propto r$ of a line, giving $E \propto r/r^2 = 1/r$, and an area $\propto r^2$ of a sheet, giving $E \propto r^2/r^2$, a constant.
\end{remark}
\begin{remark}
    For two equal positive charges, no field lines cross the midplane: the field there is parallel to the plane and vanishes at the midpoint. Gauss's law still holds, but without symmetry it cannot solve for $\vec{E}$.
\end{remark}

\subsubsection{The Curl of $\vec{E}$}
\begin{theorem}[Electrostatic fields are curl-free]
    For any electrostatic field,
    \begin{equation}
        \oint\vec{E}\cdot d\vec{l} = 0, \qquad \vec{\nabla}\times\vec{E} = 0.
    \end{equation}
\end{theorem}
\begin{proof}
    Let there be a point charge at the origin. The field is radial, $\vec{E} = \frac{q}{4\pi\epsilon_0 r^2}\hat{r}$, so the line integral from some point a to some other point b in three dimensions $\int_a^b \vec{E}\cdot d\vec{l}$, in spherical coordinates $d\vec{l} = dr\,\hat{r} + r\,d\theta\,\hat{\theta} + r\sin\theta\,d\phi\,\hat{\phi}$, we get
    \[\vec{E}\cdot d\vec{l} = \frac{q}{4\pi\epsilon_0 r^2}dr.\]
    Therefore,
    \[\int_a^b \vec{E}\cdot d\vec{l} = \frac{q}{4\pi\epsilon_0}\int_{r_a}^{r_b}\frac{dr}{r^2} = \frac{q}{4\pi\epsilon_0}\left(\frac{1}{r_a} - \frac{1}{r_b}\right).\]
    The integral around a closed path is evidently zero (for then $r_a=r_b$):
    \[\oint\vec{E}\cdot d\vec{l} = 0,\]
    and hence, by Stokes' theorem \eqref{Stokes}, $\vec{\nabla}\times\vec{E} = 0$.
\end{proof}

\subsection{Electric Potential}
\begin{definition}[Electric potential]
    Since $\vec{\nabla}\times\vec{E} = 0$, the irrotational fields theorem \ref{irrotational} guarantees a scalar $V$ with
    \begin{equation}
        \vec{E} = -\vec{\nabla}V.
    \end{equation}
    The minus sign is convention: positive charges are pushed toward lower potential energy.
\end{definition}
\begin{proposition}[Potential from the field]
    \begin{equation}
        V(\vec{r}) = -\int_{\text{ref}}^{\vec{r}}\vec{E}\cdot d\vec{l} \left[\frac{J}{C} = \text{volts (v)}\right].
    \end{equation}
    The path does not matter since the integral is path-independent\footnote{Changing the reference point adds a constant to $V$, which leaves $\vec{E} = -\vec{\nabla}V$ unchanged (gauge freedom). Statement (ii) of the irrotational fields theorem.}. By convention the reference is at infinity, unless the charge distribution extends to infinity.
\end{proposition}
\begin{proposition}[Potential of a Charge Distribution Extending to Infinity]
    If the charge distribution extends to infinity, the reference point for the potential is typically taken at a finite distance where the potential vanishes, for example a point on the plane.
\end{proposition}
\begin{definition}[Equipotential]
    An equipotential is a region, typically a surface, on which $V$ is constant.
\end{definition}
\begin{example}[Point charge]
    Along a radial path,
    \[V(r) = -\frac{q}{4\pi\epsilon_0}\int_\infty^r\frac{dr'}{r'^2} = \frac{q}{4\pi\epsilon_0 r}.\]
\end{example}
\begin{example}[Uniform solid sphere]
    Using the field of Example~\ref{ex:solid-sphere}, outside $V = \frac{q}{4\pi\epsilon_0 r}$. Inside, integrate in from infinity through both regions:
    \[V(r) = -\int_\infty^R\frac{q\,dr'}{4\pi\epsilon_0 r'^2} - \int_R^r\frac{q\,r'\,dr'}{4\pi\epsilon_0 R^3} = \frac{q}{8\pi\epsilon_0 R}\left(3 - \frac{r^2}{R^2}\right).\]
\end{example}
\begin{problem}[Midterm drills]
    Find $V$ (i) at the centre of a ring of radius $a$ and charge $Q$; (ii) at $x = 2a$ for a line of charge $Q$ from $x = -a$ to $x = a$.
\end{problem}

\subsubsection{Poisson's Equation and Laplace's Equation}
\begin{proposition}[Poisson's equation]
    Combining $\vec{\nabla}\cdot\vec{E} = \rho/\epsilon_0$ with $\vec{E} = -\vec{\nabla}V$,
    \begin{equation}
        \nabla^2 V = -\frac{\rho}{\epsilon_0}.
    \end{equation}
\end{proposition}
\begin{corollary}[Laplace's equation]
    In regions where there is no charge, so $\rho = 0$, Poisson's equation reduces to Laplace's equation:
    \begin{equation}
        \nabla^2 V = 0.
    \end{equation}
\end{corollary}

\subsubsection{The Potential of a Localized Charge Distribution}
\begin{proposition}[Potential of a distribution]
    By superposition \eqref{Superposition},
    \begin{equation}
        V(\vec{r}) = \frac{1}{4\pi\epsilon_0}\int\frac{\rho(\vec{r}')}{\mathfrak{r}}\,d\tau',
    \end{equation}
    where $\rho(\vec{r}')d\tau'=dq$. Unlike the integral for $\vec{E}$, there is no $\hat{\mathfrak{r}}$, so computing $V$ first and then $\vec{E} = -\vec{\nabla}V$ is often easier.
\end{proposition}

\subsubsection{Boundary Conditions}
A surface charge $\sigma$ makes $\vec{E}$ jump as you cross it. To fix this we use Gauss's law.

\begin{proposition}[Normal component]
    Take a pillbox of cap area $A$ straddling the surface and shrink its thickness $\varepsilon \to 0$, so the side wall carries no flux. Only the caps count, and the charge inside is $\sigma A$:
    \[E^{\perp}_{\text{above}}A - E^{\perp}_{\text{below}}A = \frac{\sigma A}{\epsilon_0} \quad\Longrightarrow\quad E^{\perp}_{\text{above}} - E^{\perp}_{\text{below}} = \frac{\sigma}{\epsilon_0}.\]
\end{proposition}
\begin{proposition}[Tangential component]
    Take a thin rectangular loop of length $l$ straddling the surface, again with $\varepsilon \to 0$ so the short sides contribute nothing. Since $\oint\vec{E}\cdot d\vec{l} = 0$,
    \[E^{\parallel}_{\text{above}}\,l - E^{\parallel}_{\text{below}}\,l = 0 \quad\Longrightarrow\quad \vec{E}^{\parallel}_{\text{above}} = \vec{E}^{\parallel}_{\text{below}}.\]
\end{proposition}
\begin{theorem}[Boundary condition on $\vec{E}$]
    \label{BC-E}
    With $\hat{n}$ the unit normal pointing from below to above, the two results combine into
    \begin{equation}
        \vec{E}_{\text{above}} - \vec{E}_{\text{below}} = \frac{\sigma}{\epsilon_0}\hat{n}.
    \end{equation}
    The normal component jumps by $\sigma/\epsilon_0$; the tangential component is continuous.
\end{theorem}
\begin{remark}
    Check with the infinite plane: $\vec{E} = +\frac{\sigma}{2\epsilon_0}\hat{n}$ above and $-\frac{\sigma}{2\epsilon_0}\hat{n}$ below, a jump of exactly $\sigma/\epsilon_0$.
\end{remark}
\begin{theorem}[Boundary conditions on $V$]
    The potential is continuous across any boundary, since
    \[V_{\text{above}} - V_{\text{below}} = -\int_{\vec{a}}^{\vec{b}}\vec{E}\cdot d\vec{l} \overset{\text{ path length}\to 0}{\implies} V_{\text{above}} = V_{\text{below}}.\]
    However, the gradient of $V$ is not continuous: substituting $\vec{E} = -\vec{\nabla}V$ into Theorem~\ref{BC-E} gives
    \begin{equation}
        \vec{\nabla}V_{\text{above}} - \vec{\nabla}V_{\text{below}} = -\frac{\sigma}{\epsilon_0}\hat{n}.
    \end{equation}
\end{theorem}

\subsection{Work and Energy in Electrostatics}
\begin{proposition}[Potential of a test charge]
    Bringing $Q$ in from infinity against the field takes
    \begin{equation}
        W = -Q\int_\infty^{\vec{r}}\vec{E}\cdot d\vec{l} = QV(\vec{r}).
    \end{equation}
\end{proposition}
\subsubsection{The Energy of a Point Charge Distribution}
\begin{proposition}[Energy of $N$ point charges]
    The first charge costs no work, since no field exists yet, and each later one feels the potential $V(\vec{r}_i)$ of those already placed, giving one term per pair:
    \begin{equation}
        W = \frac{1}{2}\sum_{i=1}^{n}q_i\left(\sum_{j\neq i}^n\frac{q_j}{4\pi\epsilon_0|\vec{r}_i - \vec{r}_j|}\right) = \frac{1}{2}\sum_{i=1}^{n}q_iV(\vec{r}_i),
    \end{equation}
    where the $\tfrac12$ corrects for counting each pair twice.
\end{proposition}

\subsubsection{The Energy of a Continuous Charge Distribution}
\begin{theorem}[Energy stored in the field]
    \label{E-stored-in-field}
    For a volume charge distribution, the energy is
    \begin{equation}
        W = \frac{1}{2}\int\rho V\,d\tau = \frac{\epsilon_0}{2}\int_{\text{all space}} E^2\,d\tau.
    \end{equation}
\end{theorem}

\subsubsection{Comments on Electrostatic Energy}
\begin{definition}[Energy Density]
    From Theorem~\ref{E-stored-in-field}, the energy per unit volume stored in the field is
    \begin{equation}
        u = \frac{\epsilon_0}{2}E^2.
    \end{equation}
    The proof is in Appendix~\ref{app:proofs}.
\end{definition}
\begin{problem}[Midterm drill]
    Two plates of area $A$, a distance $d$ apart, carry $+\sigma$ and $-\sigma$ ($A \gg d^2$). Find the energy.
\end{problem}
\begin{problem}[Problem Set 4, Q4]
    Concentric shells of radii $a < b$ carry $+q$ and $-q$. Find the energy.
\end{problem}

\subsection{Conductors}
\begin{definition}[Insulator]
    In an insulator, such as glass or ruber, each electron is on a short leash, attached to a particular atom.
\end{definition}
\begin{definition}[Conductor]
    A conductor is a material with charges that are free to move (in a metal, the electrons). 
\end{definition}
\begin{proposition}
    In an external field, a conductor's free charges move to its surface until their field $\vec{E}_{\text{int}}$ cancels $\vec{E}_{\text{ext}}$ inside. In practice, it's the negative charges (the electrons) that do the moving, leaving a positive charge on the far side.
\end{proposition}

        \foreach \y in {0.5,1.5,2.5}{

            \node[red] at (2.75,\y) {$+$};
        }

        \node[violet, below] at (1.5,1) {$\vec{E}_{\text{int}}$};
        \node at (1.5,2) {$\vec{E} = 0$};
    \end{tikzpicture}
    \caption{A conductor in an external field.}
\end{figure}
\begin{proposition}[Properties of a conductor in equilibrium]\leavevmode
    \begin{enumerate}[label=(\roman*)]
        \item $\vec{E} = 0$ inside, otherwise the free charges would still be moving.
        \item $\rho = 0$ inside, since $\rho = \epsilon_0\vec{\nabla}\cdot\vec{E} = \epsilon_0 \cdot 0 = 0$.
        \item Any net charge therefore sits on the surface.
        \item A conductor is an equipotential: for any two points within (or at the surface of) a conductor, $V(\vec{b}) - V(\vec{a}) = -\int_{\vec{a}}^{\vec{b}}\vec{E}\cdot d\vec{l} = 0$.
        \item Just outside, $\vec{E}$ is perpendicular to the surface. A tangential part would push the surface charges along the surface, and they would not be in equilibrium.
    \end{enumerate}
\end{proposition}
\begin{remark}
    Grounding a conductor connects it to $V(\infty) = 0$, so its whole body, not just its surface, sits at $V = 0$. Positive charge flows to the ground if the conductor would otherwise be at $V > 0$, or from the ground if it would be at $V < 0$, until $V = 0$.
\end{remark}

\subsubsection{Induced Charges}
\begin{remark}[Induced Charges]
    If a conductor has a cavity with a charge $q$ inside, $q$ induces $-q$ on the cavity wall and $+q$ on the outer surface (for a neutral conductor), and the outside cannot tell how the charge inside is arranged. In particular, for a sphere the outer charge spreads uniformly even if the cavity is off centre, so the field outside is that of a point charge at the centre of the sphere, not of the cavity.
\end{remark}
\begin{remark}[Induced Charges with External Field]
    Same setup as the previous remark, but now add an external field: the charge on the cavity wall is still $-q$ and the outer surface still carries $+q$, but the external field induces a nonuniform distribution on the outer surface. The field inside the cavity is unchanged, since it is due only to $q$ and the wall charge.

\begin{example}[Neutral conductor near a charged rod]
    A neutral metal block sits near an insulating rod with uniform positive charge. Negative charge is induced on the side facing the rod and an equal positive charge on the far side. The negative charges are closer, and Coulomb's force falls off with distance, so the attraction beats the repulsion: the neutral conductor is pulled toward the rod.
\end{example}
\begin{example}[Conducting versus insulating disk]
    For a disk with uniform $\sigma$ (an insulator), the disk is not an equipotential and the field lines are not perpendicular to it. That arrangement cannot last on a conductor: the field along the surface pushes charge outward, i.e.\ ``downhill'' in potential, and it piles up at the rim. So a charged conducting disk has a higher $\sigma$ near its edge than near its centre, its surface is an equipotential, and its field lines leave it at right angles.
\end{example}
\begin{remark}
    In general, charge on a conductor concentrates where the surface is most sharply curved (edges, corners, points). Since $E = \sigma/\epsilon_0$ there, the field is strongest at sharp points, which is how lightning rods work.
\end{remark}

\begin{example}[Concentric conducting shells]
    A metal sphere of radius $R$ with charge $q$ sits inside a thick neutral metal shell of inner radius $a$ and outer radius $b$.
    \begin{enumerate}[label=(\roman*)]
        \item A Gaussian sphere inside the shell's metal ($a < r < b$) has $\vec{E} = 0$ on it, so it encloses zero charge: the inner surface carries $-q$. The shell is neutral, so the outer surface carries $+q$. By symmetry each is uniform:
        \[\sigma_R = \frac{q}{4\pi R^2}, \qquad \sigma_a = -\frac{q}{4\pi a^2}, \qquad \sigma_b = \frac{q}{4\pi b^2}.\]
        \item The field is $\frac{q}{4\pi\epsilon_0 r^2}\hat{r}$ for $r > b$ and $R < r < a$, and zero in both metals. Integrating in from infinity,
        \[V(0) = -\int_\infty^b\frac{q\,dr}{4\pi\epsilon_0 r^2} - \int_a^R\frac{q\,dr}{4\pi\epsilon_0 r^2} = \frac{q}{4\pi\epsilon_0}\left(\frac{1}{b} + \frac{1}{R} - \frac{1}{a}\right).\]
        \item Grounding the outer surface sets $V(b) = 0 = V(\infty)$, so there can be no field outside and $\sigma_b = 0$. The Gauss argument in (i) is unchanged, so $\sigma_R$ and $\sigma_a$ stay the same, and the first integral in (ii) disappears: $V(0) = \frac{q}{4\pi\epsilon_0}\left(\frac{1}{R} - \frac{1}{a}\right)$.
        \item The outer surface now has zero charge; the $+q$ flowed away to the ground, leaving the shell with net charge $-q$. If the outer surface kept a charge $Q'$, the total charge inside a sphere just outside $b$ would be $Q'$, giving $V(b) = \frac{Q'}{4\pi\epsilon_0 b} \neq 0$.
    \end{enumerate}
\end{example}

\subsection{Laplace's Equation}
We want $V$ because $\vec{E} = -\vec{\nabla}V$. The integral $V = \frac{1}{4\pi\epsilon_0}\int\frac{\rho}{\mathfrak{r}}\,d\tau'$ is exact but often impossible to do, and the multipole expansion is only an approximation. Often we do not even know $\rho$ everywhere: we know the potentials of some conductors instead, and the charge on them is whatever it has to be. In the empty space between the charges, Poisson's equation simplifies.
\begin{law}[Laplace's equation]
    In a region with no charge,
    \begin{equation}
        \nabla^2 V = \frac{\partial^2 V}{\partial x^2} + \frac{\partial^2 V}{\partial y^2} + \frac{\partial^2 V}{\partial z^2} = 0.
    \end{equation}
\end{law}
\begin{remark}
    Point charges, charged objects and conductors are where $\nabla^2 V = -\rho/\epsilon_0$ holds with $\rho \neq 0$. Everywhere else $\nabla^2 V = 0$, and the charges only enter as boundary conditions on that empty region.
\end{remark}
\begin{remark}[The Laplacian compares a point with its surroundings]
    In one dimension, $V'' = 0$ gives a straight line, $V = mx + b$, and then $V(x) = \frac{1}{2}\left[V(x+a) + V(x-a)\right]$: every point is the average of its neighbours. In general $\nabla^2 V$ measures how far a point sits from the average of its surroundings, and since $\nabla^2 V = -\rho/\epsilon_0$, that tells you the sign of the charge there.
    \begin{figure}[H]
        \centering
        \begin{tikzpicture}[scale=1.1, >=stealth]
            \begin{scope}
                \draw[dashed, gray] (0,1) -- (2,1);
                \draw[thick, blue] (0,1) parabola bend (1,0.3) (2,1);
                \draw[red, thick] (1,1) -- (1,0.3);
                \draw[red, thick] (1,1) circle (2.5pt);
                \foreach \p in {(0,1),(2,1),(1,0.3)} \fill[blue] \p circle (2pt);
                \node[align=center] at (1,-0.5) {below the average\\$\nabla^2 V > 0$, so $\rho < 0$};
            \end{scope}
            \begin{scope}[xshift=3.5cm]
                \draw[thick, blue] (0,0.3) -- (2,1.3);
                \draw[green!50!black, thick] (1,0.8) circle (3.5pt);
                \foreach \p in {(0,0.3),(2,1.3),(1,0.8)} \fill[blue] \p circle (2pt);
                \node[align=center] at (1,-0.5) {equal to the average\\$\nabla^2 V = 0$, no charge};
            \end{scope}
            \begin{scope}[xshift=7cm]

\begin{theorem}[Mean value property]
    If $\nabla^2 V = 0$, the value of $V$ at any point equals its average over any sphere of radius $R$ centred on that point (containing no charge):
    \begin{equation}
        V(\vec{r}_0) = \frac{1}{4\pi R^2}\oint_{\text{sphere}} V\,da.
    \end{equation}
\end{theorem}
\begin{corollary}[No local maxima or minima]
    A solution of Laplace's equation has no local maxima or minima inside the region; extreme values occur only on the boundary. A maximum would be larger than every point around it, so it could not equal their average.
\end{corollary}
\begin{theorem}[Earnshaw's theorem]
    A charged particle cannot be held in stable equilibrium by electrostatic forces alone.
\end{theorem}
\begin{proof}
    Stable equilibrium for a charge $q$ needs its potential energy $qV$ to have a local minimum at that point. But the other charges' $V$ obeys Laplace's equation there, so it has no local minimum. Concretely, if $\frac{\partial^2 V}{\partial x^2} > 0$ (stable along $x$), then $\frac{\partial^2 V}{\partial y^2} + \frac{\partial^2 V}{\partial z^2} < 0$, so the charge escapes sideways. For example, a charge between two equal charges is stable along the line joining them but slips out perpendicular to it.
\end{proof}
\begin{theorem}[Uniqueness theorem]
    The solution of Laplace's equation in a volume is uniquely determined if $V$ is specified on the whole boundary of that volume. Similarly, if the region contains conductors, $\vec{E}$ is uniquely determined once the charge density $\rho$ and the total charge on each conductor are given.
\end{theorem}
\begin{remark}
    The significance: any function that satisfies Laplace's equation and the boundary conditions is \emph{the} answer, no matter how it was found. Guessing is allowed, which is what the method of images and separation of variables exploit.
\end{remark}
\begin{proposition}[Method of relaxation]
    On a grid, the mean value property says each cell equals the average of its neighbours. In two dimensions,
    \[V_{i,j} = \frac{1}{4}\left(V_{i+1,j} + V_{i-1,j} + V_{i,j+1} + V_{i,j-1}\right).\]
    This gives a numerical method:
    \begin{enumerate}
        \item Initialize the grid with a guess for $V$.
        \item Sweep through the grid, replacing each value by the average of its nearest neighbours, keeping the boundary values fixed.
        \item Repeat until the values stop changing.
    \end{enumerate}
\end{proposition}

\subsection{The Method of Images}
\begin{remark}[A solution in search of a problem]
    Maxwell pointed out that the inverse problem is easier: given a potential, finding conductors that fit it is simpler than finding the potential for given conductors. Any equipotential surface of a known potential can be replaced by a conductor held at that potential, and by uniqueness the field outside it does not change.
\end{remark}
\begin{example}[Point charge above a grounded plane]
    A charge $q$ sits at height $d$ above an infinite grounded conducting plane at $z = 0$. For $z > 0$ we need $\nabla^2 V = 0$ (apart from at $q$), $V = 0$ on the plane, and $V \to 0$ far away. Forget the plane and put an image charge $-q$ at $z = -d$ instead. Every point of the plane is equidistant from the two charges, so this pair gives $V = 0$ there, and it has the same charge in the region $z > 0$. By uniqueness it is the answer for $z \geq 0$:
    \[V(x,y,z) = \frac{1}{4\pi\epsilon_0}\left[\frac{q}{\sqrt{x^2 + y^2 + (z-d)^2}} - \frac{q}{\sqrt{x^2 + y^2 + (z+d)^2}}\right].\]
    \begin{figure}[H]
        \centering
        \begin{tikzpicture}[>=stealth]
            \fill[gray!20] (-2,-1.6) rectangle (2,0);
            \draw[thick] (-2,0) -- (2,0);
            \node at (0,-0.8) {grounded conductor};
            \fill (0,1.2) circle (2pt) node[right] {$+q$};
            \draw[<->] (-0.4,0) -- (-0.4,1.2) node[midway, left] {$d$};
            \draw[<->, thick] (2.3,0.3) -- (3.2,0.3);
            \node[align=center] at (2.75,0.85) {\small same $V$\\\small for $z>0$};
            \begin{scope}[xshift=5.5cm]
                \draw[dashed] (-2,0) -- (2,0) node[right] {$V = 0$};
                \fill (0,1.2) circle (2pt) node[right] {$+q$};
                \fill[red] (0,-1.2) circle (2pt) node[right] {$-q$ (image)};
                \draw[<->] (-0.4,0) -- (-0.4,1.2) node[midway, left] {$d$};
                \draw[<->] (-0.4,-1.2) -- (-0.4,0) node[midway, left] {$d$};
            \end{scope}
        \end{tikzpicture}
        \caption{The real problem (left) and the image problem (right) have the same charge and the same boundary values in $z > 0$, so they have the same potential there.}
    \end{figure}
\end{example}
\begin{remark}
    The image charge is not real; it is bookkeeping. Physically, the field comes from the negative surface charge induced on the plane, and the image reproduces its effect in the region we care about. The formula says nothing about $z < 0$, where $V = 0$ inside the conductor. The image must sit outside the region being solved, otherwise it would change $\rho$ there.
\end{remark}
\begin{remark}
    The induced charge on the plane totals $-q$, and $q$ is attracted to the plane with the force of its image, but the energy is only half of the image pair's, since there is no field in $z < 0$.
\end{remark}
\begin{example}[Two charges above a grounded plane]
    Charges $+Q$ at height $d$ and $+3Q$ at height $2d$ sit above a grounded plane. Each charge on its own is handled by one image, so by superposition we need two: $-Q$ at $-d$ and $-3Q$ at $-2d$. The method works whenever the problem is linear (superposition) and the boundary values come out right (uniqueness).
\end{example}

\subsection{Separation of Variables: Cartesian}
\begin{proposition}[Separation]
    Look for solutions $V(x,y,z) = X(x)Y(y)Z(z)$. Substituting into $\nabla^2 V = 0$ and dividing by $V$,
    \[\frac{X''}{X} + \frac{Y''}{Y} + \frac{Z''}{Z} = 0.\]
    Each term depends on a different variable, which can be varied alone, so each must be a constant: $X'' = c_xX$, $Y'' = c_yY$, $Z'' = c_zZ$ with $c_x + c_y + c_z = 0$. One PDE becomes three ODEs.
\end{proposition}
\begin{remark}
    Most solutions are not separable (e.g.\ $1/r$), but Laplace's equation is linear, so sums of separable solutions can build general ones.
\end{remark}
\begin{proposition}[Three kinds of solution]
    For each direction,
    \[c = +k^2:\ X = Ae^{kx} + Be^{-kx}, \qquad c = -k^2:\ X = A\sin kx + B\cos kx, \qquad c = 0:\ X = A + Bx.\]
    Since the constants sum to zero, they cannot all have the same sign unless all vanish.
\end{proposition}
\begin{remark}[Which direction gets which sign]
    A sinusoid crosses zero infinitely often; an exponential (or $\cosh$, $\sinh$) at most once. So a direction in which $V$ must vanish at two places gets the sinusoid, and the other gets the exponential. Symmetry picks the parity: even $\Rightarrow$ $\cos$, $\cosh$; odd $\Rightarrow$ $\sin$, $\sinh$. $\cosh$ never vanishes; $\sinh$ vanishes only at $0$.
\end{remark}
\begin{example}[Square pipe]
    A long pipe along $z$ with square cross section: sides $x = \pm a/2$ grounded, top and bottom $y = \pm a/2$ held at $V_0$. Nothing depends on $z$, so $c_z = 0$ and $c_x = -c_y$. $V$ vanishes at two values of $x$, so $X$ is sinusoidal and $Y$ hyperbolic; the problem is even in both $x$ and $y$, so
    \[V = C\cos kx\,\cosh ky.\]
    $V(\pm a/2, y) = 0$ quantises $k$: $\cos(ka/2) = 0 \Rightarrow k = (2n+1)\pi/a$. Superposing,
    \[V(x,y) = \sum_{n=0}^\infty C_n\cos\frac{(2n+1)\pi x}{a}\cosh\frac{(2n+1)\pi y}{a}.\]
    Setting $y = a/2$, multiplying by $\cos\frac{(2m+1)\pi x}{a}$ and integrating over $x$ (Fourier's trick, using orthogonality $\int_{-a/2}^{a/2}\cos\cos\,dx = \frac{a}{2}\delta_{mn}$) picks out one term:
    \[V(x,y) = \frac{4V_0}{\pi}\sum_{n=0}^{\infty}\frac{(-1)^n}{2n+1}\cos\frac{(2n+1)\pi x}{a}\,\frac{\cosh\left[(2n+1)\pi y/a\right]}{\cosh\left[(2n+1)\pi/2\right]}.\]
    Check: rotating by $90^\circ$ swaps hot and grounded faces, so $V(0,0) = V_0/2$ exactly; the series gives $0.5074\,V_0$ with one term. The centre is a saddle, as $\nabla^2 V = 0$ requires. Near the corners the series overshoots (Gibbs phenomenon), because the boundary data jumps there.
\end{example}
\begin{remark}[Summary]
    \begin{enumerate}
        \item Separate: one PDE becomes ODEs with constants summing to zero.
        \item Choose signs: two or more zeros in a direction $\Rightarrow$ sinusoid; otherwise exponential/hyperbolic.
        \item Use symmetry to keep only even or odd functions.
        \item Impose the $V = 0$ boundaries: these quantise $k$.
        \item Superpose and use Fourier's trick for the coefficients.
        \item Uniqueness guarantees the result is the answer.
    \end{enumerate}
\end{remark}

\subsection{Separation of Variables: Spherical}
\begin{proposition}[Axially symmetric solutions]
    If nothing depends on $\phi$, separating $V(r,\theta) = R(r)\Theta(\theta)$ in the spherical Laplacian gives radial solutions $r^l$ and $r^{-(l+1)}$ and angular solutions the Legendre polynomials $P_l(\cos\theta)$, so
    \begin{equation}
        V(r,\theta) = \sum_{l=0}^{\infty}\left(A_lr^l + \frac{B_l}{r^{l+1}}\right)P_l(\cos\theta),
    \end{equation}
    with $P_0 = 1$, $P_1 = x$, $P_2 = (3x^2-1)/2$, and orthogonality $\int_0^\pi P_mP_n\sin\theta\,d\theta = \frac{2}{2m+1}\delta_{mn}$.
\end{proposition}
\begin{proposition}[Sphere with given surface potential $V_0(\theta)$]
    \begin{itemize}
        \item Inside, $V$ must be finite at $r = 0$, so $B_l = 0$.
        \item Outside, $V \to 0$ at infinity, so $A_l = 0$.
        \item Write $V_0(\theta) = \sum_l C_lP_l(\cos\theta)$ (by inspection if it is a simple polynomial in $\cos\theta$, else by orthogonality). Continuity at $R$ then gives
        \[V_{\text{in}} = \sum_l C_l\left(\frac{r}{R}\right)^lP_l(\cos\theta), \qquad V_{\text{out}} = \sum_l C_l\left(\frac{R}{r}\right)^{l+1}P_l(\cos\theta).\]
        \item The surface charge comes from the jump in slope, $\sigma = -\epsilon_0\left(\frac{\partial V_{\text{out}}}{\partial r} - \frac{\partial V_{\text{in}}}{\partial r}\right)_{r=R}$, giving
        \[\sigma(\theta) = \frac{\epsilon_0}{R}\sum_l(2l+1)C_lP_l(\cos\theta).\]
    \end{itemize}
\end{proposition}
\begin{remark}
    Rewrite powers of $\cos\theta$ in Legendre polynomials, e.g.\ $\cos^2\theta = \frac{2}{3}P_2 + \frac{1}{3}P_0$. Each $l$ outside is a multipole: $l = 0$ gives $V \propto 1/r$ with total charge $Q = 4\pi\epsilon_0 RC_0$.
\end{remark}
\begin{problem}[Midterm 2025, Q5]
    A spherical shell of radius $R$ has $V(R,\theta) = V_0(1 + \cos\theta)$, no charge elsewhere, and $V(\infty) = 0$. Find $V$ outside, $V$ inside, and $\sigma(\theta)$.
\end{problem}
\begin{solution}
    $V_0(1+\cos\theta) = V_0P_0 + V_0P_1$, so $C_0 = C_1 = V_0$ and all others vanish:
    \[V_{\text{out}} = V_0\left(\frac{R}{r} + \frac{R^2}{r^2}\cos\theta\right), \qquad V_{\text{in}} = V_0\left(1 + \frac{r}{R}\cos\theta\right),\]
    \[\sigma(\theta) = \frac{\epsilon_0V_0}{R}\left(1 + 3\cos\theta\right).\]
    Check: $V_{\text{in}} = V_0 + V_0z/R$ is a uniform field inside, and $V_{\text{out}}$ is a monopole plus a dipole.
\end{solution}

\subsection{Multipole Expansion}
\begin{example}[Two charges on the $z$ axis]
    Place $q_1$ at $z = d/2$ and $q_2$ at $z = -d/2$, and measure $\theta$ from the $z$ axis.

    Then
    \[V(\vec{r}) = \frac{1}{4\pi\epsilon_0}\left(\frac{q_1}{\mathfrak{r}_1} + \frac{q_2}{\mathfrak{r}_2}\right).\]
    To get $\mathfrak{r}_1$, write $\vec{\mathfrak{r}}_1 = \vec{r} - \frac{d}{2}\hat{z}$ and square it, using $\vec{r}\cdot\hat{z} = r\cos\theta$ (this is just the law of cosines):
    \[\mathfrak{r}_1^2 = r^2 + \left(\frac{d}{2}\right)^2 - 2r\frac{d}{2}\cos\theta = r^2 + \left(\frac{d}{2}\right)^2 - rd\cos\theta.\]
    For $q_2$ the source sits at $-\frac{d}{2}\hat{z}$, so the cross term flips sign. Both cases in one line, top sign for $q_1$ and bottom sign for $q_2$:
    \[\mathfrak{r}_{1,2}^2 = r^2 + \left(\frac{d}{2}\right)^2 \mp rd\cos\theta.\]
    Writing $1/\mathfrak{r}_{1,2} = \frac{1}{r}(1+\epsilon_\mp)^{-1/2}$ with $\epsilon_\mp = \frac{d}{2r}\left(\frac{d}{2r} \mp 2\cos\theta\right)$ and expanding for $d \ll r$, $(1+\epsilon)^{-1/2} = 1 - \frac{1}{2}\epsilon + \frac{3}{8}\epsilon^2 - \dots$, gives
    \[V(\vec{r}) = \frac{1}{4\pi\epsilon_0}\left[\underbrace{\frac{q_1+q_2}{r}}_{\text{monopole}} + \underbrace{\frac{(q_1-q_2)d\cos\theta}{2r^2}}_{\text{dipole}} + \underbrace{\frac{(q_1+q_2)d^2}{4}\frac{3\cos^2\theta - 1}{2r^3}}_{\text{quadrupole}} + \dots\right].\]
    Each term gains a power of $1/r$, so $E \sim 1/r^2,\ 1/r^3,\ 1/r^4, \dots$
\end{example}
\begin{remark}
    Setting $q_1 = -q_2$ kills the monopole term, leaving $V \sim 1/r^2$ and $E \sim 1/r^3$ at leading order.
\end{remark}
\begin{definition}[Dipole Moment]
    For charges $+q$ and $-q$ separated by $\vec{d}$ (pointing from $-q$ to $+q$), the dipole moment is
    \begin{equation}
        \vec{p} = q\vec{d}, \qquad [\vec{p}] = \mathrm{C\cdot m}.
    \end{equation}
    For any distribution,
    \begin{equation}
        \vec{p} = \sum_i q_i\vec{r}_i = \int\vec{r}'\rho(\vec{r}')\,d\tau',
    \end{equation}
    which reduces to $q\vec{d}$ for the pair. It measures how far apart the positive and negative charge sit, and sets the dipole term of the potential,
    \begin{equation}
        V_{\text{dip}} = \frac{1}{4\pi\epsilon_0}\frac{\vec{p}\cdot\hat{r}}{r^2}.
    \end{equation}
\end{definition}
\begin{proposition}[Dipole field]
    For $\vec{p} = p\hat{z}$, $V_{\text{dip}} = \frac{p\cos\theta}{4\pi\epsilon_0r^2}$, and $\vec{E} = -\vec{\nabla}V$ in spherical coordinates gives
    \begin{equation}
        \vec{E}_{\text{dip}}(r,\theta) = \frac{p}{4\pi\epsilon_0r^3}\left(2\cos\theta\,\hat{r} + \sin\theta\,\hat{\theta}\right).
    \end{equation}
\end{proposition}
\begin{definition}[Ideal dipole]
    Take $q_1 = q$ and $q_2 = -q$. The monopole and quadrupole terms vanish, and the dipole term is $\frac{1}{4\pi\epsilon_0}\frac{qd\cos\theta}{r^2}$. Every term in the expansion carries a factor $q\,d^{\,n}$, which we can write as $(qd)\,d^{\,n-1}$. Now shrink $d \to 0$ while raising $q \to \infty$ so that the product $qd$ stays fixed:
    \begin{itemize}
        \item The dipole term ($n=1$) contains only $qd$, so it is unchanged.
        \item Every later term carries at least one extra factor of $d$, so it goes to zero.
    \end{itemize}
    ($q$ must grow, otherwise the dipole term would vanish too.) What is left is a point object with dipole moment $\vec{p} = q\vec{d}$ and exactly $V = \frac{1}{4\pi\epsilon_0}\frac{\vec{p}\cdot\hat{r}}{r^2}$.
\end{definition}
\begin{remark}
    The names count charges in powers of 2: monopole (1), dipole (2), quadrupole (4), octupole (8), and so on. Each is built from two opposite copies of the previous one, placed close together so the previous order cancels. For example, two opposite dipoles give a quadrupole with $V \sim 1/r^3$.
\end{remark}
\begin{theorem}[Multipole expansion]
    For $r$ large compared with the size of the distribution,
    \begin{equation}
        V(\vec{r}) = \frac{1}{4\pi\epsilon_0}\left(\frac{Q}{r} + \frac{\vec{p}\cdot\hat{r}}{r^2} + \frac{\sum_{i,j}\hat{r}_iQ_{ij}\hat{r}_j}{r^3} + \dots\right),
    \end{equation}
    with
    \[Q = \int\rho(\vec{r}')\,d\tau', \qquad
    \vec{p} = \int\vec{r}'\rho(\vec{r}')\,d\tau', \qquad
    Q_{ij} = \int\frac{1}{2}\left(3r'_ir'_j - r'^2\delta_{ij}\right)\rho(\vec{r}')\,d\tau'.\]
    Griffiths writes the same series with Legendre polynomials; see Appendix~\ref{app:legendre}.
\end{theorem}
\begin{remark}
    Each term is smaller than the previous one by a factor of roughly $d/r$, where $d$ is the size of the charge distribution. Far away ($r \gg d$) the first nonzero term dominates and the rest are small corrections. Close up ($r \sim d$) the terms no longer shrink and the expansion is useless. So to find the leading behaviour: compute $Q$; if it vanishes, compute $\vec{p}$; if that vanishes too, the leading term is the quadrupole.
\end{remark}
\begin{remark}[Origin dependence]
    Shifting the origin by $\vec{a}$ changes $\vec{p} \to \vec{p} - Q\vec{a}$, so $\vec{p}$ is well defined only when $Q = 0$ (a neutral molecule, not an ion). In general only the first non-vanishing moment is origin-independent.
\end{remark}
\begin{problem}[Midterm drill]
    Charges $3q$ at $a\hat{z}$, $q$ at $-a\hat{z}$, and $-2q$ at $\pm a\hat{y}$. (i) Sketch how to find the energy. (ii) Find $V$ to dipole order.
\end{problem}
\begin{solution}
    (i) Sum $\frac{q_iq_j}{4\pi\epsilon_0\mathfrak{r}_{ij}}$ over the $6$ pairs: the two $z$ charges are $2a$ apart, as are the two $y$ charges, and each $z$ charge is $\sqrt{2}a$ from each $y$ charge:
    \[U = \frac{1}{4\pi\epsilon_0 a}\left[\frac{3q^2}{2} + \frac{4q^2}{2} + \frac{2(3q)(-2q) + 2(q)(-2q)}{\sqrt{2}}\right] = \frac{q^2}{4\pi\epsilon_0 a}\left(\frac{7}{2} - 8\sqrt{2}\right).\]
    (ii) $Q = 3q + q - 4q = 0$, and $\vec{p} = 3qa\hat{z} - qa\hat{z} - 2qa\hat{y} + 2qa\hat{y} = 2qa\hat{z}$, so
    \[V \approx \frac{1}{4\pi\epsilon_0}\frac{2qa\cos\theta}{r^2}.\]
\end{solution}
\begin{problem}[Midterm 2025, Q4]
    Charges $+q$ and $-q$, a distance $a$ apart, sit a height $d$ above a conductor filling $z < 0$. (a) Find $\vec{E}$ and $V$ for $z < 0$. (b) Describe how to find the exact $V$ for $z > 0$. (c) Find how $V$ and $|\vec{E}|$ fall off for $r \gg d$.
\end{problem}
\begin{solution}
    \begin{enumerate}[label=(\alph*)]
        \item $\vec{E} = 0$ inside the conductor, so $V$ is constant, independent of $z$ ($V = 0$, matching the plane at infinity).
        \item Method of images: place $-q$ at depth $d$ directly below $+q$, and $+q$ directly below $-q$. The four charges give $V = 0$ on $z = 0$, and the images lie outside $z > 0$, so by uniqueness
        \[V = \frac{1}{4\pi\epsilon_0}\sum_{i=1}^{4}\frac{q_i}{|\vec{r} - \vec{r}_i|}\quad\text{for } z > 0.\]
        \item Total charge is $0$. The real pair has $\vec{p} = qa\hat{x}$ and the image pair $-qa\hat{x}$, so the dipole moment also cancels. The leading term is the quadrupole: $V \sim 1/r^3$ and $|\vec{E}| \sim 1/r^4$.
    \end{enumerate}
\end{solution}

\subsection{Current Density}
\begin{definition}[Current density]
    The current $I$ through a wire is not a local quantity, since a wire has finite thickness. The local version is the current density
    \[\vec{J} = \frac{d\vec{I}}{da_\perp},\]
    where $da_\perp$ is the differential cross section perpendicular to the flow.
\end{definition}
\begin{remark}
    Applying the divergence theorem (Theorem~\ref{FTD}) to $\vec{J}$ is how conservation of charge gets written down: the charge inside $V$ can only decrease by flowing out through $S$,
    \[\frac{d}{dt}\iiint_V\rho\, dV = - \oiint_S\vec{J}\cdot d\vec{A} = -\iiint_V \vec{\nabla}\cdot\vec{J}\,dV \quad\Longrightarrow\quad \frac{\partial\rho}{\partial t} = -\vec{\nabla}\cdot\vec{J}.\]
\end{remark}

\begin{remark}[Compactified dimensions]
    Suppose an extra dimension exists but is curled up, the way a flat sheet with coordinates $(x,y)$ can be rolled into a cylinder whose second coordinate $w$ is periodic with a small radius. At distances much smaller than that radius, field lines spread out into all the dimensions and the field falls off faster than $1/r^2$. At distances much larger than it, there is only room to spread out in the large dimensions and the usual $1/r^2$ is recovered. So a measured deviation from the inverse square law at short range would be evidence for extra dimensions.\par
    This has been tested for gravity by the E\"ot-Wash torsion balance experiments (2004, 2007, 2020), which checked the $1/r^2$ law down to separations of $52\,\mu$m. Newtonian gravity fit the data, ruling out gravitational strength Yukawa interactions with ranges above $38.6\,\mu$m at 95\% confidence.
\end{remark}
\begin{remark}
    \label{app:legendre}
    Griffiths' form of the multipole expansion, with $\alpha$ the angle between $\vec{r}$ and $\vec{r}'$ and $P_n$ the Legendre polynomials:
    \[V(\vec{r}) = \frac{1}{4\pi\epsilon_0}\sum_{n=0}^{\infty}\frac{1}{r^{n+1}}\int(r')^nP_n(\cos\alpha)\rho(\vec{r}')\,d\tau'.\]
    $P_0 = 1$ gives the monopole, $P_1 = \cos\alpha$ the dipole (since $r'\cos\alpha = \vec{r}'\cdot\hat{r}$), and $P_2 = \frac{1}{2}(3\cos^2\alpha - 1)$ the quadrupole.
\end{remark}
\begin{remark}[Self-energy]
    \label{rem:self-energy}
    The energy of a charge configuration can be written as a pair sum or as a field integral,
    \[W_{\text{pair}} = \frac{1}{2}\frac{1}{4\pi\epsilon_0}\sum_{i \neq j}\frac{q_i q_j}{\mathfrak{r}_{ij}},\qquad W_{\text{field}} = \frac{\epsilon_0}{2}\int E^2\,d\tau,\]
    but the two do not always agree. The pair sum only counts the work of bringing different charges together ($i \neq j$); it ignores the work needed to build each point charge itself. The field integral counts everything, including that self-energy. So $W_{\text{field}}$ is always positive (since $E^2 \geq 0$), while $W_{\text{pair}}$ can be negative, e.g.\ $+q$ and $-q$ at distance $d$ give $W_{\text{pair}} = -\frac{q^2}{4\pi\epsilon_0 d}$. For a true point charge the self-energy is infinite: near it $E^2 \sim 1/r^4$, and $\int \frac{1}{r^4}\,r^2\,dr$ blows up at $r = 0$. So use the pair sum for point charges and the field integral for continuous distributions.
\end{remark}

\begin{proof}
    The curl of a curl. Set $\vec{G} \equiv \vec{\nabla}\times(\vec{\nabla}\times\vec{F})$. Then,
    \[G_i = \sum_{j,k}\varepsilon_{ijk}\partial_j\sum_{m,n} \varepsilon_{kmn}\partial_m F_n=\sum_{j,k,m,n}\varepsilon_{ijk}\varepsilon_{kmn}\partial_j\partial_mF_n.\]
    Using the cyclic permutation $\varepsilon_{ijk} = \varepsilon_{kij}$, so that the summed index $k$ sits first in both epsilons, the epsilon-delta identity~\ref{Epsilon-delta-id} gives
    \[\sum_k \varepsilon_{kij}\varepsilon_{kmn} = \delta_{im}\delta_{jn} - \delta_{in}\delta_{jm}.\]
    Reading off the proposition with the summed index $k$ removed:
    \[G_i = \sum_{j,m,n}\left(\delta_{im}\delta_{jn} - \delta_{in}\delta_{jm}\right)\partial_j\partial_m F_n.\]
    Each Kronecker delta collapses one sum by setting its indices equal. In the first term $\delta_{im}$ forces $m = i$ and $\delta_{jn}$ forces $n = j$; in the second, $\delta_{in}$ forces $n = i$ and $\delta_{jm}$ forces $m = j$:
    \[G_i = \sum_j \partial_j\partial_i F_j - \sum_j \partial_j\partial_j F_i \overset{\footnotemark}{=} \partial_i\sum_j\partial_j F_j - \sum_j \partial_j\partial_j F_i = \left(\vec{\nabla}(\vec{\nabla}\cdot\vec{F})\right)_i - (\nabla^2\vec{F})_i.\]
    \footnotetext{This equality uses $\partial_j\partial_i = \partial_i\partial_j$, valid for $\vec{F}$ twice continuously differentiable.}
    \[\therefore \vec{\nabla}\times(\vec{\nabla}\times\vec{F}) = \vec{\nabla}(\vec{\nabla}\cdot\vec{F}) - \nabla^2\vec{F}.\]
\end{proof}
\begin{proof}
    Curl of a gradient. Let $f$ be twice continuously differentiable. With $(\vec{\nabla}f)_k = \partial_k f$,
    \[\left(\vec{\nabla}\times(\vec{\nabla}f)\right)_i = \sum_{j,k}\varepsilon_{ijk}\partial_j\partial_k f.\]
    The indices $j$ and $k$ are summed over, so they can be renamed $j \leftrightarrow k$:
    \[\left(\vec{\nabla}\times(\vec{\nabla}f)\right)_i = \sum_{j,k}\varepsilon_{ikj}\partial_k\partial_j f.\]
    Now use antisymmetry, $\varepsilon_{ikj} = -\varepsilon_{ijk}$, and the fact that partial derivatives commute, $\partial_k\partial_j f = \partial_j\partial_k f$:
    \[\left(\vec{\nabla}\times(\vec{\nabla}f)\right)_i = -\sum_{j,k}\varepsilon_{ijk}\partial_j\partial_k f = -\left(\vec{\nabla}\times(\vec{\nabla}f)\right)_i.\]
    A number equal to its own negative is zero, so every component vanishes and $\vec{\nabla}\times(\vec{\nabla}f) = \vec{0}$.
\end{proof}
\begin{proof}
    Using the energy of $N$ point charges, $U = \frac{1}{2}\sum_i q_i V(\vec{r}_i)$. For a continuous distribution this becomes $U = \frac{1}{2}\int\rho V\,d\tau$. Substituting $\rho = -\epsilon_0\nabla^2 V$ from Poisson's equation,
    \[U = -\frac{\epsilon_0}{2}\int V\nabla^2 V\,d\tau.\]
    To rewrite $V\nabla^2V$, use the product rule $\partial_i(fg) = (\partial_i f)g + f(\partial_i g)$ with $f = V$ and $g = \partial_i V$, summed over $i$:
    \[\underbrace{\sum_i\partial_i(V\partial_i V)}_{\vec{\nabla}\cdot(V\vec{\nabla}V)} = \underbrace{\sum_i(\partial_i V)^2}_{|\vec{\nabla}V|^2} + \underbrace{V\sum_i\partial_i\partial_i V}_{V\nabla^2 V}
    \quad\Longrightarrow\quad
    V\nabla^2 V = \vec{\nabla}\cdot(V\vec{\nabla}V) - |\vec{\nabla}V|^2.\]
    Substituting, the overall minus sign flips both terms:
    \[U = -\frac{\epsilon_0}{2}\int\left[\vec{\nabla}\cdot(V\vec{\nabla}V) - |\vec{\nabla}V|^2\right]d\tau = \frac{\epsilon_0}{2}\int|\vec{\nabla}V|^2\,d\tau - \frac{\epsilon_0}{2}\int\vec{\nabla}\cdot(V\vec{\nabla}V)\,d\tau.\]
    The second integral vanishes. To see this, integrate over a huge sphere of radius $R$ instead of all space, and let $R\to\infty$ at the end. The divergence theorem turns the integral into a flux through the sphere's surface, $\oint V\vec{\nabla}V\cdot d\vec{a}$. If the charge sits in a finite region, then far away $V \sim 1/R$ and $|\vec{\nabla}V| \sim 1/R^2$, so the integrand is $\sim 1/R^3$, while the area is $4\pi R^2$. The flux is therefore $\sim 1/R$, which goes to zero. With $\vec{E} = -\vec{\nabla}V$, $U = \frac{\epsilon_0}{2}\int E^2\,d\tau$.
\end{proof}
