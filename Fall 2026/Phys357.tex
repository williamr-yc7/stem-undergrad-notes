\section{Mathematical Background}
\begin{definition}[Kronecker delta]
    The Kronecker delta is a function of two indices that equals $1$ when they are equal and $0$ otherwise:
    \begin{equation}
        \delta_{ij} = \begin{cases} 1 & i = j, \\ 0 & i \neq j. \end{cases}
    \end{equation}
\end{definition}

\subsection{Series}
\begin{definition}
    The geometric series is
    \begin{equation}
        \frac{1}{1-x} = \sum_{n=0}^{\infty}x^n \quad (|x|<1).
    \end{equation}
\end{definition}
\begin{definition}
    The exponential series converges for all $x$:
    \begin{equation}
        e^{x} = \sum_{n=0}^{\infty}\frac{x^n}{n!}.
    \end{equation}
\end{definition}
\begin{definition}
    The cosine and sine series contain only the even and odd powers of $x$ respectively, with alternating signs:
    \begin{equation}
        \cos x = \sum_{n=0}^{\infty}\frac{(-1)^n x^{2n}}{(2n)!}, \quad \sin x = \sum_{n=0}^{\infty}\frac{(-1)^n x^{2n+1}}{(2n+1)!}.
    \end{equation}
\end{definition}
\begin{definition}
    The binomial series is a finite sum when the exponent is a natural number:
    \begin{equation}
        (1+u)^N = \sum_{n=0}^{N}\binom{N}{n}u^n \quad (N \in \mathbb{N}).
    \end{equation}
\end{definition}
\begin{definition}
    The logarithm series has no constant term, so $\ln(1+u) \approx u$ for small $u$:
    \begin{equation}
        \ln(1+u) = \sum_{n=1}^{\infty}\frac{(-1)^{n+1}u^n}{n} \quad (|u|<1).
    \end{equation}
\end{definition}

\subsection{Taylor series}
\begin{definition}[Taylor series]
    A function $f$ that is smooth near $a$ (it can be differentiated any number of times there) equals the power series
    \begin{equation}
        f(x) = \sum_{n=0}^{\infty}\frac{f^{(n)}(a)}{n!}(x-a)^n,
    \end{equation}
    where $f^{(n)}$ is the $n$th derivative. The special case $a = 0$ is called the Maclaurin series.
\end{definition}

\subsection{Euler's formula and trigonometric identities}
\begin{definition}[Euler's formula]
    Comparing the series for $e^{i\theta}$, $\cos\theta$ and $\sin\theta$ gives Euler's formula,
    \begin{equation}
        \label{eq:euler-formula}
        e^{i\theta} = \cos\theta + i\sin\theta,
    \end{equation}
    which can also be rewritten as
    \begin{equation*}
        \cos\theta = \frac{e^{i\theta}+e^{-i\theta}}{2}, \qquad \sin\theta = \frac{e^{i\theta}-e^{-i\theta}}{2i}.
    \end{equation*}
\end{definition}
\begin{definition}[Double-angle identities]
    Squaring both sides of Euler's formula (Equation~\eqref{eq:euler-formula}) and comparing with just taking $\theta \to 2\theta$ in Euler's formula, then matching real and imaginary parts, gives
    \begin{equation}
        \cos^2\theta - \sin^2\theta = \cos 2\theta, \qquad 2\sin\theta\cos\theta = \sin 2\theta.
    \end{equation}
\end{definition}

\subsection{Bases}
\begin{definition}[Basis]
    A basis is a set of vectors that are linearly independent and span the space.\footnote{A more rigorous definition is treated in detail in my \href{https://github.com/williamr-yc7/stem-undergrad-notes/blob/1479320110013b60a8c4eddb42c7916cda1b2e6e/Winter\%202026/Math223.pdf}{Math 223 notes}, in the section ``Basis and Dimension''.}
\end{definition}

\subsection{The classical picture}

\subsubsection{Angular momentum}
\begin{definition}[Orbital angular momentum]
    The orbital angular momentum of a particle is
    \[\vec{L} = \vec{r} \times \vec{p},\]
    where $\vec{r}$ is the position vector and $\vec{p} = m\vec{v}$ is the linear momentum.
\end{definition}
For motion confined to the $x$-$y$ plane, only the $z$ component survives. On a circular path of radius $r$ with speed $v$,
\[L_{z} = mvr.\]
\begin{definition}[Angular momentum of a rigid body]
    For a rigid body\footnote{A rigid body is one that does not deform under the action of external forces.} rotating about a fixed axis,
    \[L = I_{0}\omega,\]
    with $\omega$ the angular velocity and $I_{0}$ the moment of inertia, which measures how the mass is distributed relative to the rotation axis.
\end{definition}

\subsubsection{Magnetic moment of a circulating charge}
\begin{definition}[Magnetic moment]
    A vector $\vec{\mu}$ measuring the net circulation of electric charge. It sets the torque, and in a nonuniform field the force, that a magnet experiences in an external magnetic field.
\end{definition}
A point particle of mass $m$ and charge $q$ on a circular orbit of radius $r$ and period $T$ passes any given point once per period, so it acts as a steady current $I = q/T = qv/2\pi r$. Its moment is
\[\mu = \frac{IA}{c}\footnotemark = \frac{1}{c}\left(\frac{qv}{2\pi r}\right)\left(\pi r^{2}\right) = \frac{q}{2mc}\,mvr = \frac{q}{2mc}\,L_{z},\]
where $A$ is the area enclosed by the orbit.
\footnotetext{Gaussian (CGS) units, where a current loop has moment $IA/c$. In SI units it is $IA$.}

\subsubsection{Gyromagnetic ratio}
\begin{definition}[Gyromagnetic ratio]
    The constant $\gamma$ relating magnetic moment to angular momentum,
    \[\vec{\mu} = \gamma \vec{L}, \qquad \gamma = \frac{q}{2mc} \text{ for a circulating charge.}\]
\end{definition}
\begin{remark}
    This is purely classical: a point charge on a definite orbit, nothing quantized. It matters because measuring $\vec{\mu}$ is how we measure angular momentum, which is what the Stern-Gerlach apparatus does.
\end{remark}

\subsection{Spin}
\begin{definition}[Spin]
    Spin $\vec{S}$ is an \emph{intrinsic} angular momentum: a fixed property of the particle type, like charge or mass, not caused by any motion through space. A stationary electron alone in empty space still has spin. It comes in addition to any orbital $\vec{L}$, not in place of it.
\end{definition}

\subsubsection{Intrinsic moment and the g-factor}
Spin also carries a magnetic moment proportional to it, but with an extra dimensionless factor,
\begin{equation}
    \vec{\mu} = \gamma\vec{S}, \qquad \gamma = \frac{gq}{2mc}.
\end{equation}
\begin{definition}[g-factor]
    The dimensionless number $g$ above. It says how much stronger or weaker the moment is than the classical value ($g = 1$). It must be measured or derived from relativistic quantum theory.
\end{definition}

\subsubsection{Magnetons}
\begin{definition}[Planck constant]
    The constant that sets the scale at which quantization matters. It relates photon energy to frequency, $E = h\nu$, and
    \[h \approx 6.626 \times 10^{-34}\ \mathrm{J\,s}, \qquad \hbar = \frac{h}{2\pi} = 1.055 \times 10^{-27}\ \mathrm{erg\,s} = 6.582 \times 10^{-16}\ \mathrm{eV\,s}.\]
    Both have units of angular momentum, $[\hbar] = [L_z] = \mathrm{kg\,m^2/s}$.
\end{definition}
Writing $\vec{S} = \hbar(\vec{S}/\hbar)$ with $\vec{S}/\hbar$ dimensionless, the $\hbar$ joins the gyromagnetic ratio. The combination $|e|\hbar/2mc$ then carries all the units and sets the natural size of the moment for a given mass.
\begin{definition}[Magnetons]
    The natural units of magnetic moment for the electron (Bohr magneton) and for nucleons, i.e.\ protons and neutrons (nuclear magneton):
    \begin{equation}
        \mu_{B} = \frac{|e|\hbar}{2m_{e}c}, \qquad \mu_{N} = \frac{|e|\hbar}{2m_{p}c} \approx \frac{\mu_B}{1836},
    \end{equation}
    since $m_{p}/m_{e} \approx 1836$.
\end{definition}
The magnetic moments are then
\begin{equation}
    \vec{\mu}_{e} = -g_{e}\,\mu_{B}\,\frac{\vec{S}}{\hbar}, \qquad \vec{\mu}_{p,n} = g_{p,n}\,\mu_{N}\,\frac{\vec{S}}{\hbar},
\end{equation}
where the minus sign for the electron comes from its negative charge.
\begin{table}[H]
\centering
\begin{tabular}{lcccc}
\hline
Particle & Charge $q$ & Mass $m$ & $g$-factor & Gyromagnetic ratio $\gamma$ \\
\hline
Electron & $-|e|$ & $m_{e}$ & $g_{e} \approx 2$ & $-g_{e}\mu_{B}/\hbar$ \\
Proton   & $+|e|$ & $m_{p}$ & $g_{p} \approx 5.59$ & $g_{p}\mu_{N}/\hbar$ \\
Neutron  & $0$    & $m_{n} \approx m_{p}$ & $g_{n} \approx -3.83$ & $g_{n}\mu_{N}/\hbar$ \\
\hline
\end{tabular}
\caption{Charges, masses, g-factors and gyromagnetic ratios.}
\end{table}
\begin{remark}
    Classically $\mu = (q/2mc)L_z$ vanishes when $q = 0$, yet the neutral neutron has a moment. The reason is that the neutron is not a point particle: it is made of three charged quarks (one up quark with charge $+\tfrac23|e|$ and two down quarks with $-\tfrac13|e|$ each). The charges add to zero, but the quarks' spins and motion inside the neutron each give a moment, and these moments do not cancel. So its $\vec{\mu}$ cannot come from a single circulating point charge: a first sign the classical picture fails. The same internal structure is why $g_p \approx 5.59$ is far from 2.
\end{remark}
\begin{remark}
    Since $m_{p} \gg m_{e}$, $\mu_{N} \ll \mu_{B}$, so nuclei are deflected far less than electrons in a Stern-Gerlach device.
\end{remark}

\subsection{The Stern-Gerlach apparatus}
\begin{definition}[Stern-Gerlach (SG) apparatus]
    A device that measures one component of a particle's magnetic moment. A beam of neutral atoms passes through a magnet shaped so that the field strength changes along a chosen axis, here $z$ (a field \emph{gradient}). The gradient pushes each atom by an amount proportional to $\mu_z$, and a screen records where it lands. Since $\vec{\mu} = \gamma\vec{S}$, the landing position measures $S_z$.
\end{definition}
\begin{figure}[H]
    \centering
    \includegraphics[width=0.5\textwidth]{../Images/Stern-Gerlach_apparatus.png}
    \caption{Illustration of the Stern-Gerlach apparatus.}
    \label{fig:stern-gerlach-apparatus}
\end{figure}

\subsubsection{Force in a nonuniform field}
A magnetic moment in a field has potential energy $U = -\vec{\mu}\cdot\vec{B}$, so the force experienced by the particle is
\begin{equation}
    \vec{F} = -\vec{\nabla}U = \vec{\nabla}(\vec{\mu}\cdot\vec{B}), \qquad F_{z} = \mu_{z}\frac{\partial B_{z}}{\partial z}.
\end{equation}
\begin{remark}
    A uniform field gives no net force, only a torque (a twist that turns $\vec{\mu}$ without moving the atom). That is why the magnet needs a gradient.
\end{remark}
\begin{definition}[Deflection]
    The sideways displacement of an atom on the screen caused by $F_z$. It is proportional to $\mu_z$, so the screen pattern is a record of the moments in the beam.
\end{definition}

\subsubsection{Silver atoms}
Silver has zero net orbital angular momentum, $\vec{L} = 0$, and its whole moment comes from one unpaired outermost (valence) electron, so the atom behaves like a single electron spin. With
\[\mu_{z} = -\frac{g_{e}|e|}{2m_{e}c}\,S_{z}\]
and a gradient $\partial B_{z}/\partial z < 0$,
\[F_{z} = g_{e}\frac{|e|}{2m_{e}c}\left|\frac{\partial B_{z}}{\partial z}\right| S_{z}.\]
\begin{result}
    $F_{z} \propto S_{z}$: the deflection measures the $z$ component of spin directly.
\end{result}

\subsubsection{Classical prediction versus observation}
Atoms leave the oven with spins pointing in random directions, so $S_z = |\vec{S}|\cos\theta$ takes every value in $[-|\vec{S}|,|\vec{S}|]$.
\begin{remark}
    Classically the screen should show a continuous smear. It shows two spots.
\end{remark}
\begin{result}[Quantization of spin]
    The electron's spin component, if prepared in the $\ket{+\hat{z}}$ state, is quantized:
    \begin{equation}
        S_{z} = \pm\frac{\hbar}{2}.
    \end{equation}
\end{result}

\subsection{Spin quantization}
Now if we generalize the SG experiment for any random particles (electron, deutron, atoms, proton, neutron...), we get a specific amount of spots on the screen. The number of spots is the number of beams $n$, which is related to the spin quantum number $s$ by
\begin{equation}
    \label{eq:spin-quantization}
    n=2s+1.
\end{equation}
\begin{definition}[Spin quantum number]
    From the amount of beams on the screen, we can determine the spin quantum number $s$ of the particle.
    \[s = \frac{n-1}{2}.\]
\end{definition}
The number $s$, fixed by the particle type, taking values $s = 0, \tfrac12, 1, \tfrac32, \dots$ A measurement of $S_z$ gives
\begin{equation}
    \label{eq:Sz-quantized}
    S_z = m\hbar.
\end{equation}
\begin{definition}[Magnetic quantum number]
    The number $m$ is called the magnetic quantum number, it defines the orientation of the spin, and it takes values
    \[m = -s, -s+1, \dots, +s.\]
\end{definition}
\begin{example}
    To get the values of the spin of an electron we use the silver atom in the SG apparatus, since the spin of silver is defined by its lonely valence electron. We get 2 beams, so $n=2$, therefore $s=1/2$, and the possible values of $m$ are $m = -1/2, +1/2$, which gives the two possible values of $S_z$:
    \[S_z = \pm\frac{\hbar}{2}.\]
\end{example}
\begin{remark}
    $S_z$ is a component. The magnitude is $|\vec{S}| = \hbar\sqrt{s(s+1)}$, which for the electron ($s=1/2$) is $\hbar\sqrt{3}/2$.
\end{remark}

We write SG$z$ for a device whose gradient points along $z$, and SG$x$ for the same device rotated by $90^\circ$ about the beam axis.
\begin{definition}[Ket]
    \label{Ket-def}
    A ket $\ket{\,\cdot\,}$ is a label for a quantum state, treated as a vector. For electron spin, the two outcomes of an SG$z$ give
    \[\ket{+\hat{z}} \equiv \ket{\uparrow} \; \text{for } S_{z} = +\frac{\hbar}{2}, \qquad \ket{-\hat{z}} \equiv \ket{\downarrow} \; \text{for } S_{z} = -\frac{\hbar}{2},\]
    and likewise $\ket{\pm\hat{x}}$ for $S_x = \pm\hbar/2$, and so on for any axis. We follow Townsend and use the $\ket{\pm\hat{n}}$ labels.
\end{definition}
\begin{definition}[Filter]
    Each device sends atoms into two beams. A filter blocks one of them.
\end{definition}

\end{remark}

\subsubsection{\texorpdfstring{Experiment 1: SG$z$ then SG$z$}{Experiment 1: SGz then SGz}}
Filter the $+\hat{z}$ beam and send it into a second SG$z$. Every atom comes out $+\hat{z}$. The measurement is repeatable: an atom found with $S_z = +\hbar/2$ gives the same answer again, so it left the first device in $\ket{+\hat{z}}$. We end with $N_0/2$ atoms in $\ket{+\hat{z}}$.

\subsubsection{\texorpdfstring{Experiment 2: SG$z$ then SG$x$}{Experiment 2: SGz then SGx}}
Filter $+\hat{z}$ and send it into an SG$x$. Half come out $+\hat{x}$, half $-\hat{x}$: knowing $S_z$ tells us nothing about $S_x$. We end with $N_0/4$ in each of $\ket{\pm\hat{x}}$.

\subsubsection{\texorpdfstring{Experiment 3: SG$z$, SG$x$, then SG$z$}{Experiment 3: SGz, SGx, then SGz}}
Filter $+\hat{z}$, filter $+\hat{x}$, then measure $S_z$ again. The beam splits 50/50 even though every atom was $+\hat{z}$ at the start. Blocking $+\hat{x}$ instead gives the same result. We end with $N_0/8$ in each of $\ket{\pm\hat{z}}$.
\begin{result}
    No state has definite values of both $S_x$ and $S_z$. Measuring $S_x$ destroys what we knew about $S_z$.
\end{result}

\subsubsection{\texorpdfstring{Experiment 4: SG$z$, modified SG$x$, then SG$z$}{Experiment 4: SGz, modified SGx, then SGz}}
\begin{definition}[Modified SG device]
    An SG device that splits the beam according to $S_n$ and then recombines the two beams, so both outcomes exit at the same place without being recorded.
\end{definition}
Filter $+\hat{z}$, pass it through a modified SG$x$ with neither path blocked, then measure $S_z$. Every atom comes out $+\hat{z}$: all $N_0/2$ end in $\ket{+\hat{z}}$.
\begin{result}
    The atoms cannot be going half through $+\hat{x}$ and half through $-\hat{x}$, since Experiment 3 showed each half alone gives 50/50. Instead the two paths \emph{interfere}: as with water or light waves, contributions from the two paths add and can cancel, here cancelling the $-\hat{z}$ outcome. No definite path can be assigned to an atom. Section~\ref{sec:states} makes this precise.
\end{result}
\begin{remark}
    With one path blocked, a modified SG acts exactly like an ordinary SG.
\end{remark}


\section{Quantum States}
\label{sec:states}
\marginpar{September 11-18, 2026}
Throughout, an \emph{$n$-level system} is one whose measurements have $n$ possible outcomes \eqref{eq:spin-quantization}.

\subsection{Kets, bras and bases}
\begin{definition}[Bra]
    Every ket (Def.~\ref{Ket-def}) $\ket{\psi}$ has a matching bra $\bra{\psi}$, its conjugate transpose (adjoint): the ket's coefficients are complex conjugated, and the column becomes a row, its derivation is shown in Section~\ref{sec:bras}.
\end{definition}
\begin{definition}[Inner product]
    A bra followed by a ket gives a complex number, the inner product $\braket{\phi|\psi} \in \mathbb{C}$.
\end{definition}
\begin{definition}[Orthonormal basis]
    The outcome states $\{\ket{j} : j = 1,\dots,n\}$ of one measurement form an orthonormal basis of the $n$-dimensional state space:
    \begin{equation}
        \braket{i|j} = \delta_{ij} = \begin{cases}1 & \text{if } i = j, \\ 0 & \text{if } i \neq j.\end{cases} \quad \forall i, j \in \{1,\dots,n\},
    \end{equation}
\end{definition}
\begin{example}
    For spin-$\tfrac12$ this is $\braket{\pm\hat{z}|\pm\hat{z}} = 1$ and $\braket{\pm\hat{z}|\mp\hat{z}} = 0$, and the same holds for $\{\ket{\pm\hat{n}}\}$.
\end{example}
\begin{remark}
    The label $j$ counts \emph{outcomes}, not axes. Choosing to measure $S_x$ gives $\ket{1} = \ket{+\hat{x}}$, $\ket{2} = \ket{-\hat{x}}$; choosing $S_z$ gives $\ket{1} = \ket{+\hat{z}}$, $\ket{2} = \ket{-\hat{z}}$. Either way $n = 2$.
\end{remark}
\begin{remark}
    A ket may also carry several labels, usually the quantum numbers that pin it down: $\ket{s,m}$. For the deuteron $\ket{1,+1}, \ket{1,0}, \ket{1,-1}$.
\end{remark}
\begin{definition}[Superposition]
    Any state is a linear combination of basis kets,
    \begin{equation}
        \ket{\psi} = \sum_{j=1}^{n} c_j\ket{j}, \qquad c_j \in \mathbb{C}.
    \end{equation}
\end{definition}
\begin{example}
    A state in the $S_z$ basis is a superposition of the two outcomes, with complex coefficients $c_\pm$:
    \[\text{spin-}\tfrac12: \ \ket{\psi} = c_{+}\ket{+\hat{z}} + c_{-}\ket{-\hat{z}}.\]
\end{example}
\begin{proposition}[Coefficients are inner products]
    \begin{equation}
        c_j = \braket{j|\psi}.
    \end{equation}
\end{proposition}
\begin{proof}
    $\braket{i|\psi} = \sum_j c_j\braket{i|j} = \sum_j c_j\delta_{ij} = c_i$.
\end{proof}
\begin{definition}[Projector]
    The outer product (a ket times a bra) $P_j = \ket{j}\bra{j}$. It is an operator that keeps only the $\ket{j}$ part of a state,
    \[P_j\ket{\psi} = \ket{j}\braket{j|\psi} = c_j\ket{j},\]
    and projecting twice changes nothing: $P_j^2 = \ket{j}\braket{j|j}\bra{j} = P_j$.
\end{definition}
\begin{proposition}[Completeness relation]
    Because every state can be expanded in the basis, the projectors sum to the identity:
    \begin{equation}
        \id = \sum_{j=1}^{n}\ket{j}\bra{j}.
    \end{equation}
\end{proposition}
\begin{example}
    \[\text{spin-}\tfrac12: \ \id = \ket{+\hat{n}}\bra{+\hat{n}} + \ket{-\hat{n}}\bra{-\hat{n}} \text{ for any axis } \hat{n}.\]
\end{example}
\begin{remark}[Inserting the identity]
    Writing $\id$ in this form in front of a state reproduces its expansion, $\ket{\psi} = \id\ket{\psi} = \sum_j\ket{j}\braket{j|\psi}$. This trick is used in almost every derivation below.
\end{remark}

\subsection{Measurement and probability}
\begin{definition}[Observable]
    A measurable quantity $A$. Its possible outcomes $a_j$ each come with a state $\ket{a_j}$, and these states form an orthonormal basis.
\end{definition}
\begin{example}
    For spin-$\tfrac12$, $A$ can be $S_x$, $S_y$ or $S_z$, with outcomes $\pm\hbar/2$.
\end{example}
\begin{principle}[Born rule]
    The probability that a system in $\ket{\psi}$ is found in $\ket{\phi}$ is
    \begin{equation}
        P(\phi\,|\,\psi) = \left|\braket{\phi|\psi}\right|^2 \in [0,1].
    \end{equation}
    In particular $P(A = a_j) = |\braket{a_j|\psi}|^2$, e.g.\ $P(S_z = \pm\hbar/2) = |c_\pm|^2$.
\end{principle}
\begin{definition}[Probability amplitude]
    The amplitude of the probability of outcome $a_j$ is the complex number $\braket{a_j|\psi}$.
\end{definition}
\begin{remark}
    The Born rule has no deeper derivation: it is a postulate that agrees with every experiment. As David Mermin put it, "shut up and calculate."
\end{remark}

\subsubsection{Bras}
\label{sec:bras}
To find which values a bra gets compared to its ket, we firstly define it with unknown coefficients, $\bra{\psi} = c_{+}'\bra{+\hat{z}} + c_{-}'\bra{-\hat{z}}$. By orthonormality,
\[\braket{\psi|\psi} = c_{+}'c_{+} + c_{-}'c_{-}.\]
A system in $\ket{\psi}$ is found in $\ket{\psi}$ with certainty, and it is always found in one of the two beams, so this must equal $|c_+|^2 + |c_-|^2 = 1$. That forces $c_{\pm}' = c_{\pm}^{*}$.
\begin{result}
    Going from a ket to its bra:
    \begin{equation}
        \bra{\psi} = \sum_j c_j^*\bra{j}, \qquad \braket{\phi|\psi} = \braket{\psi|\phi}^{*}.
    \end{equation}
\end{result}

\subsubsection{Normalization}
\begin{proposition}[Normalization]
    $\braket{\psi|\psi} = 1$ is the statement that the probabilities of all outcomes add to 1.
\end{proposition}
\begin{proof}
    \[\braket{\psi|\psi} = \sum_{i,j}c_i^*c_j\braket{a_i|a_j} = \sum_j|c_j|^2 = \sum_j P(A = a_j) = 1.\]
\end{proof}
\begin{example}
    For $\ket{\psi} = \frac{1}{\sqrt{3}}\ket{+\hat{z}} + i\sqrt{\frac{2}{3}}\ket{-\hat{z}}$, $c_+ = 1/\sqrt{3}$ and $c_- = i\sqrt{2/3}$, so $P(S_z = +\hbar/2) = 1/3$ and $P(S_z = -\hbar/2) = |i|^2\cdot 2/3 = 2/3$. They add to 1, so the state is normalized.
\end{example}

\subsubsection{Global and relative phase}
\begin{definition}[Global and relative phase]
    A global phase multiplies the whole ket: $e^{i\alpha}\ket{\psi}$ with $\alpha \in \mathbb{R}$. 
\end{definition}
\begin{proposition}
    A global phase changes no probability: $\left|\braket{\phi|e^{i\alpha}\psi}\right|^2 = |e^{i\alpha}|^2\left|\braket{\phi|\psi}\right|^2 = \left|\braket{\phi|\psi}\right|^2$ for every $\ket{\phi}$.
\end{proposition}
\begin{definition}[Relative phase]
    A relative phase sits between the terms of a superposition, like the $e^{i\delta}$ in $c_+\ket{+\hat{z}} + e^{i\delta}c_-\ket{-\hat{z}}$, and cannot be pulled out front.
\end{definition}
\begin{definition}[Physical state]
    What the system is actually like, as far as any experiment can tell. Two kets that give the same probabilities for all measurements describe the same physical state.
\end{definition}
\begin{result}
    $\ket{\psi}$ and $e^{i\alpha}\ket{\psi}$ are the same physical state, so global phases can be dropped. Relative phases cannot: they leave the $S_z$ probabilities alone but change the $S_x$ and $S_y$ ones (Section~\ref{sec:xy-states}).
\end{result}

\subsection{Expectation value and uncertainty}
\begin{definition}[Expectation value]
    The average outcome, weighted by the probabilities:
    \begin{equation}
        \braket{A} = \sum_j a_j\,P(A = a_j) = \sum_j a_j\left|\braket{a_j|\psi}\right|^2.
    \end{equation}
\end{definition}
\begin{example}
    \[\braket{S_z} = \frac{\hbar}{2}|c_+|^2 - \frac{\hbar}{2}|c_-|^2.\]
\end{example}
\begin{definition}[Uncertainty]
    The standard deviation (typical spread of outcomes around the average):
    \begin{equation}
        (\Delta A)^2 = \braket{(A - \braket{A})^2} = \braket{A^2} - \braket{A}^2, \qquad \braket{A^2} = \sum_j a_j^2\left|\braket{a_j|\psi}\right|^2.
    \end{equation}
\end{definition}
\begin{remark}
    For spin-$\tfrac12$ both outcomes square to $\hbar^2/4$, so $\braket{S_z^2} = \hbar^2/4$ for every state.
\end{remark}
\begin{example}
    For $\ket{+\hat{x}}$, Experiment 3 gives both $S_z$ probabilities $1/2$, so $\braket{S_z} = 0$ and $\Delta S_z = \hbar/2$.
\end{example}
\begin{example}
    For $\ket{\psi} = \sqrt{\tfrac{1}{3}}\ket{+\hat{z}} + i\sqrt{\tfrac{2}{3}}\ket{-\hat{z}}$,
    \[\braket{S_z} = \frac{\hbar}{2}\cdot\frac{1}{3} - \frac{\hbar}{2}\cdot\frac{2}{3} = -\frac{\hbar}{6}, \qquad (\Delta S_z)^2 = \frac{\hbar^2}{4} - \frac{\hbar^2}{36} = \frac{2}{9}\hbar^2, \qquad \Delta S_z = \frac{\sqrt{2}}{3}\hbar.\]
\end{example}

\subsubsection{Measured averages and shot noise}
\begin{remark}
    For $N$ measurements on identically prepared atoms, with $N_\pm$ giving $\pm\hbar/2$, the measured average is
    \begin{equation}
        \braket{S_z}_N = \frac{\hbar}{2}\frac{N_+ - N_-}{N},
    \end{equation}
    which tends to $\braket{S_z}$ as $N \to \infty$.
\end{remark}
\begin{remark}[Shot noise]
    The measured average $\braket{S_z}_N$ scatters around the true expectation value, and its average deviation from it is
    \begin{equation}
        \left|\braket{S_z}_N - \braket{S_z}\right| \sim \frac{\Delta S_z}{\sqrt{N}}.
    \end{equation}
    This scatter is called shot noise. It comes from the randomness of individual quantum outcomes, not from a faulty apparatus, and is absent only when $\Delta S_z = 0$, i.e.\ when the state has a definite $S_z$.
\end{remark}

\subsection{Spin states along \texorpdfstring{$x$}{x} and \texorpdfstring{$y$}{y}}
\label{sec:xy-states}

\subsubsection{The state \texorpdfstring{$\ket{+\hat{x}}$}{|+x>}}
Experiment 3 gives $|\braket{\pm\hat{z}|+\hat{x}}|^2 = \tfrac12$, so both amplitudes have modulus $1/\sqrt{2}$ with their own phases:
\[\ket{+\hat{x}} = \frac{e^{i\delta_{+}}}{\sqrt{2}}\left(\ket{+\hat{z}} + e^{i\delta}\ket{-\hat{z}}\right), \qquad \delta = \delta_{-} - \delta_{+}.\]
Dropping the global phase $e^{i\delta_+}$,
\[\ket{+\hat{x}} = \frac{1}{\sqrt{2}}\left(\ket{+\hat{z}} + e^{i\delta}\ket{-\hat{z}}\right).\]
Experiment 3 alone does not fix the relative phase $\delta$.

\subsubsection{Experiment 5: measuring \texorpdfstring{$S_y$}{Sy}}
\begin{definition}[Cyclic permutation]
    Relabelling the axes $x \to y \to z \to x$ (or the reverse). It is a rotation of the coordinate system, so no physics changes.
\end{definition}
Experiment 5 is Experiment 3 cyclically permuted: filter $+\hat{z}$, then measure $S_y$. By symmetry it also splits 50/50, so every cross-axis overlap has modulus squared $\tfrac12$:
\[\left|\braket{+\hat{x}|+\hat{z}}\right|^2 = \left|\braket{+\hat{y}|+\hat{z}}\right|^2 = \left|\braket{-\hat{x}|+\hat{y}}\right|^2 = \frac{1}{2}.\]
The same argument as for $\ket{+\hat{x}}$ gives, after dropping global phases,
\[\ket{\pm\hat{x}} = \frac{1}{\sqrt{2}}\left(\ket{+\hat{z}} \pm e^{i\delta}\ket{-\hat{z}}\right), \qquad \ket{\pm\hat{y}} = \frac{1}{\sqrt{2}}\left(\ket{+\hat{z}} \pm e^{i\gamma}\ket{-\hat{z}}\right).\]

\subsubsection{Fixing the phases}
Using orthonormality of the $\hat{z}$ basis,
\[\left|\braket{+\hat{x}|+\hat{y}}\right|^2 = \left|\frac{1}{2}\left(1 + e^{i(\gamma - \delta)}\right)\right|^2 = \frac{1}{2}\left(1 + \cos(\gamma - \delta)\right).\]
Setting this to $\tfrac12$ forces $\gamma - \delta = \pm\pi/2$. The rest is convention.
\begin{result}[Phase convention]
    With $\delta = 0$ and $\gamma = +\pi/2$,
    \begin{equation}
        \ket{\pm\hat{x}} = \frac{1}{\sqrt{2}}\left(\ket{+\hat{z}} \pm \ket{-\hat{z}}\right), \qquad
        \ket{\pm\hat{y}} = \frac{1}{\sqrt{2}}\left(\ket{+\hat{z}} \pm i\ket{-\hat{z}}\right),
    \end{equation}
    \begin{equation}
        \bra{\pm\hat{x}} = \frac{1}{\sqrt{2}}\left(\bra{+\hat{z}} \pm \bra{-\hat{z}}\right), \qquad \bra{\pm\hat{y}} = \frac{1}{\sqrt{2}}\left(\bra{+\hat{z}} \mp i\bra{-\hat{z}}\right).
    \end{equation}
\end{result}
\begin{remark}
    Complex coefficients are required: with real ones the relative phases could only be $0$ or $\pi$, and $\gamma - \delta = \pm\pi/2$ would be impossible.
\end{remark}
\begin{example}
    For $\ket{\psi} = \sqrt{\tfrac{1}{3}}\ket{+\hat{z}} + i\sqrt{\tfrac{2}{3}}\ket{-\hat{z}}$, compute $\braket{S_y}$. Using $\braket{\pm\hat{y}|+\hat{z}} = \tfrac{1}{\sqrt{2}}$ and $\braket{\pm\hat{y}|-\hat{z}} = \mp\tfrac{i}{\sqrt{2}}$,
    \[\braket{\pm\hat{y}|\psi} = \frac{1}{\sqrt{2}}\left(\sqrt{\tfrac{1}{3}} \pm \sqrt{\tfrac{2}{3}}\right), \qquad P\!\left(S_y = \pm\tfrac{\hbar}{2}\right) = \frac{1}{2} \pm \frac{\sqrt{2}}{3},\]
    which add to 1. Then
    \[\braket{S_y} = \frac{\hbar}{2}\left[\left(\frac{1}{2} + \frac{\sqrt{2}}{3}\right) - \left(\frac{1}{2} - \frac{\sqrt{2}}{3}\right)\right] = \frac{\sqrt{2}}{3}\hbar.\]
    The factor $i$ did nothing for $S_z$, but as a relative phase it is exactly what tilts the spin toward $+\hat{y}$.
\end{example}

\subsection{States as column vectors}
\begin{definition}[Representation of a state]
    In a basis, a ket is represented by the column of its coefficients and its bra by the conjugate row (the adjoint, $\dagger$). In the $S_z$ basis:
    \begin{equation}
        \ket{\psi} \xrightarrow{S_z} \psi = \begin{pmatrix} c_+ \\ c_- \end{pmatrix}, \qquad
        \bra{\psi} \xrightarrow{S_z} \psi^\dagger = \begin{pmatrix} c_+^* & c_-^* \end{pmatrix}, \qquad c_\pm = \braket{\pm\hat{z}|\psi}.
    \end{equation}
\end{definition}
\begin{remark}
    Write $\ket{\psi} \xrightarrow{S_z} \psi$, not $\ket{\psi} = \psi$: the ket is basis-independent, the column is not.
\end{remark}
\begin{example}
    \[\ket{+\hat{z}} \xrightarrow{S_z} \begin{pmatrix} 1 \\ 0 \end{pmatrix}, \qquad \ket{-\hat{z}} \xrightarrow{S_z} \begin{pmatrix} 0 \\ 1 \end{pmatrix},\]
    \[\ket{\pm\hat{x}} \xrightarrow{S_z} \frac{1}{\sqrt{2}}\begin{pmatrix} 1 \\ \pm 1 \end{pmatrix}, \qquad \ket{\pm\hat{y}} \xrightarrow{S_z} \frac{1}{\sqrt{2}}\begin{pmatrix} 1 \\ \pm i \end{pmatrix}.\]
\end{example}
\begin{proposition}[Inner product as a matrix product]
    If $\ket{\phi} \xrightarrow{S_z} \phi$ and $\ket{\psi} \xrightarrow{S_z} \psi$, then $\braket{\phi|\psi} = \phi^\dagger\psi$ (row times column), so $P(\phi\,|\,\psi) = \left|\phi^\dagger\psi\right|^2$ and $\braket{\psi|\psi} = \psi^\dagger\psi$.
\end{proposition}
\begin{remark}
    For a state given in another basis, replace each basis ket by its $S_z$ column and add. For $\ket{\psi} = a\ket{+\hat{y}} + b\ket{-\hat{y}}$,
    \[\psi \xrightarrow{S_z} \frac{a}{\sqrt{2}}\begin{pmatrix} 1 \\ i \end{pmatrix} + \frac{b}{\sqrt{2}}\begin{pmatrix} 1 \\ -i \end{pmatrix} = \frac{1}{\sqrt{2}}\begin{pmatrix} a+b \\ i(a-b) \end{pmatrix}.\]
    Checking $\psi^\dagger\psi = 1$ before going further catches most mistakes. A cleaner systematic method is in Section~\ref{sec:change-basis}.
\end{remark}

\subsection{Operators as matrices}
\begin{definition}[Linear operator]
    A map $A$ from kets to kets that respects superpositions, $A\left(a\ket{\psi} + b\ket{\phi}\right) = aA\ket{\psi} + bA\ket{\phi}$. Every operator in this course is linear.
\end{definition}
\begin{definition}[Matrix of an operator]
    In a basis $\{\ket{j}\}$, $A$ is represented by the matrix of \emph{matrix elements}
    \begin{equation}
        A_{ij} = \bra{i}A\ket{j}.
    \end{equation}
\end{definition}
\begin{proposition}[Operator acting on a state]
    $A\ket{\psi}$ is represented by the matrix times the column, $A\psi$.
\end{proposition}
\begin{proof}
    The $k$th entry of the column for $A\ket{\psi}$ is $\bra{k}A\ket{\psi}$. Inserting the identity,
    \[\bra{k}A\ket{\psi} = \sum_j\bra{k}A\ket{j}\braket{j|\psi} = \sum_j A_{kj}\,c_j,\]
    which is row $k$ of $A$ times the column $\psi$.
\end{proof}
\begin{example}
    The projectors and identity in the $S_z$ basis:
    \[P_{+\hat{z}} \xrightarrow{S_z} \begin{pmatrix} 1 & 0 \\ 0 & 0 \end{pmatrix}, \qquad P_{-\hat{z}} \xrightarrow{S_z} \begin{pmatrix} 0 & 0 \\ 0 & 1 \end{pmatrix}, \qquad \id \xrightarrow{S_z} \begin{pmatrix} 1 & 0 \\ 0 & 1 \end{pmatrix},\]
    and indeed $P_{+\hat{z}}\psi = (c_+, 0)^T$.
\end{example}
\begin{definition}[Adjoint]
    The adjoint $A^\dagger$ is the conjugate transpose: $(A^\dagger)_{ij} = A_{ji}^*$. In operator form, writing $A = \sum_{i,j}A_{ij}\ket{i}\bra{j}$,
    \begin{equation}
        A^\dagger = \sum_{i,j}A_{ij}^*\ket{j}\bra{i}.
    \end{equation}
    For real entries it is just the transpose.
\end{definition}
\begin{proposition}[Bra of a transformed state]
    If $\ket{\psi'} = A\ket{\psi}$, then $\bra{\psi'} = \bra{\psi}A^\dagger$.
\end{proposition}
\begin{proof}
    The column of $\ket{\psi'}$ is $A\psi$, and its bra is the conjugate transpose, $(A\psi)^\dagger = \psi^\dagger A^\dagger$.
\end{proof}
\begin{proposition}[Adjoint of a product]
    \begin{equation}
        \left(AB\right)^\dagger = B^\dagger A^\dagger.
    \end{equation}
    The order reverses; $A^\dagger B^\dagger$ is a different matrix in general.
\end{proposition}

\subsection{Observables as operators}
\begin{definition}[Eigenstate and eigenvalue]
    $\ket{a}$ is an eigenstate of $A$ with eigenvalue $a$ if
    \begin{equation}
        A\ket{a} = a\ket{a},
    \end{equation}
    the familiar $Mv = \lambda v$ in matrix form. A general operator turns a state into a different state; eigenstates come back as themselves times a number. A basis made of eigenstates of $A$ is called an \emph{eigenbasis} of $A$.
\end{definition}
\begin{remark}[Eigenstate vs eigenvector]
    Same thing in two languages. ``Eigenstate'' is the physics word for the ket, e.g.\ $\ket{+\hat{x}}$. ``Eigenvector'' is the linear-algebra word, usually for its column in some basis, e.g.\ $\frac{1}{\sqrt{2}}(1,1)^T$ in the $S_z$ basis.
\end{remark}
\begin{proposition}[Operator of an observable]
    For an observable with outcomes $a_j$ and states $\ket{a_j}$, define
    \begin{equation}
        \hat{A} = \sum_j a_j\ket{a_j}\bra{a_j}.
    \end{equation}
    Then $\hat{A}\ket{a_j} = a_j\ket{a_j}$ (the outcomes are its eigenvalues, the outcome states its eigenstates), and
    \begin{equation}
        \braket{A} = \bra{\psi}\hat{A}\ket{\psi}, \qquad \braket{A^2} = \bra{\psi}\hat{A}^2\ket{\psi}.
    \end{equation}
\end{proposition}
\begin{proof}
    $\hat{A}\ket{a_k} = \sum_j a_j\ket{a_j}\delta_{jk} = a_k\ket{a_k}$, and
    $\bra{\psi}\hat{A}\ket{\psi} = \sum_j a_j\braket{\psi|a_j}\braket{a_j|\psi} = \sum_j a_j\left|\braket{a_j|\psi}\right|^2 = \braket{A}$. Likewise $\hat{A}^2 = \sum_j a_j^2\ket{a_j}\bra{a_j}$.
\end{proof}
\begin{proposition}[An operator is diagonal in its own eigenbasis]
    \begin{equation}
        \bra{a_i}\hat{A}\ket{a_j} = a_j\delta_{ij}, \qquad \hat{A} \xrightarrow{\ket{a_j}} D = \begin{pmatrix} a_1 & & 0 \\ & \ddots & \\ 0 & & a_n \end{pmatrix}.
    \end{equation}
\end{proposition}

\subsubsection{Spin operators and Pauli matrices}
\begin{definition}[Spin operators]
    Applying the operator of an observable to spin, for $\hat{n} \in \{\hat{x},\hat{y},\hat{z}\}$,
    \begin{equation}
        \hat{S}_n = \frac{\hbar}{2}\left(\ket{+\hat{n}}\bra{+\hat{n}} - \ket{-\hat{n}}\bra{-\hat{n}}\right), \qquad \hat{S}_n\ket{\pm\hat{n}} = \pm\frac{\hbar}{2}\ket{\pm\hat{n}}.
    \end{equation}
\end{definition}
\begin{example}
    Each spin operator in its own eigenbasis, using $\hat{S}_n\ket{\pm\hat{n}} = \pm\frac{\hbar}{2}\ket{\pm\hat{n}}$ and orthonormality:
    \[\bra{\pm\hat{n}}\hat{S}_n\ket{\pm\hat{n}} = \pm\frac{\hbar}{2}, \qquad \bra{\pm\hat{n}}\hat{S}_n\ket{\mp\hat{n}} = \mp\frac{\hbar}{2}\braket{\pm\hat{n}|\mp\hat{n}} = 0,\]
    so
    \[\hat{S}_x \xrightarrow{S_x} \frac{\hbar}{2}\begin{pmatrix} 1 & 0 \\ 0 & -1 \end{pmatrix}, \qquad
    \hat{S}_y \xrightarrow{S_y} \frac{\hbar}{2}\begin{pmatrix} 1 & 0 \\ 0 & -1 \end{pmatrix}, \qquad
    \hat{S}_z \xrightarrow{S_z} \frac{\hbar}{2}\begin{pmatrix} 1 & 0 \\ 0 & -1 \end{pmatrix}.\]
\end{example}
In the $S_z$ basis the matrix elements of $\hat{S}_n$ are
\[\bra{i}\hat{S}_n\ket{j} = \frac{\hbar}{2}\Big(\braket{i|+\hat{n}}\braket{+\hat{n}|j} - \braket{i|-\hat{n}}\braket{-\hat{n}|j}\Big), \qquad i,j \in \{+\hat{z},-\hat{z}\}.\]
\begin{remark}[Shortcut]
    Write each projector as a column times its conjugate row, e.g.
    \[\ket{\pm\hat{x}}\bra{\pm\hat{x}} \xrightarrow{S_z} \frac{1}{2}\begin{pmatrix} 1 \\ \pm1 \end{pmatrix}\begin{pmatrix} 1 & \pm1 \end{pmatrix} = \frac{1}{2}\begin{pmatrix} 1 & \pm1 \\ \pm1 & 1 \end{pmatrix},\]
    then subtract and multiply by $\hbar/2$.
\end{remark}
\begin{definition}[Pauli matrices]
    The results, all in the $S_z$ basis:
    \begin{equation}
        \hat{S}_j = \frac{\hbar}{2}\sigma_j, \qquad
        \sigma_x = \begin{pmatrix} 0 & 1 \\ 1 & 0 \end{pmatrix}, \quad
        \sigma_y = \begin{pmatrix} 0 & -i \\ i & 0 \end{pmatrix}, \quad
        \sigma_z = \begin{pmatrix} 1 & 0 \\ 0 & -1 \end{pmatrix}.
    \end{equation}
    Each equals its own adjoint and squares to $\id$, so $\hat{S}_j^2 = \tfrac{\hbar^2}{4}\id$. Collected into the vector of Pauli matrices $\vec{\sigma} = \sigma_x\hat{x} + \sigma_y\hat{y} + \sigma_z\hat{z}$, this reads $\vec{S} \xrightarrow{S_z} \frac{\hbar}{2}\vec{\sigma}$.
\end{definition}

\subsubsection{Hermitian operators}
\begin{definition}[Hermitian operator]
    \label{def:hermitian}
    $A$ is Hermitian if it equals its own adjoint, $A = A^\dagger$ (e.g.\ the Pauli matrices).
\end{definition}
\begin{principle}
    \label{prin:hermitian}
    Every observable is represented by a Hermitian operator.
\end{principle}
\begin{proposition}
    The eigenvalues of a Hermitian operator are real, so measurement outcomes are real.
\end{proposition}
\begin{proof}
    If $A\ket{a} = a\ket{a}$, the bra is $\bra{a}A^\dagger = a^*\bra{a}$, so $a\braket{a|a} = \bra{a}A\ket{a} = \bra{a}A^\dagger\ket{a} = a^*\braket{a|a}$, and $a = a^*$.
\end{proof}

\subsection{Change of basis}
\label{sec:change-basis}
Take an $n$-level system with two orthonormal bases, $\{\ket{j}\}$ and $\{\ket{\alpha}\}$, $j,\alpha = 1,\dots,n$. Within each basis the overlaps are Kronecker deltas, but across bases they are not: $\braket{+\hat{x}|+\hat{z}} = \tfrac{1}{\sqrt{2}}$ is neither $0$ nor $1$. The same ket then has two columns,
\[\ket{\psi} \xrightarrow{\ket{j}} \psi^{j} = \begin{pmatrix} \braket{1|\psi} \\ \vdots \\ \braket{n|\psi} \end{pmatrix}, \qquad
\ket{\psi} \xrightarrow{\ket{\alpha}} \psi^{\alpha} = \begin{pmatrix} \braket{\alpha{=}1|\psi} \\ \vdots \\ \braket{\alpha{=}n|\psi} \end{pmatrix}.\]
Inserting $\id = \sum_\alpha\ket{\alpha}\bra{\alpha}$ into each $\ket{j}$ amplitude,
\[\braket{j|\psi} = \sum_{\alpha}\braket{j|\alpha}\braket{\alpha|\psi}.\]
\begin{result}[Change-of-basis matrix]
    \begin{equation}
        \psi^{j} = U\psi^{\alpha}, \qquad \psi^{\alpha} = U^\dagger\psi^{j}, \qquad U_{j\alpha} = \braket{j|\alpha}.
    \end{equation}
    The columns of $U$ are the new basis kets $\ket{\alpha}$ written in the old basis $\{\ket{j}\}$.
\end{result}
\begin{definition}[Unitary operator]
    $U$ is unitary if $U^\dagger U = UU^\dagger = \id$, i.e.\ its inverse is its adjoint.
\end{definition}
\begin{theorem}
    The change-of-basis matrix $U$ is unitary.
\end{theorem}
\begin{proof}
    With $(U^\dagger)_{\alpha j} = \braket{j|\alpha}^* = \braket{\alpha|j}$ and completeness of $\{\ket{j}\}$,
    \[(U^\dagger U)_{\alpha\beta} = \sum_j\braket{\alpha|j}\braket{j|\beta} = \braket{\alpha|\beta} = \delta_{\alpha\beta}.\]
    Completeness of $\{\ket{\alpha}\}$ gives $UU^\dagger = \id$ the same way.
\end{proof}
\begin{remark}
    So normalization does not depend on the basis: $\psi^{j\dagger}\psi^{j} = \psi^{\alpha\dagger}U^\dagger U\psi^{\alpha} = \psi^{\alpha\dagger}\psi^{\alpha}$.
\end{remark}
\begin{remark}
    If a unitary $U$ maps $\ket{\psi}$ to $c\ket{\phi}$ with $\ket{\phi}$ normalized, then $1 = \bra{\psi}U^\dagger U\ket{\psi} = |c|^2$, so $c = e^{i\varphi}$. In particular, eigenvalues of a unitary are pure phases, and $U$ leaves a physical state unchanged only when that state is an eigenstate of $U$.
\end{remark}
\begin{example}[Between $S_x$ and $S_z$]
    With $\ket{j} = \ket{\pm\hat{z}}$ and $\ket{\alpha} = \ket{\pm\hat{x}}$, the columns of $U$ are the $S_z$ columns of $\ket{\pm\hat{x}}$:
    \begin{equation}
        \label{eq:Uzx}
        U = \begin{pmatrix} \braket{+\hat{z}|+\hat{x}} & \braket{+\hat{z}|-\hat{x}} \\ \braket{-\hat{z}|+\hat{x}} & \braket{-\hat{z}|-\hat{x}} \end{pmatrix} = \frac{1}{\sqrt{2}}\begin{pmatrix} 1 & 1 \\ 1 & -1 \end{pmatrix}.
    \end{equation}
    The two directions are
    \[S_x \to S_z: \ \psi^z = U\psi^x, \qquad S_z \to S_x: \ \psi^x = U^\dagger\psi^z = U\psi^z,\]
    since $U$ is real and symmetric, so $U^\dagger = U$: the same matrix goes both ways ($U^2 = \id$). Bill calls $U$ ``$R\left(\tfrac{\pi}{2}\hat{y}\right)$'', but it is not the actual rotation matrix (see the warning in Section~\ref{sec:any-axis}).
\end{example}
\begin{remark}
    Between $S_y$ and $S_z$ the matrix has columns $\frac{1}{\sqrt{2}}(1, \pm i)^T$,
    \[U_y = \frac{1}{\sqrt{2}}\begin{pmatrix} 1 & 1 \\ i & -i \end{pmatrix}, \qquad \psi^z = U_y\psi^y, \qquad \psi^y = U_y^\dagger\psi^z,\]
    and here $U_y^\dagger \neq U_y$, so the reverse direction needs the actual adjoint.
\end{remark}

\subsubsection{Operators in another basis}
Let $\{\ket{j}\} = \{\ket{a_j}\}$ be the eigenbasis of $\hat{A}$, so $\hat{A} \xrightarrow{\ket{a_j}} D$ (diagonal), and $U_{j\alpha} = \braket{a_j|\alpha}$.
\begin{proposition}[Similarity transformation]
    \begin{equation}
        \hat{A} \xrightarrow{\ket{\alpha}} A^{\alpha} = U^\dagger D\,U.
    \end{equation}
    Rewriting the same operator in a new basis this way is called a similarity transformation.
\end{proposition}
\begin{proof}
    Insert the identity twice:
    \[\bra{\alpha}\hat{A}\ket{\beta} = \sum_{i,j}\braket{\alpha|a_i}\bra{a_i}\hat{A}\ket{a_j}\braket{a_j|\beta} = \sum_{i,j}(U^\dagger)_{\alpha i}D_{ij}U_{j\beta} = (U^\dagger D\,U)_{\alpha\beta}.\]
\end{proof}
\begin{result}[Expectation value in any basis]
    \begin{equation}
        \braket{A} = \psi^{j\dagger}D\,\psi^{j} = \psi^{\alpha\dagger}\left[U^\dagger D\,U\right]\psi^{\alpha}.
    \end{equation}
    Either move the state to the eigenbasis, or move the operator to the state's basis. Both give the same number.
\end{result}

\subsection{The rotation operator}
\begin{definition}[Rotation operator]
    $R(\theta\hat{n})$ is the linear operator that rotates a state by angle $\theta$ about the axis $\hat{n}$: $\ket{\psi'} = R(\theta\hat{n})\ket{\psi}$.
\end{definition}
\begin{example}
    $R\left(\tfrac{\pi}{2}\hat{y}\right)$ turns $\hat{z}$ into $\hat{x}$, so it takes $\ket{\pm\hat{z}}$ to $\ket{\pm\hat{x}}$ up to phase, and by linearity
    \[R\left(\tfrac{\pi}{2}\hat{y}\right)\left(c_+\ket{+\hat{z}} + c_-\ket{-\hat{z}}\right) = c_+\ket{+\hat{x}} + c_-\ket{-\hat{x}}.\]
    Up to a phase on one column, its $S_z$ matrix is the change-of-basis matrix $U$ of Eq.~\eqref{eq:Uzx} (exact form in Section~\ref{sec:any-axis}).
\end{example}
\begin{remark}[Why the bra carries $R^\dagger$]
    By the bra-of-a-transformed-state proposition, $\bra{\psi'} = \bra{\psi}R^\dagger$. Using $\bra{\psi}R$ instead fails: with $\ket{+\hat{x}} = R\left(\tfrac{\pi}{2}\hat{y}\right)\ket{+\hat{z}}$ it would give
    \[\braket{+\hat{x}|+\hat{x}} = \bra{+\hat{z}}R\left(\tfrac{\pi}{2}\hat{y}\right)R\left(\tfrac{\pi}{2}\hat{y}\right)\ket{+\hat{z}} = \bra{+\hat{z}}R(\pi\hat{y})\ket{+\hat{z}} \propto \braket{+\hat{z}|-\hat{z}} = 0,\]
    since a $\pi$ rotation about $\hat{y}$ flips $+\hat{z}$ to $-\hat{z}$. That contradicts $\braket{+\hat{x}|+\hat{x}} = 1$.
\end{remark}
\begin{theorem}
    Rotation operators are unitary: $R^\dagger R = RR^\dagger = \id$.
\end{theorem}
\begin{proof}
    A rotation must preserve normalization: $\braket{\psi'|\psi'} = \bra{\psi}R^\dagger R\ket{\psi} = 1$ for every $\ket{\psi}$, which forces $R^\dagger R = \id$.
\end{proof}

\subsection{Generators of rotations}
\label{sec:generators}
Rotations about the same axis add, $R(\theta_1\hat{z})R(\theta_2\hat{z}) = R\big((\theta_1+\theta_2)\hat{z}\big)$, so a finite rotation is $N \gg 1$ small ones,
\begin{equation}
    R(\theta\hat{z}) = \left[R\left(\tfrac{\theta}{N}\hat{z}\right)\right]^{N}.
\end{equation}
Taylor expanding one small step about the identity and naming the first-order operator $-iJ_z/\hbar$,
\begin{equation}
    R\left(\tfrac{\theta}{N}\hat{z}\right) = \id - \frac{i}{\hbar}J_z\frac{\theta}{N} + \mathcal{O}\!\left(\frac{1}{N^2}\right).
\end{equation}
\begin{result}
    By the problem ``The exponential as a limit'',
    \begin{equation}
        R(\theta\hat{z}) = \lim_{N\to\infty}\left(\id - \frac{i}{\hbar}J_z\frac{\theta}{N}\right)^{N} = e^{-iJ_z\theta/\hbar},
    \end{equation}
    and likewise $R(\theta\hat{n}) = e^{-iJ_n\theta/\hbar}$ for any axis.
\end{result}
\begin{definition}[Generator of rotations]
    The operator $J_z$ above: the thing that, exponentiated, produces rotations about $\hat{z}$. It has units of $\hbar$, i.e.\ of angular momentum ($[\hbar] = [\vec{r}\times\vec{p}]$), so $\vec{J}$ is identified as the angular momentum operator. For spin-$\tfrac12$, $\vec{J} = \vec{S}$ and $R(\theta\hat{n}) = e^{-i\theta\hat{S}_n/\hbar}$.
\end{definition}
\begin{proposition}
    The generator $J_z$ is Hermitian (Def.~\ref{def:hermitian}).
\end{proposition}
\begin{proof}
    To first order in $\theta$, $R = \id - i\theta J_z/\hbar$ and $R^\dagger = \id + i\theta J_z^\dagger/\hbar$. Unitarity gives
    \[\id = R^\dagger R = \id + \frac{i\theta}{\hbar}\left(J_z^\dagger - J_z\right) + \mathcal{O}(\theta^2),\]
    so $J_z^\dagger = J_z$.
\end{proof}
\begin{remark}
    So $J_z$ qualifies as an observable (Principle~\ref{prin:hermitian}), consistent with identifying it as angular momentum.
\end{remark}

\subsection{Functions of an operator}
\begin{proposition}[Function of an operator on an eigenstate]
    \label{prop:function-eigenstate}
    If $\hat{A}\ket{a} = a\ket{a}$ and $f$ has a Taylor series, then
    \begin{equation}
        f(\hat{A})\ket{a} = f(a)\ket{a}.
    \end{equation}
\end{proposition}
\begin{proof}
    $f(\hat{A})$ is defined by its Taylor series, and $\hat{A}^n\ket{a} = a^n\ket{a}$, so term by term
    \[f(\hat{A})\ket{a} = \sum_{n}\frac{f^{(n)}(0)}{n!}\hat{A}^n\ket{a} = \sum_{n}\frac{f^{(n)}(0)}{n!}a^n\ket{a} = f(a)\ket{a}.\]
\end{proof}
\begin{proposition}[Spectral form]
    Writing an operator as a sum of its eigenvalues times the projectors onto its eigenstates is its spectral form. Inserting $\id = \sum_j\ket{a_j}\bra{a_j}$ and using Proposition~\ref{prop:function-eigenstate},
    \begin{equation}
        f(\hat{A}) = \sum_j f(a_j)\ket{a_j}\bra{a_j}.
    \end{equation}
    With $f(x) = x$ this is the observable operator $\hat{A} = \sum_j a_j\ket{a_j}\bra{a_j}$.
\end{proposition}

\subsubsection{Rotations about \texorpdfstring{$\hat{z}$}{z}}
\begin{result}
    With $f(x) = e^{-i\phi x/\hbar}$ and $\hat{S}_z\ket{\pm\hat{z}} = \pm\frac{\hbar}{2}\ket{\pm\hat{z}}$,
    \begin{equation}
        R(\phi\hat{z})\ket{\pm\hat{z}} = e^{\mp i\phi/2}\ket{\pm\hat{z}}, \qquad R(\phi\hat{z}) \xrightarrow{S_z} \begin{pmatrix} e^{-i\phi/2} & 0 \\ 0 & e^{i\phi/2} \end{pmatrix}.
    \end{equation}
    This makes sense: a rotation about $\hat{z}$ cannot change $S_z$, so it can only attach phases to the $S_z$ states. In other bases the matrix is not diagonal.
\end{result}
\begin{remark}
    \label{rem:2pi}
    A full turn, $\phi = 2\pi$, gives $R = -\id$: a spin-$\tfrac12$ state needs $4\pi$ to return to itself. The sign is a global phase, so it is invisible on its own.
\end{remark}

\subsection{Rotations about any axis}
\label{sec:any-axis}
\begin{proposition}[Spectral form of a rotation]
    For $\hat{n} \in \{\hat{x},\hat{y},\hat{z}\}$,
    \begin{equation}
        R(\phi\hat{n}) = e^{-i\phi/2}\ket{+\hat{n}}\bra{+\hat{n}} + e^{i\phi/2}\ket{-\hat{n}}\bra{-\hat{n}}.
    \end{equation}
\end{proposition}
\begin{proof}
    Spectral form with $f(x) = e^{-i\phi x/\hbar}$, $\hat{A} = \hat{S}_n$, eigenvalues $\pm\hbar/2$.
\end{proof}
\begin{remark}[Recipe]
    In any basis $\{\ket{i}\}$ this reads $V D V^\dagger$, with $D = \mathrm{diag}\left(e^{-i\phi/2}, e^{i\phi/2}\right)$ and $V$ the matrix whose columns are $\ket{\pm\hat{n}}$ written in $\{\ket{i}\}$. It is the similarity transformation with the roles of the two bases swapped.
\end{remark}
\begin{proposition}[$R(\phi\hat{x})$ in the $S_z$ basis]
    \begin{equation}
        R(\phi\hat{x}) \xrightarrow{S_z} \begin{pmatrix} \cos\frac{\phi}{2} & -i\sin\frac{\phi}{2} \\ -i\sin\frac{\phi}{2} & \cos\frac{\phi}{2} \end{pmatrix}.
    \end{equation}
\end{proposition}
\begin{proof}
    Here $V = \frac{1}{\sqrt{2}}\begin{pmatrix} 1 & 1 \\ 1 & -1 \end{pmatrix} = V^\dagger$, so
    \[R(\phi\hat{x}) \xrightarrow{S_z} \frac{1}{2}\begin{pmatrix} 1 & 1 \\ 1 & -1 \end{pmatrix}\begin{pmatrix} e^{-i\phi/2} & 0 \\ 0 & e^{i\phi/2} \end{pmatrix}\begin{pmatrix} 1 & 1 \\ 1 & -1 \end{pmatrix} = \frac{1}{2}\begin{pmatrix} e^{-i\phi/2} + e^{i\phi/2} & e^{-i\phi/2} - e^{i\phi/2} \\ e^{-i\phi/2} - e^{i\phi/2} & e^{-i\phi/2} + e^{i\phi/2} \end{pmatrix},\]
    and Euler's formula gives $2\cos\frac{\phi}{2}$ and $-2i\sin\frac{\phi}{2}$.
\end{proof}
\begin{remark}
    This is the same matrix as $R(\phi\hat{z})$ in the $S_x$ basis. It is also the same as $R(\phi\hat{z})$ in the $S_y$ basis (Problem~\ref{prob:Rz-y}): with $\ket{\pm\hat{y}} = \frac{1}{\sqrt{2}}\left(\ket{+\hat{z}} \pm i\ket{-\hat{z}}\right)$,
    \[\bra{\pm\hat{y}}R(\phi\hat{z})\ket{\pm\hat{y}} = \cos\frac{\phi}{2}, \qquad \bra{\pm\hat{y}}R(\phi\hat{z})\ket{\mp\hat{y}} = -i\sin\frac{\phi}{2},\]
    so in both bases
    \[R(\phi\hat{z}) \longrightarrow \begin{pmatrix} \cos\frac{\phi}{2} & -i\sin\frac{\phi}{2} \\ -i\sin\frac{\phi}{2} & \cos\frac{\phi}{2} \end{pmatrix}.\]
\end{remark}
\begin{proposition}[$R(\phi\hat{y})$ in the $S_z$ basis]
    \begin{equation}
        R(\phi\hat{y}) \xrightarrow{S_z} \begin{pmatrix} \cos\frac{\phi}{2} & -\sin\frac{\phi}{2} \\ \sin\frac{\phi}{2} & \cos\frac{\phi}{2} \end{pmatrix}.
    \end{equation}
\end{proposition}
\begin{proof}
    Now $V = \frac{1}{\sqrt{2}}\begin{pmatrix} 1 & 1 \\ i & -i \end{pmatrix}$ and $V^\dagger = \frac{1}{\sqrt{2}}\begin{pmatrix} 1 & -i \\ 1 & i \end{pmatrix} \neq V$. Then
    \[R(\phi\hat{y}) \xrightarrow{S_z} \frac{1}{2}\begin{pmatrix} e^{-i\phi/2} + e^{i\phi/2} & -i\left(e^{-i\phi/2} - e^{i\phi/2}\right) \\ i\left(e^{-i\phi/2} - e^{i\phi/2}\right) & e^{-i\phi/2} + e^{i\phi/2} \end{pmatrix},\]
    and with $e^{-i\phi/2} - e^{i\phi/2} = -2i\sin\frac{\phi}{2}$ the off-diagonal entries are $\mp\sin\frac{\phi}{2}$.
\end{proof}
\begin{remark}
    This matrix is real, so $R(\phi\hat{y})$ never introduces factors of $i$ into an $S_z$ column. At $\phi = \pi$ it gives $R(\pi\hat{y})\ket{+\hat{z}} = \ket{-\hat{z}}$, the flip used above to show bras carry $R^\dagger$.
\end{remark}
\begin{remark}[WARNING: rotation versus change of basis]
    Plugging $\phi = \pi/2$ into this proposition does \emph{not} give Bill's change-of-basis matrix:
    \[R\left(\tfrac{\pi}{2}\hat{y}\right) \xrightarrow{S_z} \frac{1}{\sqrt{2}}\begin{pmatrix} 1 & -1 \\ 1 & 1 \end{pmatrix} \neq U = \frac{1}{\sqrt{2}}\begin{pmatrix} 1 & 1 \\ 1 & -1 \end{pmatrix}.\]
    The true rotation sends $\ket{-\hat{z}}$ to $-\ket{-\hat{x}}$, not $\ket{-\hat{x}}$, so its second column is $-\ket{-\hat{x}}$ while $U$ in Eq.~\eqref{eq:Uzx} has $+\ket{-\hat{x}}$. They differ only by the phase of one basis ket, which is a free choice, so both are valid. For a change of basis with our convention for $\ket{\pm\hat{x}}$, always use $U$ from Eq.~\eqref{eq:Uzx}.
\end{remark}

\subsection{Commutation}
Lay a cuboid on its largest face, so that its top face points along $+\hat{z}$, its smallest side face along $+\hat{x}$, and a shaded side face along $+\hat{y}$. Rotate it twice by $90^\circ$ counterclockwise:
\begin{itemize}
    \item About $\hat{x}$ then $\hat{y}$: the shaded face goes $+\hat{y} \to +\hat{z} \to +\hat{x}$.
    \item About $\hat{y}$ then $\hat{x}$: the shaded face goes $+\hat{y} \to +\hat{y} \to +\hat{z}$.
\end{itemize}
Thus $R_{\hat{x}}(\frac{\pi}{2})R_{\hat{y}}(\frac{\pi}{2}) \neq R_{\hat{y}}(\frac{\pi}{2})R_{\hat{x}}(\frac{\pi}{2})$. Since $R(\theta\hat{n}) = e^{-i\theta\frac{\hat{J}_n}{\hbar}}$, if the generators were numbers we could add the exponents, $R(\theta\hat{m})R(\theta\hat{n}) = e^{-i\theta\frac{\hat{J}_m + \hat{J}_n}{\hbar}}$, and the order would not matter. Since it does, the order of the generators must matter too.
\begin{proposition}
    The order of the generators matters:
    \begin{equation*}
        \hat{J}_m\hat{J}_n - \hat{J}_n\hat{J}_m = i\hbar\,\hat{J}_{m\times n},
    \end{equation*}
    where $\hat{J}_m = \hat{m}\cdot\vec{J}$ and $\hat{J}_{m\times n} = (\hat{m}\times\hat{n})\cdot\vec{J}$.
\end{proposition}
\begin{proof}
    Geometrically, two small rotations by $\Delta\theta$ about $\hat{m}$ and $\hat{n}$ performed in opposite orders differ, to second order, by a rotation by $\Delta\theta^2$ about $\hat{m}\times\hat{n}$:
    \[R(\Delta\theta\hat{m})R(\Delta\theta\hat{n}) - R(\Delta\theta\hat{n})R(\Delta\theta\hat{m}) = R(\Delta\theta^2\,\hat{m}\times\hat{n}) - 1.\]
    By Taylor expanding both sides using small angles,
    \begin{multline*}
        \left\{1-\frac{i\hat{J}_m\Delta\theta}{\hbar}-\frac{1}{2}\left(\frac{\hat{J}_m\Delta\theta}{\hbar}\right)^2\right\}\left\{1-\frac{i\hat{J}_n\Delta\theta}{\hbar}-\frac{1}{2}\left(\frac{\hat{J}_n\Delta\theta}{\hbar}\right)^2\right\} \\
        -\left\{1-\frac{i\hat{J}_n\Delta\theta}{\hbar}-\frac{1}{2}\left(\frac{\hat{J}_n\Delta\theta}{\hbar}\right)^2\right\}\left\{1-\frac{i\hat{J}_m\Delta\theta}{\hbar}-\frac{1}{2}\left(\frac{\hat{J}_m\Delta\theta}{\hbar}\right)^2\right\} \\
        = \left(1-\frac{i\hat{J}_{m\times n}\Delta\theta^2}{\hbar}\right)-1,
    \end{multline*}
    the lowest order nonvanishing terms involve $\Delta\theta^2$. Equating these terms we get
    \[-\frac{\Delta\theta^2}{\hbar^2}\left(\hat{J}_m\hat{J}_n - \hat{J}_n\hat{J}_m\right) = -\frac{i\Delta\theta^2}{\hbar}\hat{J}_{m\times n} \implies \hat{J}_m\hat{J}_n - \hat{J}_n\hat{J}_m = i\hbar\,\hat{J}_{m\times n},\]
    which is nonzero for noncollinear axes since $\hat{m}\times\hat{n} \neq \vec{0}$, therefore order matters.
\end{proof}
\begin{definition}[Commutator]
    \label{def:commutator}
    The commutator of two operators $A$ and $B$ is
    \begin{equation}
        [A,B] = AB - BA.
    \end{equation}
    $A$ and $B$ commute if $[A,B] = 0$, which means the order in which they act does not matter.
\end{definition}
\begin{example}[Commutation of Generators]
    The generators are non commutative:
    \[[J_x, J_y] = i\hbar J_z.\]
    The same holds for the cyclic permutations of $x$, $y$, $z$, in agreement with $[J_i, J_j] = i\hbar\,\epsilon_{ijk}J_k$.
\end{example}

\subsubsection{Commuting Operators}
\begin{proposition}
    \label{prop:simultaneous-eigenstates}
    Observables which commute have simultaneous eigenstates: if $[A,B] = 0$, there is a basis of states that are eigenstates of both $A$ and $B$.
\end{proposition}
\begin{proof}
    Let $A$ and $B$ be observables which commute, $[A,B] = 0$, with eigenvalues $a$ and $b$ respectively, and $A\ket{a} = a\ket{a}$.
    \begin{equation}
        \label{eq:nondegenerate-eigenstate}
        A(B\ket{a}) = BA\ket{a} = a(B\ket{a}),
    \end{equation}
    thus $B\ket{a}$ is also an eigenstate of $A$ with the same eigenvalue $a$, or $B\ket{a} = 0$. Since $a$ is nondegenerate, $B\ket{a}$ must be proportional to $\ket{a}$, so $B\ket{a} = b\ket{a}$ and $\ket{a}$ is also an eigenstate of $B$.
\end{proof}
\begin{definition}[$N$-level system]
    An $N$-level system is a quantum system whose state space is $N$-dimensional, with $N$ linearly independent eigenstates.
\end{definition}
\begin{definition}[Degeneracy]
    An eigenvalue $a$ is degenerate if several linearly independent eigenstates share it. These eigenstates are then labelled $\ket{a,j}$, where $j$ distinguishes them.
\end{definition}
\begin{proposition}
    \label{prop:degenerate-mapping}
    If $[A,B] = 0$ and $a$ is degenerate, then $B\ket{a,i}$ cannot involve any states $\ket{a',j}$ with $a' \neq a$, and $B$ maps the states $\ket{a,j}$ among themselves:
    \begin{equation*}
        B\ket{a,i} = \sum_j\bra{a,j}B\ket{a,i}\ket{a,j}.
    \end{equation*}
\end{proposition}
\begin{proof}
    By Equation~\eqref{eq:nondegenerate-eigenstate}, $A(B\ket{a,i}) = a(B\ket{a,i})$. Write $B\ket{a,i}$ in the eigenstates of $A$,
    \[B\ket{a,i} = \sum_{a',j} c_{a'j}\ket{a',j}, \qquad c_{a'j} = \bra{a',j}B\ket{a,i},\]
    where $a'$ runs over all eigenvalues of $A$. Applying $A$ term by term multiplies each $\ket{a',j}$ by its own $a'$, but by the first equation it must multiply everything by $a$:
    \[\sum_{a',j} a'\,c_{a'j}\ket{a',j} = \sum_{a',j} a\,c_{a'j}\ket{a',j} \implies \sum_{a',j}(a'-a)\,c_{a'j}\ket{a',j} = 0.\]
    Since the $\ket{a',j}$ are linearly independent, each $(a'-a)\,c_{a'j} = 0$, so $c_{a'j} = 0$ whenever $a' \neq a$. Only the $a' = a$ terms remain:
    \[B\ket{a,i} = \sum_j \bra{a,j}B\ket{a,i}\ket{a,j}.\]
\end{proof}
\begin{definition}[Block Diagonal]
    The matrix representation of $B$ in the eigenbasis $\{\ket{a,j}\}$ of $A$ looks block diagonal, where each block is associated with a different eigenvalue of $A$:
    \begin{equation}
        B \overset{\{\ket{a,j}\}}{\to} \underline{\underline{B}} = \begin{pmatrix}
            B^{(a_1)} & 0 & \cdots & 0 \\
            0 & B^{(a_2)} & \cdots & 0 \\
            \vdots & \vdots & \ddots & \vdots \\
            0 & 0 & \cdots & B^{(a_n)}
        \end{pmatrix}, \qquad B^{(a)}_{jk} = \bra{a,j}B\ket{a,k}.
    \end{equation}
\end{definition}
\begin{example}
    Let $A$ have eigenvalues $a_1$ with two states $\ket{a_1,1}, \ket{a_1,2}$, and $a_2$ with one state $\ket{a_2}$. In the basis $\{\ket{a_1,1}, \ket{a_1,2}, \ket{a_2}\}$,
    \begin{equation*}
        \underline{\underline{B}} = \begin{pmatrix}
            \bra{a_1,1}B\ket{a_1,1} & \bra{a_1,1}B\ket{a_1,2} & 0 \\
            \bra{a_1,2}B\ket{a_1,1} & \bra{a_1,2}B\ket{a_1,2} & 0 \\
            0 & 0 & \bra{a_2}B\ket{a_2}
        \end{pmatrix},
    \end{equation*}
    where the $2\times2$ block is $B^{(a_1)}$ and the $1\times1$ block is $B^{(a_2)}$. The zeros come from $\bra{a',j}B\ket{a,i} = 0$ for $a' \neq a$ (Proposition~\ref{prop:degenerate-mapping}).
\end{example}

\subsection{Uncertainty Relations}
\begin{definition}[Incompatible observables]
    Two observables $A$ and $B$ are incompatible if they do not commute.
\end{definition}
\begin{example}
    $J_x$ and $J_y$ are incompatible observables, since $[J_x, J_y] = i\hbar J_z \neq 0$. For spin-$\frac{1}{2}$ particles, send a beam through three Stern--Gerlach devices in a row:
    \begin{enumerate}
        \item SG$\hat{x}$: keep the particles in $\ket{+x}$, which have a definite $J_x = \frac{\hbar}{2}$.
        \item SG$\hat{y}$: half come out in $\ket{+y}$ and half in $\ket{-y}$. Keep $\ket{+y}$.
        \item SG$\hat{x}$ again: half come out in $\ket{+x}$ and half in $\ket{-x}$.
    \end{enumerate}
    Measuring $J_y$ destroyed the definite value of $J_x$, so no state has definite values of both. In step 2, knowing $J_x$ exactly made $J_y$ a 50/50 coin flip: the outcome of an incompatible observable is maximally uncertain.
\end{example}
\begin{proposition}[Generalized Uncertainty Relation]
    For any observables $A$ and $B$,
    \begin{equation}
        \Delta A\,\Delta B \geq \frac{1}{2}\left|\langle[A,B]\rangle\right|.
    \end{equation}
\end{proposition}
\begin{example}[Heisenberg's uncertainty relation]
    Position $x$ and momentum $p_x$ are non-commuting observables, $[x,p_x] = i\hbar$, which gives Heisenberg's uncertainty relation
    \begin{equation}
        \label{eq:heisenberg}
        \Delta x\,\Delta p_x \geq \frac{\hbar}{2}.
    \end{equation}
\end{example}

\subsection{Eigenstates of Angular Momentum}
Since the components of $\vec{J}$ are incompatible with each other, we cannot label states by $J_x$, $J_y$ and $J_z$ together. However, $\vec{J}^2 = \vec{J}\cdot\vec{J} = J_x^2 + J_y^2 + J_z^2$ is compatible with each component.
\begin{proposition}
    \label{prop:J2-Jz}
    $[J_z, \vec{J}^2] = 0$, and similarly for $J_x$ and $J_y$.
\end{proposition}
\begin{proof}
    Using $[A,BC] = B[A,C] + [A,B]C$,
    \begin{align*}
        [J_z, J_x^2] &= J_x[J_z,J_x] + [J_z,J_x]J_x = i\hbar(J_xJ_y + J_yJ_x),\\
        [J_z, J_y^2] &= J_y[J_z,J_y] + [J_z,J_y]J_y = -i\hbar(J_yJ_x + J_xJ_y),\\
        [J_z, J_z^2] &= 0,
    \end{align*}
    which sum to $[J_z, \vec{J}^2] = 0$.
\end{proof}
Thus by Proposition~\ref{prop:simultaneous-eigenstates}, there is a basis of simultaneous eigenstates of $\vec{J}^2$ and $J_z$, labelled $\ket{j,m}$.
\begin{remark}
    $\vec{J}^2$ commutes with every component, but the components do not commute with each other. Thus $\vec{J}^2$ can be paired with only one component at a time, and choosing $J_z$ is a convention.
\end{remark}
\begin{definition}[Angular momentum eigenstates]
    The angular momentum eigenstates $\ket{j,m}$ are the simultaneous eigenstates of $\vec{J}^2$ and $J_z$. Conventionally, their eigenvalues are written as
    \begin{equation}
        \vec{J}^2\ket{j,m} = \hbar^2 j(j+1)\ket{j,m}, \qquad J_z\ket{j,m} = \hbar m\ket{j,m}.
    \end{equation}
\end{definition}

\subsubsection{Allowed Values of $j$ and $m$}
To find the allowed values of $j$ and $m$, we use the ladder operators.
\begin{definition}[Ladder Operators]
    \label{def:ladder}
    The raising and lowering operators are
    \begin{equation}
        J_\pm = J_x \pm iJ_y.
    \end{equation}
    They are not Hermitian, since $J_\pm^\dagger = J_\mp$, so they are not observables.
\end{definition}
\begin{remark}
    The commutation relation is
    \[[J_z, J_\pm] = [J_z,J_x] \pm i[J_z,J_y] = \pm\hbar(J_x\pm i J_y) = \pm\hbar J_\pm.\]
\end{remark}
\begin{proposition}[Action of the ladder operators]
    \label{prop:ladder}
    The lowering operator acts as
    \begin{equation}
        J_-\ket{j,m} = \hbar[j(j+1)-m(m-1)]^{1/2}\ket{j,m-1},
    \end{equation}
    and the raising operator as
    \begin{equation}
        J_+\ket{j,m} = \hbar[j(j+1)-m(m+1)]^{1/2}\ket{j,m+1}.
    \end{equation}
\end{proposition}
The proof is given in Appendix~\ref{app:ladder-proof}.
\begin{proposition}[Allowed values of $j$ and $m$]
    $j = 0, \frac{1}{2}, 1, \frac{3}{2}, \ldots$, and for each $j$, $m \in \{-j, -j+1, \ldots, j-1, j\}$.
\end{proposition}
\begin{proof}
    In the state $\ket{j,m}$, $J_x^2 + J_y^2$ has a nonnegative expectation value, so
    \begin{equation*}
        \bra{j,m}(\vec{J}^2 - J_z^2)\ket{j,m} = (\hbar^2 j(j+1) - \hbar^2 m^2)\bra{j,m}\ket{j,m} \geq 0 \implies m^2 \leq j(j+1).
    \end{equation*}
    Thus $m$ is bounded, and the ladder must stop at some $m_{\max}$ and $m_{\min}$:
    \begin{equation*}
        J_+\ket{j,m_{\max}} = 0, \qquad J_-\ket{j,m_{\min}} = 0.
    \end{equation*}
    By Proposition~\ref{prop:ladder}, this gives $m_{\max} = j$ and $m_{\min} = -j$. Since $J_-$ lowers $m$ by exactly 1, going from $j$ to $-j$ takes a whole number $n$ of steps:
    \begin{equation*}
        j - (-j) = 2j = n \implies j = \frac{n}{2} = 0, \frac{1}{2}, 1, \frac{3}{2}, \ldots
    \end{equation*}
\end{proof}

\subsubsection{Integer and Half-Integer $j$}
\begin{proposition}
    If a $2\pi$ rotation leaves every state unchanged, $R(2\pi\hat{z}) = \id$, then $j$ must be an integer.
\end{proposition}
\begin{proof}
    Inserting $\id = \sum_m \ket{j,m}\bra{j,m}$,
    \begin{equation*}
        R(2\pi\hat{z}) = e^{-i2\pi J_z/\hbar}\sum_m \ket{j,m}\bra{j,m} = \sum_m e^{-i2\pi m}\ket{j,m}\bra{j,m},
    \end{equation*}
    which equals $1$ only if $e^{-i2\pi m} = 1$ for every $m\in\mathbb{Z}$, by Euler's formula Equation.\eqref{eq:euler-formula}. Since $m$ goes up to $j$, $j = 0, 1, 2, \ldots$
\end{proof}
\begin{example}
    Electrons have $j = \frac{1}{2}$, since they give two SG beams (Equation~\eqref{eq:spin-quantization}), so $m = \pm\frac{1}{2}$ and
    \begin{equation*}
        R(2\pi\hat{z})\ket{j,m} = e^{-i2\pi J_z/\hbar}\ket{j,m} = e^{-i2\pi m}\ket{j,m} = e^{\mp i\pi}\ket{j,m} = -\ket{j,m}.
    \end{equation*}
    This minus sign doesn't mean anything for the electron, since physically it is a global phase, which changes no probability.
\end{example}
\begin{definition}[Fermion]
    Any particle with half-integer $j$ is called a fermion, i.e an electron is a fermion.
\end{definition}
\begin{definition}[Boson]
    Any particle with full-integer $j$ is called a boson, i.e a photon $j=1$ is a boson.
\end{definition}

\subsection{Matrix Elements}
Applying $\bra{j,m'}$ to Proposition~\ref{prop:ladder} and to $J_z\ket{j,m} = \hbar m\ket{j,m}$ gives the matrix elements
\begin{equation}
    \bra{j,m'}J_\pm\ket{j,m} = \hbar\sqrt{j(j+1)-m(m\pm1)}\,\delta_{m',m\pm1}, \qquad \bra{j,m'}J_z\ket{j,m} = \hbar m\,\delta_{m'm}.
\end{equation}
Inverting $J_\pm = J_x \pm iJ_y$,
\begin{equation}
    J_x = \frac{1}{2}(J_+ + J_-), \qquad J_y = \frac{1}{2i}(J_+ - J_-),
\end{equation}
so the matrices of $J_x$ and $J_y$ follow from those of $J_\pm$.
\begin{example}[Spin-1]
    For $j = 1$, $m \in \{1, 0, -1\}$, giving three states. The entry in row $m'$, column $m$ is $\bra{1,m'}J\ket{1,m}$, with rows and columns ordered $m = 1, 0, -1$. For $J_z$, $\bra{1,m'}J_z\ket{1,m} = \hbar m\,\delta_{m'm}$, so only the diagonal is nonzero, with entries $\hbar m = \hbar, 0, -\hbar$:
    \begin{equation*}
        J_z \xrightarrow{\ket{1,m}} \hbar\begin{pmatrix} 1 & 0 & 0 \\ 0 & 0 & 0 \\ 0 & 0 & -1 \end{pmatrix}.
    \end{equation*}
    The same steps with $\bra{1,m'}J_\pm\ket{1,m}$ give
    \begin{equation*}
        J_+ \xrightarrow{\ket{1,m}} \hbar\begin{pmatrix} 0 & \sqrt{2} & 0 \\ 0 & 0 & \sqrt{2} \\ 0 & 0 & 0 \end{pmatrix}, \qquad
        J_- \xrightarrow{\ket{1,m}} \hbar\begin{pmatrix} 0 & 0 & 0 \\ \sqrt{2} & 0 & 0 \\ 0 & \sqrt{2} & 0 \end{pmatrix},
    \end{equation*}
    and $J_x = \frac{1}{2}(J_+ + J_-)$, $J_y = \frac{1}{2i}(J_+ - J_-)$ give
    \begin{equation*}
        J_x \xrightarrow{\ket{1,m}} \frac{\hbar}{\sqrt{2}}\begin{pmatrix} 0 & 1 & 0 \\ 1 & 0 & 1 \\ 0 & 1 & 0 \end{pmatrix}, \qquad
        J_y \xrightarrow{\ket{1,m}} \frac{i\hbar}{\sqrt{2}}\begin{pmatrix} 0 & -1 & 0 \\ 1 & 0 & -1 \\ 0 & 1 & 0 \end{pmatrix}.
    \end{equation*}
\end{example}

\subsection{Direct Product States}
For two spin-$\frac{1}{2}$ particles, a complete basis is
\begin{equation*}
    \ket{+\hat{z},+\hat{z}}, \quad \ket{+\hat{z},-\hat{z}}, \quad \ket{-\hat{z},+\hat{z}}, \quad \ket{-\hat{z},-\hat{z}},
\end{equation*}
where the first label is particle 1 and the second is particle 2.
\begin{remark}
    If particle 2 is in a state along $\hat{x}$, we can still write it in the $\hat{z}$ basis using $\ket{+\hat{x}} = \frac{1}{\sqrt{2}}(\ket{+\hat{z}} + \ket{-\hat{z}})$, for example
    \begin{equation*}
        \ket{+\hat{z},+\hat{x}} = \frac{1}{\sqrt{2}}\left(\ket{+\hat{z},+\hat{z}} + \ket{+\hat{z},-\hat{z}}\right).
    \end{equation*}
\end{remark}
\begin{definition}[Direct Product]
    The state of two particles is written as the direct product of the state of each particle:
    \begin{equation}
        \ket{+\hat{z},-\hat{z}} = \ket{+\hat{z}}_1 \otimes \ket{-\hat{z}}_2 = \ket{+\hat{z}}_1\ket{-\hat{z}}_2,
    \end{equation}
    where the $\otimes$ is often left implied. Similarly for operators,
    \begin{equation}
        S_{1z} = S_z \otimes \id, \qquad S_{2z} = \id \otimes S_z, \qquad S_{1z}S_{2z} = S_z \otimes S_z,
    \end{equation}
    where each factor acts only on its own particle, so the two spins are independent: $[\vec{S}_1, \vec{S}_2] = 0$.
\end{definition}
\begin{example}
    \[S_{2z}\ket{+\hat{z},-\hat{z}} = (\id \otimes S_z)\left(\ket{+\hat{z}}_1 \otimes \ket{-\hat{z}}_2\right) = \left(\id\ket{+\hat{z}}_1\right) \otimes \left(S_z\ket{-\hat{z}}_2\right) = -\frac{\hbar}{2}\ket{+\hat{z},-\hat{z}}.\]
\end{example}
\subsection{Hyperfine Splitting}
Denote $\vec{I}$ to be the nuclear (proton) spin and $\vec{S}$ to be the electron spin, both spin-$\frac{1}{2}$.
\begin{definition}[Hyperfine Interaction]
    The hyperfine interaction is the interaction of the magnetic moment of the proton with the magnetic field generated by the electron spin.
\end{definition}

\subsubsection{Hamiltonian}
\begin{definition}[Hyperfine Hamiltonian]
    The Hamiltonian of the hyperfine interaction in hydrogen is
    \begin{equation}
        H = \frac{A}{\hbar^2}\vec{S}\cdot\vec{I},
    \end{equation}
    where $\vec{S}\cdot\vec{I} = S_zI_z + \frac{1}{2}(S_+I_- + S_-I_+)$, using the ladder operators $S_\pm = S_x \pm iS_y$ and $I_\pm = I_x \pm iI_y$ (Definition~\ref{def:ladder}).
\end{definition}
For the next few explanations, we define $\ket{1} = \ket{+\hat{z},+\hat{z}}$, $\ket{2} = \ket{+\hat{z},-\hat{z}}$, $\ket{3} = \ket{-\hat{z},+\hat{z}}$, $\ket{4} = \ket{-\hat{z},-\hat{z}}$.
\begin{proposition}
    $\ket{1}$ and $\ket{4}$ are eigenstates of $H$, with eigenvalue $\frac{A}{4}$.
\end{proposition}
\begin{proof}
        Since $\ket{+\hat{z}} = \ket{\frac{1}{2},\frac{1}{2}}$ is the top of the ladder, $S_+\ket{+\hat{z}} = I_+\ket{+\hat{z}} = 0$ (Proposition~\ref{prop:ladder}), both $S_+I_-$ and $S_-I_+$ give $0$ on $\ket{1}$, so
    \begin{equation*}
        H\ket{1} = \frac{A}{\hbar^2}S_zI_z\ket{1} = \frac{A}{\hbar^2}\left(\frac{\hbar}{2}\right)^2\ket{1} = \frac{A}{4}\ket{1}.
    \end{equation*}
    Similarly, $S_-\ket{-\hat{z}} = I_-\ket{-\hat{z}} = 0$ gives $H\ket{4} = \frac{A}{4}\ket{4}$.
\end{proof}
However, $\ket{2}$ and $\ket{3}$ are not eigenstates. Acting on $\ket{2} = \ket{+\hat{z},-\hat{z}}$ with each term of $H$, using Proposition~\ref{prop:ladder},
\begin{align*}
    S_zI_z\ket{2} &= \left(\frac{\hbar}{2}\right)\left(-\frac{\hbar}{2}\right)\ket{2} = -\frac{\hbar^2}{4}\ket{2},\\
    S_+I_-\ket{2} &= \left(S_+\ket{+\hat{z}}\right)\left(I_-\ket{-\hat{z}}\right) = 0,\\
    S_-I_+\ket{2} &= \left(\hbar\ket{-\hat{z}}\right)\left(\hbar\ket{+\hat{z}}\right) = \hbar^2\ket{3},
\end{align*}
so
\begin{equation*}
    H\ket{2} = \frac{A}{\hbar^2}\left(-\frac{\hbar^2}{4}\ket{2} + \frac{1}{2}\hbar^2\ket{3}\right) = -\frac{A}{4}\ket{2} + \frac{A}{2}\ket{3}.
\end{equation*}
Similarly, $H\ket{3} = \frac{A}{2}\ket{2} - \frac{A}{4}\ket{3}$. Thus in the basis $\{\ket{1}, \ket{2}, \ket{3}, \ket{4}\}$,
\begin{equation*}
    H \to \underline{\underline{H}}=\begin{pmatrix} \frac{A}{4} & 0 & 0 & 0 \\ 0 & -\frac{A}{4} & \frac{A}{2} & 0 \\ 0 & \frac{A}{2} & -\frac{A}{4} & 0 \\ 0 & 0 & 0 & \frac{A}{4} \end{pmatrix}.
\end{equation*}
\begin{definition}[Energies]
    The eigenvalues of the Hamiltonian are the allowed energies of the system.
\end{definition}
\begin{remark}
    $\ket{1}$ and $\ket{4}$ have energy $E = \frac{A}{4}$. $\ket{2}$ and $\ket{3}$ are not eigenstates, so their energies come from diagonalizing the $2\times2$ block in $\{\ket{2}, \ket{3}\}$.
\end{remark}
To find the energies of the other eigenstates of $H$, we diagonalize this block:
\begin{equation*}
    \underline{\underline{h}} = \begin{pmatrix} -\frac{A}{4} & \frac{A}{2} \\ \frac{A}{2} & -\frac{A}{4} \end{pmatrix} = -\frac{A}{4}\id + \frac{A}{2}\sigma_x.
\end{equation*}
Since $\sigma_x$ has eigenvalues $\pm1$ with eigenvectors $\frac{1}{\sqrt{2}}(1, \pm1)^T$, $\underline{\underline{h}}$ has eigenvalues $-\frac{A}{4} \pm \frac{A}{2}$:
\begin{equation*}
    E = \frac{A}{4} \text{ for } \frac{1}{\sqrt{2}}(\ket{2} + \ket{3}), \qquad E = -\frac{3A}{4} \text{ for } \frac{1}{\sqrt{2}}(\ket{2} - \ket{3}).
\end{equation*}
\begin{definition}[Triplet and singlet]
    The three states with $E = \frac{A}{4}$ are the triplet states,
    \begin{equation*}
        \ket{T_+} = \ket{1}, \qquad \ket{T_0} = \frac{1}{\sqrt{2}}(\ket{2} + \ket{3}), \qquad \ket{T_-} = \ket{4},
    \end{equation*}
    and the state with $E = -\frac{3A}{4}$ is the singlet state, $\ket{S} = \frac{1}{\sqrt{2}}(\ket{2} - \ket{3})$.
\end{definition}

\subsubsection{The 21-cm Line}
The transition from a triplet state $\ket{T_\alpha}$, $\alpha \in \{0, \pm\}$, to the singlet state $\ket{S}$ lowers the energy of an atom with one electron and a spin-$\frac{1}{2}$ nucleus by $\frac{A}{4}-(-\frac{3A}{4}) = A$, so a photon of energy $A$ is emitted. For hydrogen,
\begin{equation}
    \nu = \frac{A}{2\pi\hbar} = 1.42\ \mathrm{GHz}, \qquad \lambda = \frac{c}{\nu} = 21\ \mathrm{cm}.
\end{equation}
\begin{remark}
    This is the ``21-cm line''. The transition is very long lived (lifetime $\sim 10^8$ years), and hydrogen is everywhere in the universe, so this microwave radiation tells us how the universe cooled after the Big Bang.
\end{remark}

\subsubsection{Total Spin}
\begin{proposition}
    With the total spin $\vec{J} = \vec{S} + \vec{I}$, the Hamiltonian is
    \begin{equation}
        H = \frac{A}{2\hbar^2}\left(\vec{J}^2 - \frac{3}{2}\hbar^2\right),
    \end{equation}
    so its eigenstates are $\ket{j,m}$, with energies
    \begin{equation*}
        E_1 = \frac{A}{4} \text{ for } \ket{1,m} = \ket{T_m}, \qquad E_0 = -\frac{3A}{4} \text{ for } \ket{0,0} = \ket{S}.
    \end{equation*}
\end{proposition}
\begin{proof}
    $\vec{J}^2 = \vec{S}^2 + \vec{I}^2 + 2\vec{S}\cdot\vec{I}$, and since both are spin-$\frac{1}{2}$, $\vec{S}^2 = \vec{I}^2 = \frac{3}{4}\hbar^2$, so
    \begin{equation*}
        \vec{S}\cdot\vec{I} = \frac{1}{2}\left(\vec{J}^2 - \frac{3}{2}\hbar^2\right) \implies H = \frac{A}{2\hbar^2}\left(\vec{J}^2 - \frac{3}{2}\hbar^2\right).
    \end{equation*}
    Since $H$ only contains $\vec{J}^2$, its eigenstates are the $\ket{j,m}$, and with $\vec{J}^2\ket{j,m} = \hbar^2 j(j+1)\ket{j,m}$,
    \begin{equation*}
        H\ket{j,m} = \frac{A}{2}\left(j(j+1) - \frac{3}{2}\right)\ket{j,m},
    \end{equation*}
    which gives $\frac{A}{4}$ for $j = 1$ and $-\frac{3A}{4}$ for $j = 0$.
\end{proof}

\subsection{Density Operator}
\begin{definition}[Pure State]
    A state that can be written as a single ket $\ket{\psi}$, e.g.\ $\ket{+\hat{z}}$.
\end{definition}
A pure state can still give random outcomes, e.g.\ measuring $S_x$ on $\ket{+\hat{z}}$, but measuring $S_z$ always gives $+\frac{\hbar}{2}$. In reality, we cannot prepare $\ket{+\hat{z}}$ perfectly every time, so to account for this imperfection we need mixed states.
\begin{definition}[Mixed State]
    A statistical mixture of pure states $\ket{\psi^{(k)}}$, each prepared with probability $p_k$, where $\sum_k p_k = 1$.
\end{definition}
\begin{definition}[Density Operator]
    The operator describing both pure and mixed states:
    \begin{equation}
        \rho = \ket{\psi}\bra{\psi} \quad \text{(pure)}, \qquad \rho = \sum_k p_k\ket{\psi^{(k)}}\bra{\psi^{(k)}} \quad \text{(mixed)}.
    \end{equation}
\end{definition}
\begin{proposition}
    The density operator is Hermitian, $\rho = \rho^\dagger$, so it has real eigenvalues.
\end{proposition}
\begin{proof}
    For a pure state, $\rho_{ij} = \braket{i|\psi}\braket{\psi|j}$, so $\rho_{ij}^* = \braket{j|\psi}\braket{\psi|i} = \rho_{ji}$. For a mixed state, each term $p_k\ket{\psi^{(k)}}\bra{\psi^{(k)}}$ is Hermitian by the same argument.
\end{proof}
\begin{definition}[Trace]
    The trace of an operator is the sum of its diagonal elements, $\text{Tr}\,A = \sum_i \bra{i}A\ket{i}$.
\end{definition}
\begin{proposition}
    $\text{Tr}\,\rho = 1$, for both pure and mixed states.
\end{proposition}
\begin{proof}
    For a pure state, $\text{Tr}\,\rho = \sum_i\braket{i|\psi}\braket{\psi|i} = \bra{\psi}\left(\sum_i\ket{i}\bra{i}\right)\ket{\psi} = \braket{\psi|\psi} = 1$. For a mixed state, each term gives $p_k$, and $\sum_k p_k = 1$.
\end{proof}
\begin{remark}
    Since $\text{Tr}\,\rho = 1$ for both, the trace cannot tell pure and mixed states apart, but $\text{Tr}\,\rho^2$ can.
\end{remark}
\begin{definition}[Purity]
    The purity of a state is $\text{Tr}\,\rho^2$:
    \begin{equation}
        \text{Tr}\,\rho^2 = 1 \implies \text{pure}, \qquad \text{Tr}\,\rho^2 < 1 \implies \text{mixed}.
    \end{equation}
\end{definition}
\begin{proposition}[Probability]
    The probability to measure $\ket{\phi}$ is $\text{Tr}\{P_\phi\,\rho\}$, where $P_\phi = \ket{\phi}\bra{\phi}$. For a pure state, this is $|\braket{\phi|\psi}|^2$.
\end{proposition}
\begin{proposition}[Expectation Value]
    \begin{equation}
        \langle A\rangle = \text{Tr}\{\rho A\}.
    \end{equation}
\end{proposition}
\begin{proof}
    For a pure state, inserting the identity twice, $\langle A\rangle = \sum_{ij}\braket{\psi|i}\bra{i}A\ket{j}\braket{j|\psi} = \sum_{ij}\rho_{ji}A_{ij} = \text{Tr}\{\rho A\}$. For a mixed state, $\langle A\rangle = \sum_k p_k\bra{\psi^{(k)}}A\ket{\psi^{(k)}} = \text{Tr}\{\rho A\}$ the same way.
\end{proof}
\begin{proposition}[von Neumann equation]
    \begin{equation}
        \frac{d\rho}{dt} = -\frac{i}{\hbar}[H, \rho].
    \end{equation}
\end{proposition}
\begin{proof}
    With $\rho(t) = \ket{\psi(t)}\bra{\psi(t)}$ and the Schrödinger equation, $\frac{d}{dt}\ket{\psi} = -\frac{i}{\hbar}H\ket{\psi}$ and $\frac{d}{dt}\bra{\psi} = \frac{i}{\hbar}\bra{\psi}H$,
    \begin{equation*}
        \frac{d\rho}{dt} = \left(\frac{d}{dt}\ket{\psi}\right)\bra{\psi} + \ket{\psi}\left(\frac{d}{dt}\bra{\psi}\right) = -\frac{i}{\hbar}(H\rho - \rho H).
    \end{equation*}
\end{proof}

\subsection{Action of the ladder operators}

\begin{proof}
    From the remark, $[J_z, J_-] = -\hbar J_-$, so $J_zJ_- = J_-(J_z - \hbar)$. Then
    \begin{equation*}
        J_z(J_-\ket{j,m}) = J_-(J_z - \hbar)\ket{j,m} = \hbar(m-1)(J_-\ket{j,m}),
    \end{equation*}
    so $J_-\ket{j,m}$ is an eigenstate of $J_z$ with eigenvalue $\hbar(m-1)$, thus $J_-\ket{j,m} = c_-\ket{j,m-1}$ for some constant $c_-$. To normalize, since $J_-^\dagger = J_+$ and the $\ket{j,m}$ are normalized,
    \begin{equation*}
        |c_-|^2 = \bra{j,m}J_+J_-\ket{j,m}.
    \end{equation*}
    Expanding, $J_+J_- = J_x^2 + J_y^2 - i[J_x,J_y] = \vec{J}^2 - J_z^2 + \hbar J_z$, so
    \begin{equation*}
        |c_-|^2 = \hbar^2[j(j+1) - m^2 + m] = \hbar^2[j(j+1) - m(m-1)].
    \end{equation*}
    Choosing $c_-$ real and positive gives the result. Similarly, $[J_z, J_+] = \hbar J_+$ gives $J_+\ket{j,m} = c_+\ket{j,m+1}$, and $J_-J_+ = \vec{J}^2 - J_z^2 - \hbar J_z$ gives $|c_+|^2 = \hbar^2[j(j+1) - m(m+1)]$.
\end{proof}
