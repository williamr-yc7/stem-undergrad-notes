\section{Temperature}
\subsection{Basic Concepts}
\begin{definition}[System]
    A system is the part of the universe under study.
\end{definition}
\begin{definition}[Surroundings]
    The surroundings are everything outside the system that can interact with it.
\end{definition}
\begin{definition}[Boundary]
    The boundary, or also named partition, is the wall separating the system from its surroundings.
\end{definition}
\begin{definition}[Closed System]
    A closed system exchanges energy but not matter with its surroundings.
\end{definition}
\begin{definition}[Equilibrium State]
    An equilibrium state is one in which all the bulk physical properties of the system (i.e pressure, volume...) are uniform throughout the system and do not change with time.
\end{definition}

\subsubsection{State Functions}
\begin{definition}[State Function]
    A state function is a property of a system that depends only on its current equilibrium state, not on how that state was reached. Its change between two states is therefore path independent.
\end{definition}
\begin{example}
    Pressure $P$, volume $V$, and temperature $T$ are state functions; heat $Q$ and work $W$ are not.
\end{example}
\begin{remark}[State Property]
    State functions are also called state properties, since they describe the state itself.
\end{remark}
\begin{remark}[State Variables]
    The state variables are the state functions that are easily measured and allow the specification of an equilibrium state, e.g.\ $P$, $V$, $T$. For a simple fluid, only two are independent, the third being fixed by the equation of state.
\end{remark}

\subsubsection{Adiabatic and Diathermal Walls}
\begin{definition}[Mechanical Interaction]
    A mechanical interaction is an exchange of energy between the system and its surroundings as work, e.g.\ through a moving boundary.
\end{definition}
\begin{example}
    A piston pushing a gas (system) in a cylinder.
\end{example}
\begin{definition}[Adiabatic Wall]
    An adiabatic (or adiathermal) wall allows only mechanical interactions: no heat can pass through it.
\end{definition}
\begin{definition}[Diathermal Wall]
    A diathermal wall allows thermal interaction: heat can pass through it.
\end{definition}
\begin{definition}[Thermal Contact]
    Two systems are in thermal contact when separated by a diathermal wall.
\end{definition}
\begin{example}[Testing a Wall]
    \label{ex:testing-wall}
    A rigid cylinder contains two gases separated by a rigid wall; a piston compresses the second gas. Since the first gas cannot do or receive work, its pressure can only change if heat crosses the wall (at fixed volume, a change in $T$ shows up as a change in $P$). If its pressure stays constant, the wall is adiabatic; otherwise it is diathermal.
\end{example}

\subsection{Equilibrium State}
Returning to Example~\ref{ex:testing-wall}, if the two gases are put into thermal contact, after a time their pressures and volumes stop changing: each gas has reached an equilibrium state.
\begin{definition}[Thermal Equilibrium]
    Two systems in thermal contact are in thermal equilibrium when no further changes occur in their state variables.
\end{definition}

\subsubsection{The Zeroth Law of Thermodynamics}
\begin{law}[Zeroth Law of Thermodynamics]
    If systems $A$ and $B$ are each in thermal equilibrium with a third system $C$, then $A$ and $B$ are in thermal equilibrium with each other.
\end{law}
\begin{figure}[H]
    \centering
    \includegraphics[width=0.5\textwidth]{../images/zerothlaw.png}
    \caption{If $A$ and $B$ are in thermal equilibrium upon contact, as are $A$ and $C$, then so are $B$ and $C$.}
    \label{fig:zeroth-law}
\end{figure}

\subsubsection{Temperature}
\begin{definition}[Temperature]
    Temperature is the property that tells whether two systems are in thermal equilibrium: they are if and only if they have the same temperature $T$.
\end{definition}

\subsubsection{Thermodynamic Equilibrium}
\begin{definition}[Thermodynamic Equilibrium]
    Systems in thermodynamic equilibrium must meet the following conditions:
    \begin{itemize}
        \item Thermal equilibrium, with no heat flow (uniform temperature).
        \item Mechanical equilibrium, with no unbalanced forces acting.
        \item Chemical equilibrium, with no net chemical reaction occurring.
        \item Diffusive equilibrium, with no flow of matter from one system to another.
    \end{itemize}
\end{definition}

\subsubsection{PV Diagrams}
\begin{definition}[PV Diagram]
    A PV diagram is a plot of pressure versus volume, on which each equilibrium state of a gas, such as the gases in Example~\ref{ex:testing-wall}, is a single point $(P, V)$.
\end{definition}
\begin{figure}[H]
    \centering
    \includegraphics[width=0.6\textwidth]{../images/pvdiagrams.png}
    \caption{The four basic processes on a PV diagram: isobaric (constant $P$), isochoric (constant $V$), isothermal (constant $T$, $PV = \text{const}$ for an ideal gas) and adiabatic (no heat exchanged, steeper than an isotherm).}
    \label{fig:pvdiagrams}
\end{figure}
\begin{definition}[Isobar]
An isobar is the curve on a PV diagram joining all states with the same pressure, a horizontal line.
\end{definition}
\begin{definition}[Isochore]
An isochore is the curve on a PV diagram joining all states with the same volume, a vertical line.
\end{definition}
\begin{definition}[Adiabat]
    An adiabat is the curve on a PV diagram followed by a system undergoing a reversible adiabatic process (no heat exchanged). It is steeper than an isotherm, since compression raises the temperature.
\end{definition}
\begin{definition}[Isotherm]
    An isotherm is the curve on a PV diagram joining all states with the same temperature.
\end{definition}
\begin{remark}
    To trace an isotherm, put a gas in state $(P, V)$ in thermal equilibrium with a reference system, then separate them and push the piston to reach a new state $(P', V')$. If $(P', V')$ is also in thermal equilibrium with the unchanged reference system, the zeroth law says both states have the same temperature, so they lie on the same isotherm (Figure~\ref{fig:isotherms}).
\end{remark}
\begin{example}[Same Isotherm]
    For an ideal gas, $T = \frac{PV}{nR}$, so two states lie on the same isotherm if and only if $PV = P'V'$. For one mole in the states $(100~\text{kPa}, 24.9~\text{L})$ and $(200~\text{kPa}, 12.45~\text{L})$,
    \[PV = P'V' = 2490~\text{J} \implies T = \frac{2490~\text{J}}{(1~\text{mol})(8.31~\text{J/(mol K)})} \approx 300~\text{K},\]
    so both states lie on the $300$~K isotherm, and two identical gases in these states would be in thermal equilibrium.
\end{example}

\subsection{Equations of State}
\begin{definition}[Equation of State]
    An equation of state is a functional relationship between the state variables of a system in equilibrium, which leaves only two of them independent.
\end{definition}
\begin{example}
    For a gas, an equation of state is
    \[f(P,V,T) = 0.\]
\end{example}
\begin{example}
    For an ideal gas (where there are no intermolecular attractions and the molecules themselves have no volume), the equation of state is determined empirically\footnote{Determined from experiment and observation rather than derived from theory.} to be
    \begin{equation}
        \label{eq:ideal-gas-law}
        PV = nRT,
    \end{equation}
    where $n$ is the number of moles present and $R = 8.314~\text{J/(mol K)}$ is the universal gas constant.
\end{example}
It follows from Equation~\eqref{eq:ideal-gas-law} that the isotherms for an ideal gas are a family of hyperbolae, as shown in Figure~\ref{fig:isotherms}.
\begin{figure}[H]
    \centering
    \includegraphics[width=0.4\textwidth]{../images/isotherms.png}
    \caption{The isotherms for an ideal gas. They form a family of hyperbolae $P = \frac{nRT}{V}$.}
    \label{fig:isotherms}
\end{figure}

\section{Reversible Processes and Work}
\begin{definition}[Process]
    A process is the mechanism of bringing about a change in a system from one equilibrium state to another.
\end{definition}
\begin{example}
    Pushing the piston and compressing the second gas in Example~\ref{ex:testing-wall} is a process.
\end{example}
\begin{definition}[End Points]
    The end points of a process are its initial and final equilibrium states.
\end{definition}

\subsection{Reversible Processes}
\begin{definition}[Reversible Process]
    A process is reversible if, after it occurs, both the system and its surroundings can be returned to their original states.
\end{definition}
\begin{definition}[Quasistatic Process]
    A quasistatic process is one that can be thought of as a succession of equilibrium states.
\end{definition}
\begin{remark}
    Reversible processes are quasistatic processes in which no dissipative forces, such as friction, are present.
\end{remark}

\subsubsection{Isothermal Compression}
\begin{definition}[Reservoir]
    A reservoir (or heat or thermal reservoir) is a system so large that it can absorb or release heat without any significant change in temperature.
\end{definition}
\begin{example}[Reversible Isothermal Compression]
    A gas in contact with a reservoir at $T$ is compressed from $(P_1, V_1)$ to $(P_2, V_2)$ by infinitesimal pushes on a frictionless piston, re-equilibrating at $T$ each time. It passes only through equilibrium states, tracing an isotherm (Figure~\ref{fig:isotherms}).
\end{example}
\begin{example}[Irreversible Compression]
    Pushing the piston in suddenly causes turbulence and finite gradients: the intermediate stages have no well-defined $P$ or $T$, so the process cannot be plotted on a PV diagram.
\end{example}
\begin{remark}
    In a reversible process there are never finite pressure or temperature differences, within the system or with the surroundings, so its direction can be reversed by an infinitesimal change in external conditions.
\end{remark}

\section{The First Law of Thermodynamics}
\subsection{The Work of Thompson and Joule}
Joseph Black, professor of medicine at the University of Glasgow, discovered the following principle:
\begin{principle}[Black's Principle of Calorimetry]
    When two bodies are put in thermal contact, the heat lost by one is equal to the heat gained by the other.
\end{principle}
This suggested that heat was a conserved substance, called caloric.
\begin{definition}[Caloric]
    Caloric is a hypothetical weightless, invisible fluid, once believed to be heat itself, conserved and flowing from hotter to colder bodies.
\end{definition}
Then came Benjamin Thompson, working in the arsenal in Munich, who noticed while boring cannons that friction produced seemingly unlimited heat:
\begin{principle}[Thompson's Observation]
    Heat is not a finite substance such as caloric, and there is a direct relation between the work done and the heat produced.
\end{principle}
About 50 years later, James Joule finished the job with a careful series of experiments, using a tub containing a paddle wheel rotated by weights falling outside the tub. The water inside was stirred (irreversibly, because of turbulence), raising its temperature between two equilibrium states. The walls of the tub were insulating, so the work was performed under adiabatic conditions.
\begin{definition}[Adiabatic Work]
    Adiabatic work is work done on a system enclosed by adiabatic (thermally insulating) walls, so that no heat is exchanged with the surroundings.
\end{definition}
Joule found:
\begin{result}[Mechanical Equivalent of Heat]
    $4.2$~kJ of work is required to raise the temperature of $1$~kg of water by $1$~K.
\end{result}
\begin{result}[Adiabatic Work]
    No matter how the adiabatic work is performed, the same amount of work is required to take a system between the same two equilibrium states.
\end{result}
This second result is exactly the First Law of Thermodynamics, stated in the next subsection.

\subsection{The First Law of Thermodynamics}
\begin{law}[First Law of Thermodynamics]
    If a thermally isolated system is brought from one equilibrium state to another, the work necessary to achieve this change is independent of the process used.
\end{law}
This adiabatic work therefore defines a state function, the internal energy, through $\Delta U = W_{\text{adiabatic}}$. For a non-adiabatic process, the difference defines the heat:
\begin{equation}
    \label{eq:first-law-thermo}
    \Delta U = W + Q,
\end{equation}
where $W$ is the work done on the system and $Q$ the heat added to it.
\begin{definition}[Heat]
    Heat $Q$ is the nonmechanical exchange of energy between the system and the surroundings due to their temperature difference, flowing spontaneously from the hotter to the colder one.
\end{definition}
\begin{remark}[Sign convention for Heat]
    Heat $Q$ is positive when it enters the system and is negative when it leaves the system.
\end{remark}
\begin{remark}[Differential Form]
    For an infinitesimal process, the first law takes the form
    \begin{equation}
        \label{eq:first-law-differential}
        dU = \delta W + \delta Q.
    \end{equation}
    The $\delta$ are used to indicate that their integrals are in general path dependent (inexact differentials\footnote{Exactness, including the test $\left(\frac{\partial M}{\partial y}\right)_x = \left(\frac{\partial N}{\partial x}\right)_y$, is treated in detail in my \href{https://github.com/williamr-yc7/stem-undergrad-notes/blob/86199275ed081fd521621a48faec61bef0cb1069/Winter\%202026/Math325.pdf}{Math 325 notes}, in the section ``First-Order Scalar Equation''.}).
\end{remark}
\begin{proof}
    For work, $\delta W = -P\,dV$. Take two paths between the same states $(P_1, V_1)$ and $(P_2, V_2)$, with $P_1 \neq P_2$. Along path A, expand at $P_1$, then change pressure at constant $V_2$ (no work):
    \[W_A = -P_1(V_2 - V_1).\]
    Along path B, change pressure at constant $V_1$ (no work), then expand at $P_2$:
    \[W_B = -P_2(V_2 - V_1).\]
    Same end states, but $W_A \neq W_B$. In general, for an ideal gas write $\delta W = M\,dV + N\,dT$ with $M = -\frac{nRT}{V}$ and $N = 0$. Exactness would require $\left(\frac{\partial M}{\partial T}\right)_V = \left(\frac{\partial N}{\partial V}\right)_T$, but
    \[\left(\frac{\partial M}{\partial T}\right)_V = -\frac{nR}{V} \neq 0 = \left(\frac{\partial N}{\partial V}\right)_T,\]
    so no function $W(V, T)$ satisfies $dW = \delta W$. Since $dU$ is exact, $\delta Q = dU - \delta W$ is then inexact too.
\end{proof}

\section{Entropy}
As previously mentioned, Sadi Carnot was the first to recognize that heat flowing from hot to cold can be used to produce work, but never all of it. The explanation behind this limit is entropy. The second law of thermodynamics can be restated in terms of entropy:
\begin{law}[The Second Law of Thermodynamics (Entropy Formulation)]
    the entropy of an isolated system never decreases with time. As a result, isolated systems evolve toward a thermodynamic equilibrium state where the entropy is maximized.
\end{law}

\subsection{The Clausius Inequality}
\begin{figure}[H]
    \centering
    \includegraphics[width=0.5\textwidth]{../images/clausius-inequality.png}
    \caption{Clausius' thought experiment: Carnot engines route all heat exchanged by the gas through a single reservoir at $T_0$.}
    \label{fig:clausius-inequality}
\end{figure}
Could an engine run on heat drawn from the surrounding air alone, without burning fuel?
\begin{remark}[Clausius' Thought Experiment]
    Consider Figure~\ref{fig:clausius-inequality}. Heat $\delta Q_1 \frac{T_0}{T_1}$ flows from the principal reservoir (PR) at $T_0$ (e.g.\ the surrounding air) to an auxiliary reservoir at $T_0$. A Carnot engine takes this heat, transforms part of it into work, and transfers the rest, $\delta Q_1$, to an auxiliary reservoir at the gas's temperature $T_1$, which passes it to the gas, changing its state from $(P_1, V_1, T_1)$ to $(P_2, V_2, T_2)$. Each auxiliary reservoir gives exactly what it receives, so it ends unchanged. This continues until the gas is back at its original state. The question is whether heat from a single reservoir can thus be fully converted into work, with no other effect on the surroundings.
\end{remark}
\begin{figure}[H]
    \centering
    \includegraphics[width=0.5\textwidth]{../images/clausius-inequality2.png}
    \caption{The composite system exchanges heat only with the reservoir at $T_0$.}
    \label{fig:clausius-inequality-2}
\end{figure}
After a full cycle, everything in the composite system (Figure~\ref{fig:clausius-inequality-2}) is back to its original state, so $\Delta U = 0$ and the first law gives
\begin{equation}
    W = Q = T_0\sum_i \frac{\delta Q_i}{T_i},
\end{equation}
where $W = \sum_i \delta W_i$ is the total work done and $Q$ the heat drawn from $T_0$. A positive $W = Q$ is forbidden by:
\begin{law}[Kelvin Statement of the Second Law of Thermodynamics]
    No process is possible whose sole result is the absorption of heat from a reservoir and the conversion of this heat into work.
\end{law}
Hence $W = Q \le 0$, and since $T_0 > 0$, taking the steps infinitesimal gives:
\begin{law}[Clausius Inequality]
    For any cyclic process,
    \begin{equation}
        \oint \frac{\delta Q}{T_{\text{surr}}} \leq 0,
    \end{equation}
    where $T_{\text{surr}}$ is the temperature of the auxiliary reservoirs supplying $\delta Q_i$ (not of the system). If the cycle is reversible, $T_{\text{surr}} = T_{\text{sys}}$ and running it in reverse gives $\oint \frac{\delta Q}{T_{\text{surr}}} \geq 0$, so equality holds if and only if the cycle is reversible.
\end{law}
\begin{remark}
    Physically, an engine needs a hot source and a colder sink: a car takes heat from burning fuel, turns part of it into work, and dumps the rest into the colder air. With only one reservoir, there is nowhere to dump heat, so no net work is possible.
\end{remark}

\subsection{Entropy}
Suppose a system is taken along a reversible path $\text{Route}_1$ from an initial state $i$ to a final state $f$ and then back again to the initial state along another reversible path $\text{Route}_2$, completing a reversible cycle. Since the cycle is reversible, the Clausius inequality holds with equality:
\[\oint\frac{\delta Q_{\text{Route}}}{T} = \underset{\text{Route}_{1}}{\int_i^f}\frac{\delta Q_{\text{Route}}}{T} + \underset{\text{Route}_{2}}{\int_f^i}\frac{\delta Q_{\text{Route}}}{T} = 0,\]
therefore
\[\underset{\text{Route}_{1}}{\int_i^f}\frac{\delta Q_{\text{Route}}}{T} = -\underset{\text{Route}_{2}}{\int_f^i}\frac{\delta Q_{\text{Route}}}{T}.\]
Since $\text{Route}_2$ is reversible, traversing it from $i$ to $f$ instead flips the sign of every $\delta Q$, so $\underset{\text{Route}_{2}}{\int_f^i} = -\underset{\text{Route}_{2}}{\int_i^f}$ and
\[\underset{\text{Route}_{1}}{\int_i^f}\frac{\delta Q_{\text{Route}}}{T} = \underset{\text{Route}_{2}}{\int_i^f}\frac{\delta Q_{\text{Route}}}{T}.\]
The integral is the same along any reversible path between $i$ and $f$, so it is path independent (depends only on the end states) and defines a state function.
\begin{definition}[Entropy]
    The entropy $S$ is the state function defined by
    \begin{equation}
        \Delta S = S_f - S_i = \int_i^f \frac{\delta Q_{\text{rev}}}{T},
    \end{equation}
    where the integral is taken along any reversible path\footnote{Irreversibility comes from the temperature gap between system and source. The source must be at $T + dT$ hotter than the system at every moment.}, for which $T_{\text{sys}} = T_{\text{surr}}$. Only entropy differences ($\Delta S$) are defined, in units of J/K, using absolute temperature $T$ in Kelvin. For an infinitesimal reversible process
    \begin{equation}
        dS = \frac{\delta Q_{\text{rev}}}{T}.
    \end{equation}
\end{definition}
\begin{remark}[Entropy Change in Irreversible Processes]
    Since $S$ is a state function, $\Delta S$ for an irreversible process equals $\Delta S$ along any reversible process between the same end states.
\end{remark}
\begin{example}[Entropy Change of Heating Water]
    Consider heating water from $T_i$ to $T_f$, isobarically, through a series of reservoirs, each at $T + dT$ when the water is at $T$. The heat entering the water reversibly at each step is
    \[\delta Q_{\text{rev}} = C_P\,dT,\]
    where $C_P$ is the water's heat capacity at constant pressure. Hence, the entropy change of the water is
    \[dS = \frac{\delta Q_{\text{rev}}}{T} = \frac{C_P\,dT}{T},\]
    or, for constant $C_P$,
    \[\Delta S = C_P\int_{T_i}^{T_f}\frac{dT}{T} = C_P \ln\frac{T_f}{T_i}.\]
\end{example}
\begin{definition}[Isentropic]
    A process is isentropic if the entropy of the system stays constant, $dS = 0$ at every step.
\end{definition}
\begin{remark}[Isentropic Conditions]
    A process that is both adiabatic ($\delta Q = 0$) and reversible is isentropic, since $dS = \frac{\delta Q_{\text{rev}}}{T} = 0$. An adiabatic but irreversible process increases the system's entropy.
\end{remark}

\subsection{The Principle of Increasing Entropy}
\begin{proposition}[Clausius Inequality for a Process]
    For any process between equilibrium states $i$ and $f$,
    \[\int_i^f \frac{\delta Q}{T_{\text{surr}}} \leq \Delta S, \qquad \text{or, infinitesimally,} \qquad \frac{\delta Q}{T_{\text{surr}}} \leq dS,\]
    where the equality holds if the process is reversible, and then $T = T_{\text{surr}}$.\footnote{Here $\delta Q$ is the actual heat, not the heat of the imaginary reversible process used to calculate $dS$.}
\end{proposition}
\begin{proof}
    Consider a cycle consisting of an irreversible path $i$ to $f$ followed by a reversible path $f$ to $i$. The Clausius inequality gives
    \[\int_i^f \frac{\delta Q}{T_{\text{surr}}} + \underset{R}{\int_f^i}\frac{\delta Q}{T_{\text{surr}}} \leq 0 \implies \int_i^f \frac{\delta Q}{T_{\text{surr}}} \leq - \underset{R}{\int_f^i}\frac{\delta Q}{T_{\text{surr}}} = \underset{R}{\int_i^f}\frac{\delta Q}{T} = \Delta S,\]
    where the last equality is the definition of entropy, since the path is reversible.
\end{proof}
For an isolated system $\delta Q = 0$, so $\Delta S \geq 0$:
\begin{law}[Principle of Increasing Entropy]
    \label{poie}
    The entropy of a thermally isolated system increases in any irreversible process and is constant in a reversible process; it never decreases.
\end{law}
\begin{remark}
    Applied to the universe (system plus surroundings), a process can occur spontaneously only if the total entropy increases or stays the same. The principle applies to net changes of the entire system: the entropy of a part of a composite system, or of a non-isolated system, can decrease.
\end{remark}
\begin{figure}[H]
    \centering
    \includegraphics[width=0.5\textwidth]{../images/Entropy-1.png}
    \caption{Although the entropy of a thermally isolated system can only increase or stay constant, the entropy of a part of the system can decrease.}
    \label{fig:entropy-subsystems}
\end{figure}
\begin{example}
    Consider Figure~\ref{fig:entropy-subsystems}, a system consisting of two subsystems which can exchange heat, the whole system being enclosed by adiabatic walls. If initially $T_A > T_B$, a small heat $Q$ naturally flows from subsystem A to subsystem B. This reduces the entropy of subsystem A, $\Delta S_A = -\frac{Q}{T_A}$, while increasing the entropy of subsystem B, $\Delta S_B = \frac{Q}{T_B}$. But the total change is
    \[\Delta S = Q\left(\frac{1}{T_B} - \frac{1}{T_A}\right) > 0,\]
    since $T_A > T_B$, so the entropy of the whole system increases.
\end{example}
Since the entropy of an isolated system can only increase, spontaneous changes continue until no further increase is possible, so equilibrium is reached when $S$ is maximum.
\begin{proposition}[Principle of Entropy Maximization]
    For a system that is thermally isolated from the surroundings, $S$ tends to its maximum, whilst for a system that is totally isolated (thermally and mechanically), $S$ tends to its maximum with $U$ remaining constant.
\end{proposition}

\subsubsection{Entropy and the Arrow of Time}
\begin{remark}[Entropy and the Arrow of Time]
    Most physical laws are unchanged by replacing $t$ with $-t$ (e.g.\ $F = m\frac{d^2\vec{r}}{dt^2}$), so they allow processes to run backward as well as forward. Yet a smashed teacup never reassembles itself: by the principle of increasing entropy (Law~\ref{poie}), the broken pieces have higher entropy than the unbroken cup. Entropy increase thus gives time its direction.
\end{remark}

\section{Statistical Mechanics}
\subsection{Introduction to Probability and Statistics}
\begin{definition}[Microstate]
    A microstate is a specific configuration of the particles in a system.
\end{definition}
\begin{definition}[Macrostate]
    A macrostate is a description of the system in terms of macroscopic properties, such as temperature, pressure, and volume. Different macrostates can be distinguished by a measurement, while different microstates of the same macrostate cannot.
\end{definition}
Assuming all microstates are equally probable, the probability of any macrostate is given by the multiplicity of that macrostate divided by the sum of all the multiplicities:
\begin{equation}
    P(n) = \frac{\Omega(n)}{\sum_i \Omega(i)}.
\end{equation}
For $N$ two-state particles, the multiplicity $\Omega(N, n)$ of the macrostate with $n$ out of the $N$ particles in one of the two states can be calculated using the binomial coefficient:
\begin{equation}
    \Omega(N, n) = \binom{N}{n} = \frac{N!}{n!(N-n)!}.
\end{equation}
There are $N+1$ macrostates and $2^N$ microstates in total:
\begin{equation}
    \sum_{n=0}^{N} \binom{N}{n} = 2^N.
\end{equation}
The distribution has mean $N/2$ and standard deviation $\sigma = \sqrt{N}/2$, so its relative width is
\begin{equation}
    \frac{\sigma}{N} = \frac{1}{2\sqrt{N}},
\end{equation}
meaning the multiplicity becomes sharply peaked at $n = N/2$ as $N$ grows.

For $N \gg 1$, factorials are evaluated with Stirling's approximation:
\begin{equation}
    N! \approx N^N e^{-N}\sqrt{2\pi N}, \qquad \ln N! \approx N\ln N - N.
\end{equation}
