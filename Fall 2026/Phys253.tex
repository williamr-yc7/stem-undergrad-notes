\section{Temperature}
\subsection{Basic Concepts}
\begin{definition}[System]
    A system is the part of the universe under study.
\end{definition}
\begin{definition}[Surroundings]
    The surroundings are everything outside the system that can interact with it.
\end{definition}
\begin{definition}[Boundary]
    The boundary, or also named partition, is the wall separating the system from its surroundings.
\end{definition}
\begin{definition}[Closed System]
    A closed system exchanges energy but not matter with its surroundings.
\end{definition}
\begin{definition}[Equilibrium State]
    An equilibrium state is one in which all the bulk physical properties of the system (i.e pressure, volume...) are uniform throughout the system and do not change with time.
\end{definition}

\subsubsection{State Functions}
\begin{definition}[State Function]
    A state function is a property of a system that depends only on its current equilibrium state, not on how that state was reached. Its change between two states is therefore path independent.
\end{definition}
\begin{example}
    Pressure $P$, volume $V$, and temperature $T$ are state functions; heat $Q$ and work $W$ are not.
\end{example}
\begin{remark}[State Property]
    State functions are also called state properties, since they describe the state itself.
\end{remark}
\begin{remark}[State Variables]
    The state variables are the state functions that are easily measured and allow the specification of an equilibrium state, e.g.\ $P$, $V$, $T$. For a simple fluid, only two are independent, the third being fixed by the equation of state.
\end{remark}

\subsubsection{Adiabatic and Diathermal Walls}
\begin{definition}[Mechanical Interaction]
    A mechanical interaction is an exchange of energy between the system and its surroundings as work, e.g.\ through a moving boundary.
\end{definition}
\begin{example}
    A piston pushing a gas (system) in a cylinder.
\end{example}
\begin{definition}[Adiabatic Wall]
    An adiabatic (or adiathermal) wall allows only mechanical interactions: no heat can pass through it.
\end{definition}
\begin{definition}[Diathermal Wall]
    A diathermal wall allows thermal interaction: heat can pass through it.
\end{definition}
\begin{definition}[Thermal Contact]
    Two systems are in thermal contact when separated by a diathermal wall.
\end{definition}
\begin{example}[Testing a Wall]
    \label{ex:testing-wall}
    A rigid cylinder contains two gases separated by a rigid wall; a piston compresses the second gas. Since the first gas cannot do or receive work, its pressure can only change if heat crosses the wall (at fixed volume, a change in $T$ shows up as a change in $P$). If its pressure stays constant, the wall is adiabatic; otherwise it is diathermal.
\end{example}

\subsection{Equilibrium State}
Returning to Example~\ref{ex:testing-wall}, if the two gases are put into thermal contact, after a time their pressures and volumes stop changing: each gas has reached an equilibrium state.
\begin{definition}[Thermal Equilibrium]
    Two systems in thermal contact are in thermal equilibrium when no further changes occur in their state variables.
\end{definition}

\subsubsection{The Zeroth Law of Thermodynamics}
\begin{law}[Zeroth Law of Thermodynamics]
    If systems $A$ and $B$ are each in thermal equilibrium with a third system $C$, then $A$ and $B$ are in thermal equilibrium with each other.
\end{law}
\begin{figure}[H]
    \centering
    \includegraphics[width=0.5\textwidth]{../images/zerothlaw.png}
    \caption{If $A$ and $B$ are in thermal equilibrium upon contact, as are $A$ and $C$, then so are $B$ and $C$.}
    \label{fig:zeroth-law}
\end{figure}

\subsubsection{Temperature}
\begin{definition}[Temperature]
    Temperature is the property that tells whether two systems are in thermal equilibrium: they are if and only if they have the same temperature $T$.
\end{definition}

\subsubsection{Thermodynamic Equilibrium}
\begin{definition}[Thermodynamic Equilibrium]
    Systems in thermodynamic equilibrium must meet the following conditions:
    \begin{itemize}
        \item Thermal equilibrium, with no heat flow (uniform temperature).
        \item Mechanical equilibrium, with no unbalanced forces acting.
        \item Chemical equilibrium, with no net chemical reaction occurring.
        \item Diffusive equilibrium, with no flow of matter from one system to another.
    \end{itemize}
\end{definition}

\subsubsection{PV Diagrams}
\begin{definition}[PV Diagram]
    A PV diagram is a plot of pressure versus volume, on which each equilibrium state of a gas, such as the gases in Example~\ref{ex:testing-wall}, is a single point $(P, V)$.
\end{definition}
\begin{figure}[H]
    \centering
    \includegraphics[width=0.6\textwidth]{../images/pvdiagrams.png}
    \caption{The four basic processes on a PV diagram: isobaric (constant $P$), isochoric (constant $V$), isothermal (constant $T$, $PV = \text{const}$ for an ideal gas) and adiabatic (no heat exchanged, steeper than an isotherm).}
    \label{fig:pvdiagrams}
\end{figure}
\begin{definition}[Isobar]
An isobar is the curve on a PV diagram joining all states with the same pressure, a horizontal line.
\end{definition}
\begin{definition}[Isochore]
An isochore is the curve on a PV diagram joining all states with the same volume, a vertical line.
\end{definition}
\begin{definition}[Adiabat]
    An adiabat is the curve on a PV diagram followed by a system undergoing a reversible adiabatic process (no heat exchanged). It is steeper than an isotherm, since compression raises the temperature.
\end{definition}
\begin{definition}[Isotherm]
    An isotherm is the curve on a PV diagram joining all states with the same temperature.
\end{definition}
\begin{remark}
    To trace an isotherm, put a gas in state $(P, V)$ in thermal equilibrium with a reference system, then separate them and push the piston to reach a new state $(P', V')$. If $(P', V')$ is also in thermal equilibrium with the unchanged reference system, the zeroth law says both states have the same temperature, so they lie on the same isotherm (Figure~\ref{fig:isotherms}).
\end{remark}
\begin{example}[Same Isotherm]
    For an ideal gas, $T = \frac{PV}{nR}$, so two states lie on the same isotherm if and only if $PV = P'V'$. For one mole in the states $(100~\text{kPa}, 24.9~\text{L})$ and $(200~\text{kPa}, 12.45~\text{L})$,
    \[PV = P'V' = 2490~\text{J} \implies T = \frac{2490~\text{J}}{(1~\text{mol})(8.31~\text{J/(mol K)})} \approx 300~\text{K},\]
    so both states lie on the $300$~K isotherm, and two identical gases in these states would be in thermal equilibrium.
\end{example}

\subsection{Equations of State}
\begin{definition}[Equation of State]
    An equation of state is a functional relationship between the state variables of a system in equilibrium, which leaves only two of them independent.
\end{definition}
\begin{example}
    For a gas, an equation of state is
    \[f(P,V,T) = 0.\]
\end{example}
\begin{example}
    For an ideal gas (where there are no intermolecular attractions and the molecules themselves have no volume), the equation of state is determined empirically\footnote{Determined from experiment and observation rather than derived from theory.} to be
    \begin{equation}
        \label{eq:ideal-gas-law}
        PV = nRT,
    \end{equation}
    where $n$ is the number of moles present and $R = 8.314~\text{J/(mol K)}$ is the universal gas constant.
\end{example}
It follows from Equation~\eqref{eq:ideal-gas-law} that the isotherms for an ideal gas are a family of hyperbolae, as shown in Figure~\ref{fig:isotherms}.
\begin{figure}[H]
    \centering
    \includegraphics[width=0.4\textwidth]{../images/isotherms.png}
    \caption{The isotherms for an ideal gas. They form a family of hyperbolae $P = \frac{nRT}{V}$.}
    \label{fig:isotherms}
\end{figure}

\section{Reversible Processes and Work}
\begin{definition}[Process]
    A process is the mechanism of bringing about a change in a system from one equilibrium state to another.
\end{definition}
\begin{example}
    Pushing the piston and compressing the second gas in Example~\ref{ex:testing-wall} is a process.
\end{example}
\begin{definition}[End Points]
    The end points of a process are its initial and final equilibrium states.
\end{definition}

\subsection{Reversible Processes}
\begin{definition}[Reversible Process]
    A process is reversible if, after it occurs, both the system and its surroundings can be returned to their original states.
\end{definition}
\begin{definition}[Quasistatic Process]
    A quasistatic process is one that can be thought of as a succession of equilibrium states.
\end{definition}
\begin{remark}
    Reversible processes are quasistatic processes in which no dissipative forces, such as friction, are present.
\end{remark}

\subsubsection{Isothermal Compression}
\begin{definition}[Reservoir]
    A reservoir (or heat or thermal reservoir) is a system so large that it can absorb or release heat without any significant change in temperature.
\end{definition}
\begin{example}[Reversible Isothermal Compression]
    A gas in contact with a reservoir at $T$ is compressed from $(P_1, V_1)$ to $(P_2, V_2)$ by infinitesimal pushes on a frictionless piston, re-equilibrating at $T$ each time. It passes only through equilibrium states, tracing an isotherm (Figure~\ref{fig:isotherms}).
\end{example}
\begin{example}[Irreversible Compression]
    Pushing the piston in suddenly causes turbulence and finite gradients: the intermediate stages have no well-defined $P$ or $T$, so the process cannot be plotted on a PV diagram.
\end{example}
\begin{remark}
    In a reversible process there are never finite pressure or temperature differences, within the system or with the surroundings, so its direction can be reversed by an infinitesimal change in external conditions.
\end{remark}

\subsubsection{Thermal Expansion}
\begin{definition}[Linear Coefficient of Thermal Expansion]
    For a small increase in temperature $\Delta T$ at constant tension $F$, the increase in length of a solid bar of length $L$ is $\Delta L = \alpha L \Delta T$, where
    \begin{equation}
        \alpha = \frac{1}{L}\left(\frac{\partial L}{\partial T}\right)_F
    \end{equation}
    is the linear coefficient of thermal expansion, in units of $\text{K}^{-1}$.
\end{definition}
\begin{definition}[Volume Coefficient of Thermal Expansion]
    For a small increase in temperature $\Delta T$ at constant pressure $P$, the increase in volume of a solid or liquid of volume $V$ is $\Delta V = \beta V \Delta T$, where in the limit as $\Delta T$ approaches zero,
    \begin{equation}
        \beta = \frac{1}{V}\left(\frac{\partial V}{\partial T}\right)_P
    \end{equation}
    is the volume coefficient of thermal expansion, in units of $\text{K}^{-1}$.
\end{definition}
\begin{remark}
    Slow thermal expansion is reversible: cooling by $\Delta T$ undoes the volume change from heating by $\Delta T$.
\end{remark}
\begin{remark}
    For solid material, $\alpha$ and $\beta$ are related
    \[\beta \approx 3\alpha.\]
\end{remark}

\subsubsection{Bulk Modulus}
\begin{definition}[Isothermal Bulk Modulus]
    The bulk modulus of a solid or fluid is the stress (pressure increase) divided by the fractional volume decrease, at constant temperature:
    \begin{equation}
        B = -V\left(\frac{\partial P}{\partial V}\right)_T \equiv \frac{1}{\kappa},
    \end{equation}
    where $\kappa$ is the isothermal compressibility and the negative sign ensures that $B$ stays positive.
\end{definition}

\subsection{Work}
\begin{proposition}[Reversible Work]
    The work done on a fluid in a reversible process from $V_1$ to $V_2$ is
    \begin{equation}
        W = -\int_{V_1}^{V_2} P\,dV,
    \end{equation}
    or, for an infinitesimal change,
    \begin{equation}
        \label{eq:reversible-work}
        \delta W = -PdV.
    \end{equation}
\end{proposition}
\begin{proof}
    A gas in the equilibrium state $(P_1, V_1)$ is compressed to $(P_2, V_2)$ by increasing the external force on a frictionless piston of area $A$, so that $F = PA$. If the force is increased infinitesimally so that the piston moves in by $dx$, the volume changes by $dV = -A\,dx$, and the work done on the gas is
    \[\delta W = F\,dx = PA\,dx = -P\,dV.\]
    The minus sign is necessary because during compression the volume decreases, while the work done on the gas is positive.
\end{proof}
\begin{remark}[Quasistatic Piston, Irreversible Process]
    $\delta W = -PdV$ also holds for irreversible processes in which the piston itself moves quasistatically (no finite pressure drop across it), even if irreversibility occurs elsewhere, e.g.\ a slow chemical reaction releasing gas, or gas slowly leaking through a small hole.
\end{remark}
\begin{remark}[Work as an Area]
    The magnitude of the work in a reversible process is the area under its path on a PV diagram (Figure~\ref{fig:pvdiagrams}), so it depends on the path. For a cycle, the enclosed area is the magnitude of the net work, with a sign set by the direction of the cycle.
\end{remark}
\begin{example}[Isothermal Work on an Ideal Gas]
    Along an isotherm, $P = \frac{nRT}{V}$, so
    \[W = -\int_{V_1}^{V_2}\frac{nRT}{V}\,dV = -nRT\ln\frac{V_2}{V_1},\]
    negative for an expansion ($V_2 > V_1$) and positive for a compression.
\end{example}
\begin{remark}[Sign Convention for Work]
    In this course, $W$ is the work done on the system by its surroundings: positive for compression, negative for expansion.
\end{remark}

\subsubsection{Free Expansion}
\begin{definition}[Free Expansion]
    \label{def:free-expansion}
    Free expansion is the expansion of a gas into a vacuum, with no external pressure applied.
    \begin{itemize}
        \item The process is irreversible;
        \item There is no temperature change;
        \item No heat enters the system, since the walls are adiabatic;
        \item No work is done, since there is no external pressure;
    \end{itemize}   
\end{definition}

\subsection{Other Kinds of Work}
\begin{definition}[Work on an Extensible Wire]
    The work done on a wire under tension $F$ when it is stretched by an infinitesimal length $dL$ is
    \begin{equation}
        \label{eq:wire-work}
        \delta W = F\,dL,
    \end{equation}
    positive for an extension ($dL > 0$), consistent with the sign convention for work.
\end{definition}
\begin{remark}
    Work also includes other energy transfers, such as magnetizing a material or running a current through a resistor.
\end{remark}

\subsection{Calculating Work in a Reversible Process}
\begin{remark}[Method]
    To find the change in one state function given the changes in two others, write the equation of state for the wanted one in terms of the other two; since a reversible process is a succession of equilibrium states, it holds at every step.
\end{remark}
\begin{example}[Work on a Compressible Fluid]
    For a fluid taken reversibly from $(P_1, T_1)$ to $(P_2, T_2)$, write $V = V(P, T)$:
    \[dV = \left(\frac{\partial V}{\partial P}\right)_T dP + \left(\frac{\partial V}{\partial T}\right)_P dT = -\frac{V}{B}\,dP + \beta V\,dT,\]
    where $B = -V\left(\frac{\partial P}{\partial V}\right)_T$ and $\beta = \frac{1}{V}\left(\frac{\partial V}{\partial T}\right)_P$, so
    \[W = -\int P\,dV = \int_{P_1}^{P_2}\frac{PV}{B}\,dP - \int_{T_1}^{T_2}P\beta V\,dT.\]
\end{example}


\section{The First Law of Thermodynamics}
\subsection{The Work of Thompson and Joule}
Joseph Black, professor of medicine at the University of Glasgow, discovered the following principle:
\begin{principle}[Black's Principle of Calorimetry]
    When two bodies are put in thermal contact, the heat lost by one is equal to the heat gained by the other.
\end{principle}
This suggested that heat was a conserved substance, called caloric.
\begin{definition}[Caloric]
    Caloric is a hypothetical weightless, invisible fluid, once believed to be heat itself, conserved and flowing from hotter to colder bodies.
\end{definition}
Then came Benjamin Thompson, working in the arsenal in Munich, who noticed while boring cannons that friction produced seemingly unlimited heat:
\begin{principle}[Thompson's Observation]
    Heat is not a finite substance such as caloric, and there is a direct relation between the work done and the heat produced.
\end{principle}
About 50 years later, James Joule finished the job with a careful series of experiments, using a tub containing a paddle wheel rotated by weights falling outside the tub. The water inside was stirred (irreversibly, because of turbulence), raising its temperature between two equilibrium states. The walls of the tub were insulating, so the work was performed under adiabatic conditions.
\begin{definition}[Adiabatic Work]
    \label{def:adiabatic-work}
    Adiabatic work is work done on a system enclosed by adiabatic (thermally insulating) walls, so that no heat is exchanged with the surroundings.
\end{definition}
Joule found:
\begin{result}[Mechanical Equivalent of Heat]
    $4.2$~kJ of work is required to raise the temperature of $1$~kg of water by $1$~K.
\end{result}
\begin{result}[Adiabatic Work]
    No matter how the adiabatic work is performed, the same amount of work is required to take a system between the same two equilibrium states.
\end{result}
This second result is exactly the First Law of Thermodynamics, stated in the next subsection.

\subsection{The First Law of Thermodynamics}
\begin{law}[First Law of Thermodynamics]
    \label{law:first-law}
    If a thermally isolated system is brought from one equilibrium state to another, the work necessary to achieve this change is independent of the process used.
\end{law}

\subsubsection{Internal Energy}
By Law~\ref{law:first-law}, the adiabatic work (Definition~\ref{def:adiabatic-work}) depends only on the end states, whether the process is reversible or not, so it defines a state function:
\begin{definition}[Internal Energy]
    The internal energy $U$ is the state function satisfying
    \[W_{\text{adiabatic}} = U_2 - U_1.\]
\end{definition}
\begin{remark}
    $U$ is the kinetic and potential energy of the individual molecules (random motion, mutual attraction), not of the object as a whole.
\end{remark}

\subsubsection{Heat}
\begin{remark}
    If the system is not thermally isolated, the work between two equilibrium states depends on the path and generally differs from the adiabatic work.
\end{remark}
The difference between $\Delta U$ and $W$ is called heat $Q$.
\begin{equation}
    \label{eq:first-law-thermo}
    \Delta U = W + Q.
\end{equation}
\begin{definition}[Heat]
    Heat $Q$ is the nonmechanical exchange of energy between the system and the surroundings due to their temperature difference, flowing spontaneously from the hotter to the colder one.
\end{definition}
\begin{remark}[Sign convention for Heat]
    Heat $Q$ is positive when it enters the system and is negative when it leaves the system.
\end{remark}
\begin{remark}[Differential Form]
    For an infinitesimal process, the first law takes the form
    \begin{equation}
        \label{eq:first-law-differential}
        dU = \delta W + \delta Q.
    \end{equation}
    The $\delta$ are used to indicate that their integrals are in general path dependent (inexact differentials\footnote{Exactness, including the test $\left(\frac{\partial M}{\partial y}\right)_x = \left(\frac{\partial N}{\partial x}\right)_y$, is treated in detail in my \href{https://github.com/williamr-yc7/stem-undergrad-notes/blob/86199275ed081fd521621a48faec61bef0cb1069/Winter\%202026/Math325.pdf}{Math 325 notes}, in the section ``First-Order Scalar Equation''.}).
\end{remark}
To see why non-adiabatic work is an inexact differential, see Appendix~\ref{subsection:inexact-differential}.
\begin{remark}
    For a compressible fluid, where $\delta W = -PdV$ for an infinitesimal reversible process, the first law becomes
    \[dU = -P\,dV + \delta Q,\]
    or
    \begin{equation}
        \label{eq:first-law-reversible}
        \delta Q = dU + P\,dV \quad \text{(reversible)}.
    \end{equation}
\end{remark}

\subsection{Heat Capacity}
\begin{definition}[Heat Capacity]
    The heat capacity $C$ of a system is the heat needed per unit temperature rise, in the limit of an infinitesimally small rise, for heat added reversibly:
    \begin{equation}
        \label{eq:heat-capacity}
        C = \lim_{\Delta T \to 0}\frac{Q}{\Delta T} = \frac{\delta Q}{dT}.
    \end{equation}
\end{definition}
\begin{remark}
    Heat capacities are measured with irreversible heating, but this agrees with the definition as long as the system re-equilibrates much faster than it is heated, so no significant temperature or pressure gradients form.
\end{remark}
\begin{definition}[Specific Heat]
    The specific heat $c$ is the heat capacity per unit mass
    \begin{equation}
        \label{eq:specific-heat}
        c = \frac{1}{m}\frac{\delta Q}{dT}.
    \end{equation}
\end{definition}
\begin{remark}
    Lower case $c$ is used consistently for specific heat, whether per unit mass, per mole, or another definite quantity.
\end{remark}

\subsubsection{Other Heat Capacities}
\begin{remark}
    Since $Q$ depends on the path, there is a different heat capacity for each path between the same end temperatures. The two most useful are at constant volume and at constant pressure.
\end{remark}
\begin{definition}[Heat capacity at constant volume]
    For an infinitesimal isochoric reversible process, Equation~\eqref{eq:first-law-reversible} gives $\delta Q_V = dU$, thus the heat capacity at constant volume is given
    \begin{equation}
        \label{eq:heat-capacity-volume}
        C_V = \frac{\delta Q_V}{dT} = \left(\frac{\partial U}{\partial T}\right)_V.
    \end{equation}
\end{definition}
\begin{definition}[Enthalpy]
    Let's define a new state function, the enthalpy $H$, as
    \begin{equation}
        \label{eq:enthalpy}
        H=U+PV.
    \end{equation}
\end{definition}
\begin{proposition}[Heat capacity at constant pressure]
    For an infinitesimal isobaric reversible process, the heat capacity at constant pressure is given
    \begin{equation}
        \label{eq:heat-capacity-pressure}
        C_P = \frac{\delta Q_P}{dT} = \left(\frac{\partial H}{\partial T}\right)_P,
    \end{equation}
    where $H$ is the enthalpy.
\end{proposition}
\begin{proof}
    For an infinitesimal process $dH = dU+PdV+VdP$, using the infinitesimal form of the first law Equation~\eqref{eq:first-law-reversible}
    \[dH = \delta Q + VdP.\]
    For a reversible, isobaric process
    \[dH = \delta Q_P.\]
\end{proof}

\subsubsection{Calorimetry}
If $c$ is nearly constant over the temperature range, Equation~\eqref{eq:specific-heat} integrates to
\begin{equation}
    \label{eq:heat-mcdt}
    Q = mc\,\Delta T.
\end{equation}
\begin{example}[Specific Heat of Water]
    It takes $4186$~J to heat $1$~kg of water from $20^\circ$C to $21^\circ$C, so $c = \frac{Q}{m\,\Delta T} = 4186~\text{J/(kg K)}$.
\end{example}
\begin{definition}[Calorimetry]
    Calorimetry is the measurement of an unknown specific heat by heat exchange with a standard substance of known specific heat, usually water, in a thermally insulated container.
\end{definition}
\begin{example}[Specific Heat of a Metal]
    A metal of mass $m_m$ at $T_m$ is placed in water of mass $m_w$ at $T_w$, and both reach $T_f$. By Black's principle, the heat lost by the metal equals the heat gained by the water:
    \[m_m c_m (T_m - T_f) = m_w c_w (T_f - T_w) \implies c_m = \frac{m_w c_w (T_f - T_w)}{m_m (T_m - T_f)}.\]
\end{example}

\subsubsection{Latent Heat}
\begin{remark}
    During a phase transformation at constant pressure, heat is absorbed or released with no change in temperature, so the heat capacity $\frac{\delta Q}{dT}$ is not defined (it would be infinite).
\end{remark}
\begin{definition}[Latent Heat]
    The latent heat $L$ is the heat per unit mass absorbed during a phase transformation at constant temperature and pressure:
    \begin{equation}
        \label{eq:latent-heat}
        L=\frac{Q}{m}\left[\frac{J}{kg}\right].
    \end{equation}
    It is called the latent heat of fusion $L_f$ for melting and of vaporization $L_v$ for boiling.
\end{definition}
\begin{remark}
    Since the transformation occurs at constant pressure, $Q = \Delta H$ (Equation~\eqref{eq:heat-capacity-pressure}), so the latent heat is the enthalpy change per unit mass.
\end{remark}

\section{Entropy}
As previously mentioned, Sadi Carnot was the first to recognize that heat flowing from hot to cold can be used to produce work, but never all of it. The explanation behind this limit is entropy. The second law of thermodynamics can be restated in terms of entropy:
\begin{law}[The Second Law of Thermodynamics (Entropy Formulation)]
    the entropy of an isolated system never decreases with time. As a result, isolated systems evolve toward a thermodynamic equilibrium state where the entropy is maximized.
\end{law}

\subsection{The Clausius Inequality}
\begin{figure}[H]
    \centering
    \includegraphics[width=0.5\textwidth]{../images/clausius-inequality.png}
    \caption{Clausius' thought experiment: Carnot engines route all heat exchanged by the gas through a single reservoir at $T_0$.}
    \label{fig:clausius-inequality}
\end{figure}
Could an engine run on heat drawn from the surrounding air alone, without burning fuel?
\begin{remark}[Clausius' Thought Experiment]
    Consider Figure~\ref{fig:clausius-inequality}. Heat $\delta Q_1 \frac{T_0}{T_1}$ flows from the principal reservoir (PR) at $T_0$ (e.g.\ the surrounding air) to an auxiliary reservoir at $T_0$. A Carnot engine takes this heat, transforms part of it into work, and transfers the rest, $\delta Q_1$, to an auxiliary reservoir at the gas's temperature $T_1$, which passes it to the gas, changing its state from $(P_1, V_1, T_1)$ to $(P_2, V_2, T_2)$. Each auxiliary reservoir gives exactly what it receives, so it ends unchanged. This continues until the gas is back at its original state. The question is whether heat from a single reservoir can thus be fully converted into work, with no other effect on the surroundings.
\end{remark}
\begin{figure}[H]
    \centering
    \includegraphics[width=0.5\textwidth]{../images/clausius-inequality2.png}
    \caption{The composite system exchanges heat only with the reservoir at $T_0$.}
    \label{fig:clausius-inequality-2}
\end{figure}
After a full cycle, everything in the composite system (Figure~\ref{fig:clausius-inequality-2}) is back to its original state, so $\Delta U = 0$ and the first law gives
\begin{equation}
    W = Q = T_0\sum_i \frac{\delta Q_i}{T_i},
\end{equation}
where $W = \sum_i \delta W_i$ is the total work done by the composite system (the opposite of our usual sign convention) and $Q$ the heat drawn from $T_0$. A positive $W = Q$ is forbidden by:
\begin{law}[Kelvin Statement of the Second Law of Thermodynamics]
    No process is possible whose sole result is the absorption of heat from a reservoir and the conversion of this heat into work.
\end{law}
Hence $W = Q \le 0$, and since $T_0 > 0$, taking the steps infinitesimal gives:
\begin{law}[Clausius Inequality]
    For any cyclic process,
    \begin{equation}
        \oint \frac{\delta Q}{T_{\text{surr}}} \leq 0,
    \end{equation}
    where $T_{\text{surr}}$ is the temperature of the auxiliary reservoirs supplying $\delta Q_i$ (not of the system). If the cycle is reversible, $T_{\text{surr}} = T_{\text{sys}}$ and running it in reverse gives $\oint \frac{\delta Q}{T_{\text{surr}}} \geq 0$, so equality holds if and only if the cycle is reversible.
\end{law}
\begin{remark}
    Physically, an engine needs a hot source and a colder sink: a car takes heat from burning fuel, turns part of it into work, and dumps the rest into the colder air. With only one reservoir, there is nowhere to dump heat, so no net work is possible.
\end{remark}

\subsection{Entropy}
Suppose a system is taken along a reversible path $\text{Route}_1$ from an initial state $i$ to a final state $f$ and then back again to the initial state along another reversible path $\text{Route}_2$, completing a reversible cycle. Since the cycle is reversible, the Clausius inequality holds with equality:
\[\oint\frac{\delta Q_{\text{Route}}}{T} = \underset{\text{Route}_{1}}{\int_i^f}\frac{\delta Q_{\text{Route}}}{T} + \underset{\text{Route}_{2}}{\int_f^i}\frac{\delta Q_{\text{Route}}}{T} = 0,\]
therefore
\[\underset{\text{Route}_{1}}{\int_i^f}\frac{\delta Q_{\text{Route}}}{T} = -\underset{\text{Route}_{2}}{\int_f^i}\frac{\delta Q_{\text{Route}}}{T}.\]
Since $\text{Route}_2$ is reversible, traversing it from $i$ to $f$ instead flips the sign of every $\delta Q$, so $\underset{\text{Route}_{2}}{\int_f^i} = -\underset{\text{Route}_{2}}{\int_i^f}$ and
\[\underset{\text{Route}_{1}}{\int_i^f}\frac{\delta Q_{\text{Route}}}{T} = \underset{\text{Route}_{2}}{\int_i^f}\frac{\delta Q_{\text{Route}}}{T}.\]
The integral is the same along any reversible path between $i$ and $f$, so it is path independent (depends only on the end states) and defines a state function.
\begin{definition}[Entropy]
    The entropy $S$ is the state function defined by
    \begin{equation}
        \label{eq:entropy-change}
        \Delta S = S_f - S_i = \int_i^f \frac{\delta Q_{\text{rev}}}{T},
    \end{equation}
    where the integral is taken along any reversible path\footnote{Irreversibility comes from the temperature gap between system and source. The source must be at $T + dT$ hotter than the system at every moment.}, for which $T_{\text{sys}} = T_{\text{surr}}$. Only entropy differences ($\Delta S$) are defined, in units of J/K, using absolute temperature $T$ in Kelvin. For an infinitesimal reversible process
    \begin{equation}
        \label{eq:entropy-differential}
        dS = \frac{\delta Q_{\text{rev}}}{T}.
    \end{equation}
\end{definition}
\begin{remark}[Entropy Change in Irreversible Processes]
    Since $S$ is a state function, $\Delta S$ for an irreversible process equals $\Delta S$ along any reversible process between the same end states.
\end{remark}
\begin{example}[Entropy Change of Heating Water]
    Consider heating water from $T_i$ to $T_f$, isobarically, through a series of reservoirs, each at $T + dT$ when the water is at $T$. The heat entering the water reversibly at each step is
    \[\delta Q_{\text{rev}} = C_P\,dT,\]
    where $C_P$ is the water's heat capacity at constant pressure. Hence, the entropy change of the water is
    \[dS = \frac{\delta Q_{\text{rev}}}{T} = \frac{C_P\,dT}{T},\]
    or, for constant $C_P$,
    \begin{equation}
        \label{eq:centropy-water}
        \Delta S = C_P\int_{T_i}^{T_f}\frac{dT}{T} = C_P \ln\frac{T_f}{T_i}.
    \end{equation}
\end{example}
\begin{definition}[Isentropic]
    A process is isentropic if the entropy of the system stays constant, $dS = 0$ at every step.
\end{definition}
\begin{remark}[Isentropic Conditions]
    A process that is both adiabatic ($\delta Q = 0$) and reversible is isentropic, since $dS = \frac{\delta Q_{\text{rev}}}{T} = 0$. An adiabatic but irreversible process increases the system's entropy.
\end{remark}

\subsection{The Principle of Increasing Entropy}
\begin{proposition}[Clausius Inequality for a Process]
    For any process between equilibrium states $i$ and $f$,
    \[\int_i^f \frac{\delta Q}{T_{\text{surr}}} \leq \Delta S, \qquad \text{or, infinitesimally,} \qquad \frac{\delta Q}{T_{\text{surr}}} \leq dS,\]
    where the equality holds if the process is reversible, and then $T = T_{\text{surr}}$.\footnote{Here $\delta Q$ is the actual heat, not the heat of the imaginary reversible process used to calculate $dS$.}
\end{proposition}
\begin{proof}
    Consider a cycle consisting of an irreversible path $i$ to $f$ followed by a reversible path $f$ to $i$. The Clausius inequality gives
    \[\int_i^f \frac{\delta Q}{T_{\text{surr}}} + \underset{R}{\int_f^i}\frac{\delta Q}{T_{\text{surr}}} \leq 0 \implies \int_i^f \frac{\delta Q}{T_{\text{surr}}} \leq - \underset{R}{\int_f^i}\frac{\delta Q}{T_{\text{surr}}} = \underset{R}{\int_i^f}\frac{\delta Q}{T} = \Delta S,\]
    where the last equality is the definition of entropy, since the path is reversible.
\end{proof}
For an isolated system $\delta Q = 0$, so $\Delta S \geq 0$:
\begin{law}[Principle of Increasing Entropy]
    \label{poie}
    The entropy of a thermally isolated system increases in any irreversible process and is constant in a reversible process; it never decreases.
\end{law}
\begin{remark}
    Applied to the universe (system plus surroundings), a process can occur spontaneously only if the total entropy increases or stays the same. The principle applies to net changes of the entire system: the entropy of a part of a composite system, or of a non-isolated system, can decrease.
\end{remark}
\begin{figure}[H]
    \centering
    \includegraphics[width=0.5\textwidth]{../images/Entropy-1.png}
    \caption{Although the entropy of a thermally isolated system can only increase or stay constant, the entropy of a part of the system can decrease.}
    \label{fig:entropy-subsystems}
\end{figure}
\begin{example}
    Consider Figure~\ref{fig:entropy-subsystems}, a system consisting of two subsystems which can exchange heat, the whole system being enclosed by adiabatic walls. If initially $T_A > T_B$, a small heat $Q$ naturally flows from subsystem A to subsystem B. This reduces the entropy of subsystem A, $\Delta S_A = -\frac{Q}{T_A}$, while increasing the entropy of subsystem B, $\Delta S_B = \frac{Q}{T_B}$. But the total change is
    \[\Delta S = Q\left(\frac{1}{T_B} - \frac{1}{T_A}\right) > 0,\]
    since $T_A > T_B$, so the entropy of the whole system increases.
\end{example}
Since the entropy of an isolated system can only increase, spontaneous changes continue until no further increase is possible, so equilibrium is reached when $S$ is maximum.
\begin{proposition}[Principle of Entropy Maximization]
    For a system that is thermally isolated from the surroundings, $S$ tends to its maximum, whilst for a system that is totally isolated (thermally and mechanically), $S$ tends to its maximum with $U$ remaining constant.
\end{proposition}

\subsubsection{Entropy and the Arrow of Time}
\begin{remark}[Entropy and the Arrow of Time]
    Most physical laws are unchanged by replacing $t$ with $-t$ (e.g.\ $F = m\frac{d^2\vec{r}}{dt^2}$), so they allow processes to run backward as well as forward. Yet a smashed teacup never reassembles itself: by the principle of increasing entropy (Law~\ref{poie}), the broken pieces have higher entropy than the unbroken cup. Entropy increase thus gives time its direction.
\end{remark}

\subsection{Entropy--Temperature Diagrams}
\begin{remark}
    The thermodynamic state of a system can be specified by any pair of independent state variables, for example on a PV diagram or a TS diagram.
\end{remark}
\begin{remark}[Reversible Adiabats and Isotherms]
    By Equation~\eqref{eq:entropy-differential}, a reversible adiabatic process is isentropic, so on a TS diagram (with $T$ vertical) it is a vertical line, while a reversible isothermal process is a horizontal line.
\end{remark}
\begin{figure}[H]
    \centering
    \includegraphics[width=0.5\textwidth]{../images/tsdiagram.png}
    \caption{TS diagram for a Carnot cycle: a rectangle of two isotherms and two adiabats.}
    \label{fig:ts-diagram}
\end{figure}
\begin{remark}[Area of a TS Cycle]
    Since $\delta Q_{\text{rev}} = T\,dS$, the area enclosed by a reversible cycle on a TS diagram is the net heat absorbed, $\oint T\,dS$, which by the first law equals the net work done by the system.
\end{remark}

\subsection{The Thermodynamic Identity}
Combining the first and second laws gives the most important equation in thermodynamics.
\begin{proposition}[Thermodynamic Identity]
    The thermodynamic identity, or central equation of thermodynamics, states that for any process between equilibrium states,
    \begin{equation}
        \label{eq:thermodynamic-identity}
        dU = T\,dS - P\,dV.
    \end{equation}
\end{proposition}
\begin{proof}
    For a reversible process, substitute $\delta W = -PdV$ (Equation~\eqref{eq:reversible-work}) and $\delta Q = T\,dS$ (Equation~\eqref{eq:entropy-differential}) into the first law (Equation~\eqref{eq:first-law-differential}). Since the result involves only state functions, it holds between any two neighbouring equilibrium states, so for irreversible processes too.
\end{proof}
\begin{remark}
    The whole of thermodynamics is dependent on this equation, just as the whole of mechanics is dependent on Newton's laws.
\end{remark}

\subsection{Other Examples of Entropy Calculations}
\subsubsection{Free Expansion of an Ideal Gas}
Free expansion (Definition~\ref{def:free-expansion}) is irreversible, so to apply Equation~\eqref{eq:entropy-change} we imagine a reversible isothermal expansion between the same end states, e.g.\ letting the gas expand slowly in thermal contact with a reservoir at $T$. Since $T$ is constant, $dU = C_V\,dT = 0$ for an ideal gas, so the first law and $PV = nRT$ give
\[dS = \frac{\delta Q_{\text{rev}}}{T} = \frac{dU + P\,dV}{T} = \frac{nR}{V}\,dV \implies \Delta S = nR\int_{V_i}^{V_f}\frac{dV}{V} = nR\ln\frac{V_f}{V_i},\]
which gives $\Delta S = nR\ln 2$ when the volume doubles.
\begin{remark}
    Although $Q = 0$ in the actual free expansion, $\Delta S > 0$, as expected for an adiabatic irreversible process.
\end{remark}

\subsubsection{Entropy of an Ideal Gas in General}
For an ideal gas, $U = U(T)$, since there are no intermolecular forces, so $dU = C_V\,dT$. The thermodynamic identity (Equation~\eqref{eq:thermodynamic-identity}) and the ideal gas law (Equation~\eqref{eq:ideal-gas-law}) give
\[T\,dS = C_V\,dT + P\,dV = C_V\,dT + \frac{nRT}{V}\,dV.\]
Dividing by $nT$ and using molar quantities, $s = S/n$, $c_v = C_V/n$ and $v = V/n$,
\[ds = \frac{c_v}{T}\,dT + \frac{R}{v}\,dv.\]
For constant $c_v$, integrating from a reference state $(T_0, v_0)$, usually taken at standard temperature and pressure ($T_0 = 273.15$~K, $P_0 = 101.325$~kPa), gives
\begin{equation}
    \label{eq:ideal-gas-entropy}
    s = c_v\ln\frac{T}{T_0} + R\ln\frac{v}{v_0} + s_0,
\end{equation}
where $s_0$ is the molar entropy of the reference state, which cancels when entropy differences are taken.

\subsubsection{Entropy Change of the Universe when Heating Water}
\begin{example}[Single-Stage Heating]
    Heating water from $T_i = 293$~K on a reservoir at $T_f = 373$~K, the reservoir loses $Q = C_P(T_f - T_i)$ at constant temperature $T_f$, so
    \[\Delta S_{\text{universe}} = \Delta S_{\text{water}} + \Delta S_{\text{reservoir}},\]
    where $\Delta S_{\text{water}}$ comes from Equation~\eqref{eq:centropy-water}, so
    \[\Delta S_{\text{universe}} = C_P\left[\ln\frac{T_f}{T_i} - \frac{T_f - T_i}{T_f}\right] = C_P(0.241 - 0.214) = 0.027\,C_P > 0.\]
    The water's entropy gain only slightly exceeds the reservoir's loss, but no choice of parameters makes the total negative.
\end{example}
\begin{example}[Two-Stage Heating]
    Heating in two stages, first on a reservoir at $323$~K, then at $373$~K, leaves $\Delta S_{\text{water}}$ unchanged (same end states), while
    \[\Delta S_{\text{universe}} = C_P\left[\ln\frac{373}{293} - \frac{323-293}{323} - \frac{373-323}{373}\right] = 0.014\,C_P,\]
    about half.
\end{example}
\begin{remark}
    More stages bring the process closer to reversibility.
\end{remark}
\begin{example}[Reversible Heating with a Carnot Engine]
    An infinitesimal Carnot engine runs between the reservoir at $T_f = 373$~K and the water at $T$, with cycles so small that only differential heat and work are exchanged per cycle. To deliver $\delta Q_2 = C_P\,dT$ to the water, it draws $\delta Q_1 = \frac{T_f}{T}\,\delta Q_2$ from the reservoir and outputs work $\delta W$. Then
    \[\Delta S_{\text{reservoir}} = -\int_{T_i}^{T_f}\frac{\delta Q_1}{T_f} = -\int_{T_i}^{T_f}\frac{C_P\,dT}{T} = -C_P\ln\frac{T_f}{T_i} = -\Delta S_{\text{water}},\]
    so $\Delta S_{\text{universe}} = 0$, as it must be for a reversible process.
\end{example}


\section{Statistical Mechanics}
\begin{remark}
    Many thermodynamic processes happen in one direction but never the reverse, because every irreversible process increases the total entropy. Classical thermodynamics states this but does not explain why; statistical mechanics does.
\end{remark}
\begin{definition}[Statistical Thermodynamics]
    Statistical thermodynamics is the study of how systems store energy, by counting all the ways that energy can be arranged among their constituents.
\end{definition}
\begin{definition}[Combinatorics]
    Combinatorics is the mathematics of counting ways of arranging things.
\end{definition}
\subsection{Introduction to Probability and Statistics}
\begin{definition}[Microstate]
    A microstate is a specific configuration of the particles in a system.
\end{definition}
\begin{definition}[Macrostate]
    A macrostate is a description of the system in terms of macroscopic properties, such as temperature, pressure, and volume. Different macrostates can be distinguished by a measurement, while different microstates of the same macrostate cannot.
\end{definition}
\begin{definition}[Multiplicity]
    The multiplicity $\Omega$ of a macrostate is the number of microstates corresponding to it. For $N$ two-state particles, the multiplicity of the macrostate with $n$ particles in one of the two states is the binomial coefficient
    \begin{equation}
        \Omega(N, n) = \binom{N}{n} = \frac{N!}{n!(N-n)!}.
    \end{equation}
\end{definition}
\begin{example}[Three Coins]
    Flipping three distinguishable fair coins gives $2^3 = 8$ equally probable microstates (e.g.\ HHT). The macrostate is the number of heads: $\Omega(3) = 1$ (HHH), while $\Omega(2) = 3$ (HHT, HTH, THH). Knowing the microstate gives the macrostate, but not the reverse.
\end{example}
\begin{definition}[Probability of a macrostate]
    Assuming all microstates are equally probable, the probability of a macrostate is its multiplicity divided by the total number of microstates:
    \begin{equation}
        P(n) = \frac{\Omega(n)}{\sum_i \Omega(i)}.
    \end{equation}
\end{definition}
\begin{proposition}[Total Amount of Microstates]
    There are $N+1$ macrostates and $2^N$ microstates in total:
    \begin{equation}
        \sum_{n=0}^{N} \binom{N}{n} = 2^N.
    \end{equation}
\end{proposition}

\subsubsection{The Multiplicity Function}
\begin{definition}[Mean of a Binomial Distribution]
    For $N$ independent trials, each succeeding with probability $p$, the mean number of successes is
    \[\bar{n} = Np.\]
\end{definition}
\begin{definition}[Variance of a Binomial Distribution]
    The variance measures the spread of a distribution around its mean, $\sigma^2 = \overline{(n - \bar{n})^2}$. For a binomial distribution,
    \[\sigma^2 = Np(1-p).\]
\end{definition}
\begin{definition}[Standard Deviation]
    The standard deviation is the square root of the variance,
    \[\sigma = \sqrt{Np(1-p)}.\]
\end{definition}
For fair coins, $p = \frac{1}{2}$, so $\bar{n} = \frac{N}{2}$, $\sigma^2 = \frac{N}{4}$, and the standard deviation normalized to the number of coins is
\[\frac{\sigma}{N} = \frac{1}{2\sqrt{N}}.\]
\begin{remark}
    The normalized multiplicity function gets narrower as $N$ increases, becoming sharply peaked at $n = \frac{N}{2}$.
\end{remark}
\begin{remark}[Stirling's Approximation]
    For $N \gg 1$, factorials are evaluated with Stirling's approximation:
    \begin{equation}
        N! \approx N^N e^{-N}\sqrt{2\pi N}, \qquad \ln N! \approx N\ln N - N.
    \end{equation}
\end{remark}

\subsection{Microscopic View of Entropy}
\subsubsection{The Two-State Paramagnet}
\begin{definition}[Paramagnet]
    A paramagnet is a material whose constituent particles act like tiny magnetic dipoles, due to `unpaired' electrons or atomic nuclei, which tend to align parallel to any externally applied magnetic field and do not interact significantly with neighbouring dipoles.
\end{definition}
\begin{definition}[Dipoles]
    Here, dipoles are the individual magnetic particles, which can be single electrons, groups of electrons in an atom, or atomic nuclei.
\end{definition}
\begin{remark}[Two-State Paramagnet]
    Quantum mechanics limits the projection of each dipole onto the axis defined by $\vec{B}$ to only two states:
    \begin{itemize}
        \item Up state: aligned with $\vec{B}$.
        \item Down state: anti-aligned with $\vec{B}$.
    \end{itemize}
\end{remark}
\begin{definition}[M-Meter]
    An M-meter measures the magnetization $M$ of the material, proportional to $N_\uparrow - N_\downarrow$, the paramagnet's analogue of a ``heads meter'' for coins.
\end{definition}
\begin{remark}
    In zero (or weak) $\vec{B}$, the up/down state of each dipole is effectively an independent $50/50$ `coin toss', so we expect to measure $M = 0$ ($N_\uparrow = N_\downarrow = \frac{N}{2}$).
\end{remark}
This system has one macrostate for each possible value of $N_\uparrow$, from $0$ to $N$. The multiplicity of each macrostate is
\[\Omega(N_\uparrow) = \binom{N}{N_\uparrow} = \frac{N!}{N_\uparrow!\,N_\downarrow!}.\]

\subsubsection{The Einstein Solid}
Consider a collection of microscopic systems that can each store any number of energy ``units'', all of the same size.
\begin{definition}[Quantum Harmonic Oscillator]
    A quantum harmonic oscillator is a particle in a potential $\frac{1}{2}k_s x^2$, where $k_s$ is the ``spring constant''.
\end{definition}
\begin{remark}
    Its allowed energies are equally spaced by $hf$, where $h = 6.63 \times 10^{-34}~\text{J s}$ is Planck's constant and $f = \frac{1}{2\pi}\sqrt{\frac{k_s}{m}}$ is its natural frequency.
\end{remark}
\begin{definition}[Einstein Solid]
    An Einstein solid is a model of a solid as a collection of $N$ identical, independent quantum harmonic oscillators, three per atom (one for each direction of vibration).
\end{definition}
\begin{example}
    Examples of quantum oscillators include the vibrational motions of diatomic and polyatomic gas molecules, and the oscillation of atoms in a solid.
\end{example}
\begin{figure}[H]
    \centering
    \includegraphics[width=0.5\textwidth]{../images/einsteinsolid.jpg}
    \caption{3d unrealistic view of an einstein solid with springs between atoms}
    \label{fig:einstein-solid}
\end{figure}
\begin{remark}
    In a 3-dimensional solid, each atom can oscillate in 3 independent directions, so if there are N oscillators, there are only N/3 atoms.
\end{remark}

\section{Proofs}
\label{section:proofs}
\subsection{Inexact differentials}
\label{subsection:inexact-differential}
\begin{proof}
    For work, $\delta W = -PdV$. Take two paths between the same states $(P_1, V_1)$ and $(P_2, V_2)$, with $P_1 \neq P_2$. Along path A, expand at $P_1$, then change pressure at constant $V_2$ (no work):
    \[W_A = -P_1(V_2 - V_1).\]
    Along path B, change pressure at constant $V_1$ (no work), then expand at $P_2$:
    \[W_B = -P_2(V_2 - V_1).\]
    Same end states, but $W_A \neq W_B$. In general, for an ideal gas write $\delta W = M\,dV + N\,dT$ with $M = -\frac{nRT}{V}$ and $N = 0$. Exactness would require $\left(\frac{\partial M}{\partial T}\right)_V = \left(\frac{\partial N}{\partial V}\right)_T$, but
    \[\left(\frac{\partial M}{\partial T}\right)_V = -\frac{nR}{V} \neq 0 = \left(\frac{\partial N}{\partial V}\right)_T,\]
    so no function $W(V, T)$ satisfies $dW = \delta W$. Since $dU$ is exact, $\delta Q = dU - \delta W$ is then inexact too.
\end{proof}
