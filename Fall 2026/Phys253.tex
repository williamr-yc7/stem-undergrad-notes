\section{Statistical Mechanics}
\subsection{Introduction to Probability and Statistics}
  \subsubsection{Microstates, Macrostates and Multiplicity}
    \begin{definition}[Microstate]
        A microstate is a specific configuration of the particles in a system.
    \end{definition}
    \begin{definition}[Macrostate]
        A macrostate is a description of the system in terms of macroscopic properties, such as temperature, pressure, and volume.    
    \end{definition}
    \begin{definition}[Multiplicity]
        The multiplicity $\Omega$ of a macrostate is the number of microstates corresponding to that macrostate.    
    \end{definition}
    The probability of any particular macrostate can be written as
    \begin{equation}
        P(n) = \frac{\Omega(n)}{\Omega(all)}.
    \end{equation}
