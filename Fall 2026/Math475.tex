\subsubsection{Scalar \& Vector Fields}
\begin{definition}[Scalar Field]
    A scalar field assigns a number to each point of space.
\end{definition}
\begin{definition}[Vector Field]
    A vector field assigns a vector (magnitude and direction) to each point of space.
\end{definition}

\subsubsection{The Gradient}
\begin{definition}[Gradient]
    The gradient collects the partial derivatives into a vector
    \begin{equation}
        \label{eq:gradient}
        \vec{\nabla} f(\vec{x}) = \left(\partial_1 f,\partial_2 f,\dots,\partial_n f\right)(\vec{x}).
    \end{equation}
\end{definition}
The gradient is the vector that converts a small step $\vec{h}$ into the resulting small change in $f$, by a dot product
\[f(\vec{x}+\vec{h}) = f(\vec{x}) + \vec{\nabla} f(\vec{x})\cdot \vec{h} + o(\|\vec{h}\|).\]
\begin{definition}[Directional Derivative]
    The directional derivative of $f$ along a unit vector $\hat{v}$ is
    \begin{equation}
        \label{eq:directional-derivative}
        D_{\hat{v}}f(\vec{x}) = \lim_{t\to0}\frac{f(\vec{x}+t\hat{v})-f(\vec{x})}{t}.
    \end{equation}
\end{definition}
\begin{proposition}[Directional Derivative via the Gradient]
    \begin{equation}
        \label{eq:directional-gradient}
        D_{\hat{v}}f(\vec{x}) = \vec{\nabla} f(\vec{x})\cdot \hat{v} = \|\vec{\nabla} f(\vec{x})\|\cos\theta,
    \end{equation}
    where $\theta$ is the angle between $\hat{v}$ and $\vec{\nabla} f(\vec{x})$.
\end{proposition}

\subsubsection{Level Sets}
\begin{definition}[Level Set]
    The level set of $f$ at height $c$ is
    \begin{equation}
        \label{eq:level-set}
        \{\vec{x}:f(\vec{x})=c\}.
    \end{equation}
\end{definition}
\begin{definition}[Equipotential]
    Level sets of the potential are called equipotentials.
\end{definition}
\begin{proposition}[Gradient Is Normal to Level Sets]
    The gradient is perpendicular to the level set.
\end{proposition}

\subsubsection{Critical Points}
\begin{definition}[Critical Point]
    A critical point is a point where $\vec{\nabla} f(\vec{x}) = \vec{0}$: no direction of increase, the linear approximation is flat.
\end{definition}
\begin{remark}
    Local maxima and minima of a differentiable function occur only at critical points, but the converse fails, and the failure has a name: a critical point that is neither a maximum nor a minimum is a saddle point.
\end{remark}
\begin{definition}[Hessian]
    The Hessian is the symmetric matrix of second partials
    \begin{equation}
        \label{eq:hessian}
        H_{ij} = \partial_i\partial_j f.
    \end{equation}
\end{definition}
\begin{proposition}[Second Derivative Test]
    Near a critical point $\vec{x}$, the behavior is governed by the signs of the eigenvalues of $H$:
    \begin{itemize}
        \item all eigenvalues $>0$: local minimum (a bowl);
        \item all eigenvalues $<0$: local maximum (a dome);
        \item eigenvalues of both signs: a saddle, increasing in some directions and decreasing in others, like the point at the center of a mountain pass.
    \end{itemize}
    If some eigenvalue is $0$ and the others share a sign, the test is inconclusive.
\end{proposition}

\subsubsection{Line Integrals, Work \& Conservative Fields}
\begin{definition}[Line Integral]
    The line integral of a vector field $\vec{F}$ along a curve $C$ parametrized by $\vec{\gamma}(t)$, $a\le t\le b$, is
    \begin{equation}
        \label{eq:line-integral}
        \int_C \vec{F}\cdot d\vec{l} = \int_a^b \vec{F}(\vec{\gamma}(t))\cdot\vec{\gamma}'(t)\,dt.
    \end{equation}
\end{definition}
\begin{definition}[Work]
    The work done by a force $\vec{F}$ along a path $C$ is
    \begin{equation}
        \label{eq:work}
        W = \int_C \vec{F}\cdot d\vec{l}.
    \end{equation}
\end{definition}
\begin{definition}[Conservative Field]
    A vector field is conservative if its line integral between two points depends only on the endpoints, not on the path. Equivalently, $\oint_C \vec{F}\cdot d\vec{l} = 0$ for every closed curve $C$.
\end{definition}

\subsubsection{Divergence}
\begin{definition}[Divergence]
    The divergence of a vector field is the scalar field
    \begin{equation}
        \label{eq:divergence}
        \vec{\nabla}\cdot\vec{F} = \partial_1 F_1 + \partial_2 F_2 + \partial_3 F_3.
    \end{equation}
\end{definition}
\begin{theorem}[Divergence Theorem]
    For a volume $V$ bounded by a closed surface $S$ with outward normal,
    \begin{equation}
        \label{eq:divergence-theorem}
        \oiint_S \vec{F}\cdot d\vec{a} = \iiint_V(\vec{\nabla}\cdot\vec{F})\,d\tau.
    \end{equation}
\end{theorem}

\subsubsection{The Laplacian}
\begin{definition}[Laplacian]
    The Laplacian composes the two first-order operators, the divergence of the gradient
    \begin{equation}
        \label{eq:laplacian}
        \Delta f = \nabla^2 f = \vec{\nabla}\cdot(\vec{\nabla} f) = \partial_1^2 f + \partial_2^2 f + \partial_3^2 f = \operatorname{tr} H,
    \end{equation}
    where $\operatorname{tr} H$ is the trace of the Hessian.
\end{definition}
\begin{definition}[Harmonic Function]
    A function $f\in C^2(\Omega)$ on an open set $\Omega$ is harmonic in $\Omega$ if
    \begin{equation}
        \label{eq:harmonic}
        \Delta f = 0
    \end{equation}
    at every point of $\Omega$.
\end{definition}

\subsection{Linear Algebra}
\begin{definition}[Linearity]
    \label{def:linearity}
    An operator $L$ is \emph{linear} if for all functions $u, v$ and constants $c$,
    \[L(u+cv) = L(u)+cL(v)\]
\end{definition}
\begin{definition}[Orthogonal basis]
    A basis $\{v_1,\dots,v_N\}$ of $\R^N$ is an orthogonal basis if $v_i\cdot v_j = 0, \forall i\neq j$.
\end{definition}
\begin{definition}[Inner product, norm, orthogonality]
    For real-valued functions $f,g$ on $(a,b)$ whose integrals below are finite, the inner product is 
    \[(f,g) := \int_a^b f(x)g(x)dx.\]
    The norm is 
    \[||f|| := (f,f)^{1/2},\quad ||f||^2=\int_a^b f^2 dx.\]
    Two functions are orthogonal on $(a,b)$ if $(f,g) = 0$.
\end{definition}
\begin{remark}
    For complex-valued functions the inner product is $(f,g) := \int f\overline{g}dx$.
\end{remark}
\begin{definition}[Projection]
    The projection of $f$ onto $g\neq 0$ is
    \[Proj_gf := \frac{(f,g)}{||g||^2}g.\]
\end{definition}

\subsection{Analysis}
\begin{definition}[Maximum and minimum]
    $\underset{x\in A}{\max} f(x)$ is the largest value $f$ takes on $A$, and $\underset{x\in A}{\min} f(x)$ the smallest. The subscript says which inputs $x$ we look over.
\end{definition}
\begin{definition}[Supremum and infimum]
    $\underset{x\in A}{\sup} f(x)$ is the smallest number that is $\geq$ every value of $f$ on $A$, and $\underset{x\in A}{\inf} f(x)$ the largest number that is $\leq$ every value.
\end{definition}
\begin{example}
    They exist even when no maximum or minimum is reached: for $f(x)=1-e^{-x}$ on $[0,\infty)$, $\sup f = 1$ but $f$ never equals $1$.
\end{example}
\begin{remark}
    Functions that are bounded always carry a supremum, but may not have a maximum (like the example above). However, if the function has a maximum, then its supremum is always the maximum.
\end{remark}

\subsection{Special Functions}
\begin{definition}[Dirac Delta]
    \label{def:dirac-delta}
    The Dirac delta $\delta_a$ at a point $a$ is the object with
    \begin{equation}
        \label{eq:dirac-delta}
        \intinf\delta_a(x)\,dx = 1 \quad\text{and}\quad \delta_a(x) = 0 \text{ for } x\neq a.
    \end{equation}
    The case $a=0$ is written $\delta_0$.
\end{definition}
\begin{definition}[Heaviside Function]
    The Heaviside function, named after Oliver Heaviside (1850--1925), is the step function that jumps from $0$ to $1$ at $x=a$:\footnote{The case $a=0$ is written $H(x)$.}
    \begin{equation}
        \label{eq:heaviside}
        H(x-a) :=\begin{cases}0, & x<a\\1, & x\geq a\end{cases}.
    \end{equation}
\end{definition}

\subsection{Partial Differential Equations}
\begin{definition}[Partial Differential Equation]
    A partial differential equation (PDE) is a differential equation relating an unknown function $u$ of several variables, its partial derivatives, and the independent variables.
    \[F(\text{independent variables}, u, \text{partial derivatives of }u) = 0.\]
\end{definition}
\begin{remark}
    Physics rarely hands us a function $u$ directly. It gives a law for how $u$ changes. Example: heat flows from hot to cold, so the temperature $u(x,t)$ in a room changes in time wherever it varies in space. Recovering $u$ from such a relation is the whole subject.
\end{remark}

\subsubsection{Notation}
\begin{remark}
    For the rest of this notes, we will use these terms for a general function $u(x,y,t)$
    \[\nabla u = (u_x,u_y,u_z) = \left(\frac{\partial u}{\partial x}, \frac{\partial u}{\partial y}, \frac{\partial u}{\partial t}\right),\]
    and
    \[\Delta u = u_{xx}+u_{yy}+u_{tt} = \frac{\partial^2 u}{\partial x^2} + \frac{\partial^2 u}{\partial y^2} + \frac{\partial^2 u}{\partial t^2}.\]
\end{remark}

\subsection{Classification}
\subsubsection{Order}
\begin{definition}[Order]
    The order of a PDE is the order of the highest derivative appearing in it.
\end{definition}
\begin{example}
    $u_{xx}+u_{t} = 0$ is second order, and $u_t + u u_x = 0$ is first order.
\end{example}

\subsubsection{Linearity}
\begin{definition}[Linear PDE]
    A PDE is linear if it can be written as $L(u) = f(x)$, where $f$ does not involve $u$ and $L$ satisfies linearity (Def.\ref{def:linearity}).
\end{definition}
\begin{remark}[Practical test]
    Best way to determine if a PDE is linear is to check the following rules:
    \begin{itemize}
        \item $u$ and its derivatives appear only to the first power (no $u^2$, $(u_x)^2$),
        \item they are never multiplied together (no $u\,u_x$, $u_x u_y$),
        \item they never sit inside a function (no $\sin u$, $e^u$, $\ln u$).
    \end{itemize}
\end{remark}
\begin{example}
    $u_{xx} + xy\,u_y + e^x u = \cos x$ is linear. Nonlinear: $u\,u_x + u_y = 0$ (product), $u_{xx} + (u_y)^2 = 0$ (power), $u_{xx} + \sin u = 0$ (inside a function).
\end{example}
\begin{definition}[Degrees of nonlinearity]\leavevmode
    \begin{itemize}
        \item \textbf{Semilinear:} highest derivatives appear linearly, with coefficients depending only on the independent variables.\\Example: $u_{xx} + \sin u = 0$.
        \item \textbf{Quasilinear:} highest derivatives appear linearly, but coefficients may depend on $u$ and its lower derivatives.\\Example: $u_t + u\,u_x = 0$.
        \item \textbf{Fully nonlinear:} highest derivatives appear nonlinearly.\\Example: $(u_x)^2 + (u_y)^2 = 1$.
    \end{itemize}
\end{definition}

\subsubsection{Homogeneity}
\begin{definition}[Homogeneous]
    A linear PDE $L(u) = f(x)$ is homogeneous if $f \equiv 0$.
\end{definition}
\begin{example}
    $u_t + a u_x = x^3$ is linear with $L(u) = u_t + a u_x$ and $f(x) = x^3$, so it is inhomogeneous.
\end{example}

\subsection{Classical Solutions to a PDE}
\begin{definition}[Classical solution]
    A classical solution to a PDE, with order $k$ on a domain $\Omega \subset\R^N$, is a function $u \in C^k(\Omega)$ satisfying the PDE at every point of $\Omega$.
\end{definition}
\begin{example}
    For any function $\phi \in C^1$ of one variable, $u(x,t) = \phi(x-ct)$ solves $u_t+cu_x=0$, we can check this by the chain rule, $u_t = -c\phi'(x-ct)$, $u_x = \phi'(x-ct)$, then $u_t+cu_x = -c\phi'+c\phi' = 0$.
\end{example}

\subsection{General Solutions to a PDE}
Recall from the \href{https://github.com/williamr-yc7/stem-undergrad-notes/blob/86199275ed081fd521621a48faec61bef0cb1069/Winter\%202026/Math325.pdf}{Math 325 notes} that the general solution of an ODE contains arbitrary constants.
\begin{example}
    The general solution of $y' = y$ is $y(t) = Ce^t$, where $C$ is an arbitrary constant.
\end{example}
\begin{remark}
    The general solution of a PDE contains arbitrary functions.
\end{remark}
\begin{example}
    Find all $u : \R^2 \to \R$ with $u_{xx} = 0$. Since $(u_x)_x = 0$, $u_x$ does not change with $x$, so it can only depend on $y$. Then $u_x = f(y)$. Integrating in $x$, the general solution is $u(x,y) = f(y)\,x + g(y)$, where $f, g$ are arbitrary functions.
\end{example}

\subsection{Auxiliary Conditions}
\begin{definition}[Auxiliary Conditions]
    Extra conditions imposed on a PDE to pick out a single solution from the general solution.
\end{definition}
Two types of auxiliary conditions will be studied in this course.
\begin{definition}[Initial Value Problems]
    A PDE together with conditions on $u$ at a starting time $t = t_0$, such as $u(x,t_0) = \phi(x)$. A PDE of order $n$ in $t$ needs $n$ initial conditions, on $u, u_t, \dots, \partial_t^{\,n-1}u$.
\end{definition}
\begin{definition}[Boundary Value Problems]
    A PDE on a domain $D$ together with conditions on $u$ or its derivatives on the boundary $\partial D$.
\end{definition}

\subsection{Well-Posedness}
\begin{definition}[Well-Posedness]
    A PDE with auxiliary conditions is well-posed when:
    \begin{itemize}
        \item \textbf{Existence:} a solution exists for the given data.
        \item \textbf{Uniqueness:} there is only one solution.
        \item \textbf{Stability:} small changes in the data produce small changes in the solution.
    \end{itemize}
\end{definition}

\subsection{Transport Equation}
\begin{definition}[Solution Operator (ODE)]
    For times $t_0,t_1$, the solution operator is the map
    \[S_{t_0\to t_1}(x_0):=x(t_1),\]
    where $x(\cdot)$ solves the ODE with $x(t_0)=x_0$.
\end{definition}
\begin{proposition}[Properties of the Solution Operator]
    For $x'(s) = v(x(s),s)$ with $v$ smooth, uniqueness of trajectories gives
    \[S_{t\to t} = \text{id}, \qquad S_{t_1\to t_2}\circ S_{t_0\to t_1} = S_{t_0\to t_2}, \qquad S_{t_1\to t_0} = \left(S_{t_0\to t_1}\right)^{-1}.\]
\end{proposition}
\begin{example}
    If $v = c$, then $S_{0\to t}(x_0) = x_0 + ct$ and $S_{t\to 0}(x) = x - ct$.\\
    If $v = -x$, then $S_{0\to t}(x_0) = x_0 e^{-t}$ and $S_{t\to 0}(x) = x e^{t}$.
\end{example}
For PDEs, we let each particle carry a value $u$ that stays constant as it moves along its trajectory $x'(t) = v(x,t)$.
\begin{definition}[Linear Transport Equation]
    The linear transport equation is defined to be
    \begin{equation}
        \label{eq:transport}
        u_t+v(x,t)u_x=0,\quad u(x,0)=g(x).
    \end{equation}
\end{definition}
\begin{definition}[Transport Label]
    To find which particle is at position $x$ at time $t$, we run the ODE backwards: it is the particle that started at $x_0 = S_{t\to0}(x)$, hence
    \[u(x,t) = g\big(S_{t\to 0}(x)\big).\]
\end{definition}
\begin{example}
    $v = c$ gives $u(x,t) = g(x-ct)$.\\
    $v = -x$ gives $u(x,t) = g(xe^t)$.
\end{example}
\begin{remark}
    For \eqref{eq:transport}, the solution recipe is: flow backwards, read the label.
    \[\text{ODE flow}\to\text{transported labels}\to\text{PDE}.\]
\end{remark}

\subsection{Characteristic}
\begin{definition}[Characteristic]
    A characteristic of a PDE is a curve in the domain along which the PDE degenerates into an ODE.\footnote{Knowing $u$ at one point of the curve, makes the PDE determine the value of u along the entire curve.}
\end{definition}
\begin{proposition}[Method of Characteristics]
    Given a first-order PDE with data on a curve $\Gamma$:
    \begin{enumerate}
        \item Find the characteristics.
        \item Solve the ODE for $u$ along each one, starting from the data on $\Gamma$.
        \item To get $u$ at a point, find the characteristic through it and read off the value.
    \end{enumerate}
    So $u$ at a point only depends on the data at one point of $\Gamma$.
\end{proposition}

\subsubsection{General Solution}
\begin{example}[Constant Coefficients]
    \label{ex:constant-coeff}
    Consider the PDE with $a,b \in\R$
    \[au_x+bu_y=\langle a,b\rangle\cdot\nabla u = 0.\]
    So the derivative of $u$ in the direction $\langle a,b\rangle$ is zero, and $u$ is constant along lines parallel to $\langle a,b\rangle$. These lines have the form $bx-ay=c$. Thus the general solution to this PDE is a function of $bx-ay$ only:
    \[u(x,y) = f(bx-ay),\]
    where $f$ is just an arbitrary function.
\end{example}
\begin{example}[Variable Coefficients]
    Consider the following PDE
    \[u_x+yu_y=\langle 1,y\rangle\cdot\nabla u=0.\]
    So $u$ is constant along curves whose tangent is $\langle 1,y\rangle$ at every point, i.e.\ curves with slope
    \[\frac{dy}{dx} = \frac{y}{1} = y.\]
    The general solution of $y'=y$ is $y=Ce^x$. Two points lie on the same characteristic exactly when they give the same value of $C=e^{-x}y$, so the general solution is
    \[u(x,y) = f(e^{-x}y),\]
    where $f$ is just an arbitrary function.
\end{example}

\subsubsection{Initial Value Problem}
\begin{example}
    Consider the following PDE with initial data
    \[3u_x+2u_y = 0, \quad u(x,0) = x^3.\]
    From Example~\ref{ex:constant-coeff} the general solution is $u(x,y) = f(2x-3y)$. The initial data determines $f$:
    \[u(x,0) = f(2x) = x^3.\]
    Let $w = 2x$, then $x=\frac{w}{2}$ and
    \[f(w) = \frac{w^3}{8},\]
    therefore taking $w=2x-3y$ gives the specific solution
    \[u(x,y) = f(2x-3y) = \frac{(2x-3y)^3}{8}.\]
\end{example}

\subsubsection{Linear Equations}
\begin{definition}[Linear First-Order PDE]
    The general linear first-order PDE in two variables, with data $g$ on a curve $\Gamma$, is
    \begin{equation}
        \label{eq:linear-first-order}
        a(x,y)u_x+b(x,y)u_y = c_1(x,y)u + c_2(x,y),\quad u = g \text{ on } \Gamma.
    \end{equation}
\end{definition}
\begin{proposition}[Characteristic Equations]
    \label{prop:characteristic-eqs}
    To solve \eqref{eq:linear-first-order}, define the characteristic curves $(x(s),y(s))$ by
    \[x'(s) = a(x(s),y(s)),\quad y'(s) = b(x(s),y(s)).\]
    Then let $z(s)$ denote the values of $u$ on the curve $z(s) := u(x(s),y(s))$. By the chain rule we get
    \[z'(s) = u_xx'(s) + u_yy'(s) = u_xa(x(s),y(s))+u_yb(x(s),y(s)),\]
    which is the left side of the PDE along the curve, so
    \[z'(s) = c_1(x(s),y(s))z(s)+c_2(x(s),y(s)).\]
    Collecting, we arrive at the characteristic equations
    \[\begin{cases}x'(s)=a(x(s),y(s))\\y'(s)=b(x(s),y(s))\\z'(s)=c_1(x(s),y(s))z(s)+c_2(x(s),y(s))\end{cases}\]
    with $(x(0),y(0))\in\Gamma$ and $z(0)=g(x(0),y(0))$. Solving these and eliminating $s$ and the starting point gives $u(x,y)$.
\end{proposition}
\begin{example}[Rotational Characteristics]
    Consider the following PDE
    \[-yu_x+xu_y=u,\quad u(x,0)=g(x),\quad x>0.\]
    The characteristic equations are
    \[x'=-y,\quad y'=x, \quad z'=z.\]
    With $x(0)=x_0$, $y(0) = 0$, then
    \[x(s) = x_0\cos(s),\quad y(s) = x_0\sin(s).\]
    For $z$, with $z(0)=g(x_0)$,
    \[z(s)=g(x_0)e^s.\]
    Now we need to find $x_0,s$ in terms of $x,y$. For $x_0$, take
    \[x^2+y^2 = x_0^2\left(\cos^2(s)+\sin^2(s)\right) = x_0^2 \quad\Rightarrow\quad x_0=\sqrt{x^2+y^2},\]
    taking the positive root since $x_0>0$. For $s$, take $\frac{y(s)}{x(s)} = \frac{x_0\sin(s)}{x_0\cos(s)}=\tan(s)$, so $s$ is the polar angle $\theta$ of $(x,y)$, which for $x>0$ is
    \[s = \arctan\left(\frac{y}{x}\right).\]
    Therefore
    \[u(x,y) = g\left(\sqrt{x^2+y^2}\right)e^{\theta}.\]
\end{example}
\begin{remark}
    When one variable is time and data is given at $t=0$, dividing by the coefficient of $u_t$ writes the PDE as
    \[u_t+b(x,t)u_x=c_1(x,t)u+c_2(x,t).\]
    Then the characteristic ODE for $t$ is $t'(s)=1$ with $t(0)=0$, so $t=s$ and $t$ itself can be used as the parameter.\footnote{This works for any variable, as long as its coefficient is nonzero and the data is given where that variable is constant, e.g.\ $y=0$.}
\end{remark}

\section{Distributions}
Classical solutions must be differentiable, but some natural solutions are not.
\begin{example}[Non-Differentiable Solution]
    For the transport equation $u_t+cu_x=0$ with a jump in the initial data,
    \[g(x)=\begin{cases}1, & x<0\\0, & x>0\end{cases},\]
    the solution $u(x,t)=g(x-ct)$ is a step moving at speed $c$. It is clearly the right answer, but $u_t$ and $u_x$ do not exist along the line $x=ct$, so $u$ does not satisfy the PDE in the classical sense.
\end{example}
To count such $u$ as solutions, we need distributions: instead of asking for the value of an object at a point, we ask what it does when integrated against a smooth, localized test function $\phi$.

\subsection{Classes of functions}
\begin{definition}[Integrable Function]
    A function $f$ defined on $\R$ is integrable if
    \[\intinf|f(x)|dx < \infty.\]
\end{definition}
\begin{definition}[Locally Integrable Function]
    A function $f$ defined on $\R$ is locally integrable if $\forall a<b$,
    \[\int_{a}^{b}|f(x)|dx < \infty.\]    
\end{definition}
\begin{remark}
    Integration cannot see the value at a single point, so two functions that differ only at a point are treated as the same function.
\end{remark}

\subsection{Test Functions}
\begin{definition}[Support]
    The support of a continuous function $\phi$ is the closure of the set where $\phi \neq 0$.
\end{definition}
\begin{definition}[Compact Support]
    A continuous function on $\R$ has compact support if it is zero outside some closed bounded interval $[-a,a]$.
\end{definition}
\begin{definition}[Test Functions]
    Test functions are the elements of $C_c^{\infty}(\R)$, which is the set of functions $\phi: \R \to \R$ that are infinitely differentiable and have compact support.
\end{definition}
\begin{example}
    The standard example of a test function is the bump function
    \begin{equation}
        \label{eq:bump}
        \phi_a(x) := \begin{cases}e^{-\frac{1}{a^2-x^2}}, & |x|<a\\0, & |x|\geq a\end{cases}.
    \end{equation}
\end{example}
\begin{definition}[Convergence of Test Functions]
    $\phi_n \to \phi$ in $C^{\infty}_c(\R)$ if
    \begin{enumerate}[label=(\roman*)]
        \item there is one $a>0$ such that every $\phi_n$ and $\phi$ vanish for $|x|\geq a$, and
        \item for every $k=0,1,2,\dots$, the $k$-th derivatives converge uniformly:
        \[\max_{x\in\R}\left|\phi_n^{(k)}(x)-\phi^{(k)}(x)\right| \xrightarrow{n\to\infty} 0.\]
    \end{enumerate}
\end{definition}

\subsection{Definition of a Distribution}
\begin{definition}[Functional]
    A functional is a map whose inputs are functions.
\end{definition}
\begin{definition}[Distribution]
    A distribution $F$, also called a generalized function, is a rule assigning to each test function $\phi \in C^{\infty}_{c}(\R)$ a real number, written
    \begin{equation}
        \label{eq:distribution}
        \ip{F}{\phi},
    \end{equation}
    which is linear and continuous.
\end{definition}
\begin{remark}
    The expression $\ip{F}{\phi}$ is notation; we could equally write $F(\phi)$ in functional notation. The notation emphasizes the special case of an integral.
\end{remark}
\begin{remark}
    Distributions are linear:
    \[\ip{F}{a\phi+b\psi} = a\ip{F}{\phi} + b\ip{F}{\psi}, \quad \forall a,b \in\R, \phi,\psi \in C^{\infty}_c(\R).\]
\end{remark}
\begin{definition}[Continuity of a Functional]
    A functional $F$ is continuous if $\phi_n\to\phi$ in $C^{\infty}_c(\R)$ implies $\ip{F}{\phi_n} \to \ip{F}{\phi}$.
\end{definition}

\subsection{Functions as Distributions}
\begin{proposition}
    Let $f$ be locally integrable on $\R$. Then
    \begin{equation}
        \label{eq:function-distribution}
        \ip{F_f}{\phi} := \intinf f(x)\phi(x)\,dx,\quad \phi\in C_c^{\infty}(\R)
    \end{equation}
    defines a distribution $F_f$.
\end{proposition}
\begin{proof}
    If $\phi$ vanishes outside $[-a,a]$, then $\left|\ip{F_f}{\phi}\right| \le \max|\phi|\int_{-a}^{a}|f|\,dx < \infty$, so $F_f$ is well defined. Linearity follows from linearity of the integral. For continuity, if $\phi_n\to\phi$ in $\D$, all supports lie in one $[-a,a]$, so
    \[\left|\ip{F_f}{\phi_n} - \ip{F_f}{\phi}\right| \le \max_{x}|\phi_n(x)-\phi(x)|\int_{-a}^{a}|f|\,dx \to 0.\]
\end{proof}

\subsection{Delta Function}
\begin{definition}[Delta Function]
    As a distribution, $\delta_0$ is defined by
    \[\ip{\delta_0}{\phi} := \phi(0),\quad \phi\in C_c^{\infty}(\R).\]
\end{definition}
\begin{remark}
    By tradition we call it the delta function, even though it is a distribution and not a function, so $\delta_0(x)$ has no value at a point. Likewise, $\int\delta_0(x)\phi(x)\,dx = \phi(0)$ is only shorthand for $\ip{\delta_0}{\phi}$, not an actual integral.
\end{remark}

\subsection{Derivatives of a Distribution}
\begin{definition}[Derivatives of a Distribution]
    Let $F$ be a distribution. Its derivative $F'$ is the distribution
    \begin{equation}
        \label{eq:distribution-derivative}
        \ip{F'}{\phi} := -\ip{F}{\phi'}.
    \end{equation}
    For a more general case $F^{(n)}$, the distribution becomes
    \[\ip{F^{(n)}}{\phi} = (-1)^n\ip{F}{\phi^{(n)}}\]
\end{definition}
\begin{remark}
    If $f\in C^1(\R)$, then $(F_f)' = F_{f'}$.
\end{remark}
\begin{proof}
    \[\ip{(F_f)'}{\phi} = -\ip{F_f}{\phi'} = -\int f\phi' dx = \int f'\phi dx = \ip{F_{f'}}{\phi}\]
\end{proof}
\begin{example}[Derivative of the delta function]
    \[\ip{\delta_0'}{\phi} = -\ip{\delta_0}{\phi'} = -\phi'(0).\]
\end{example}

\subsection{Derivatives of Piecewise Smooth Functions}
\begin{example}[Derivative of the Heaviside function]
    For any test function $\phi$ vanishing outside $[-a,a]$,
    \begin{align*}
        \ip{(F_H)'}{\phi}=-\ip{F_H}{\phi'} &= -\intinf H(x)\phi'(x)\,dx\\ &= -\int_0^{a}\phi'(x)\,dx\\ &=-(\phi(a)-\phi(0)) = \phi(0) = \ip{\delta_0}{\phi}.
    \end{align*}
    Thus $H'=\delta_0$ in the sense of distributions.
\end{example}
\begin{example}[Derivative of the absolute value]
    For $f(x)=|x|$ and any test function $\phi$,
    \begin{align*}
        \ip{(F_f)'}{\phi} &= -\intinf |x|\phi'(x)\,dx = -\int_{-\infty}^{0}(-x)\phi'(x)\,dx-\int_{0}^{\infty}x\phi'(x)\,dx\\ &\underset{\text{By parts}}{=} -[-x\phi(x)]_{-\infty}^0-\int_{-\infty}^0\phi(x)\,dx - [x\phi(x)]_0^{\infty} + \int_{0}^{\infty}\phi(x)\,dx\\ &= \intinf \operatorname{sgn}(x)\phi(x)\,dx.
    \end{align*}
    Thus $\tfrac{d}{dx}|x|=\operatorname{sgn}(x)=\begin{cases}-1, & x<0\\0, & x=0\\1, & x>0\end{cases}$ in the sense of distributions.
\end{example}
\begin{remark}
    Using the same steps as the previous example, we can find that $|x|'' = \operatorname{sgn}' = 2\delta_0$ in the sense of distributions.
\end{remark}
\begin{theorem}[Derivative of a function with a jump]
    Let $f$ be $C^1$ on $\R$ except at $x_0$, and suppose $f$ and $f'$ have one-sided limits at $x_0$. Denote $a=\lim_{x\to x_0^-}f(x)$, $b=\lim_{x\to x_0^+}f(x)$, and $g(x) = f'(x)$ for $x\neq x_0$ ($g(x_0) = 0$). Then, in the sense of distributions,
    \begin{equation}
        \label{eq:jump}
        f'=g+(b-a)\delta_{x_0}.
    \end{equation}
\end{theorem}

\subsection{Convergence}
\begin{definition}[Convergence of distributions]
    A sequence of distributions $F_n$ converges to a distribution $F$ in the sense of distributions if
    \[\ip{F_n}{\phi}\overset{n\to\infty}{\to}\ip{F}{\phi},\quad \forall \phi\in\D.\]
\end{definition}
\begin{definition}[Convergence of functions]
    A sequence of locally integrable functions $f_n$ converges in the sense of distributions to a distribution $F$ if $F_{f_n}\to F$,
    \begin{equation}
        \intinf f_n(x)\phi(x)\,dx \overset{n\to\infty}{\to}\ip{F}{\phi},\quad \forall\phi\in\D.
    \end{equation}
\end{definition}
\begin{proposition}[Convergence of derivatives]
    \label{prop:derivatives-converge}
    If $F_m\to F$ in the sense of distributions, then $F_m^{(n)}\to F^{(n)}$.
\end{proposition}

\subsubsection{Pointwise Convergence}
\begin{definition}[Pointwise convergence]
    Pointwise convergence of $f_n$ to $f$ means: for every $x$, the numbers $f_n(x)$ converge to $f(x)$.
\end{definition}
\begin{proposition}[Pointwise implies distributional]
    Let $f_n$ be locally integrable, converging pointwise to a locally integrable $f$, and suppose there is locally integrable $g$ with $|f_n(x)|\leq g(x), \forall n,x$. Then $f_n\to f$ in the sense of distributions.
\end{proposition}

\subsubsection{Convergence to $\delta_0$}
\begin{example}
    Four sequences converge to $\delta_0$ in the sense of distributions, with $\sigma_n=\frac1n$:
    \begin{enumerate}
        \item Hats: $f_n(x) = \begin{cases}\frac{n}{2}, & |x|\leq\sigma_n\\0, & \text{otherwise}\end{cases}$
        \item Spikes: $f_n(x) = \begin{cases}n-n^2|x|, & |x|<\sigma_n\\0, & \text{otherwise}\end{cases}$
        \item Derivative of arctangent: $f_n(x) = \frac{1}{\pi}\frac{\sigma_n}{x^2+\sigma_n^2}$
        \item Gaussians: $f_n(x) = \frac{1}{\sqrt{2\pi\sigma_n}}e^{-\frac{x^2}{2\sigma_n}}$
    \end{enumerate}
\end{example}
\begin{remark}
    Such sequences are called approximations to the identity. They share three properties:
    \begin{enumerate}
        \item Nonnegativity: $f_n(x)\geq 0,\quad\forall x,n$.
        \item Unit mass: $\intinf f_n(x)\,dx = 1,\quad \forall n$.
        \item Concentration at $0$: as $n\to\infty$, the mass of $f_n$ concentrates near $x=0$.
    \end{enumerate}
\end{remark}
\begin{theorem}[Approximations to the identity]
    Let $f$ be a nonnegative integrable function with $\intinf f(x)\,dx = 1$, and put $f_n(x) := nf(nx)$. Then $f_n\to\delta_0$ in the sense of distributions.
\end{theorem}
\begin{proof}
    Substituting $y=nx$,
    \[\ip{F_{f_n}}{\phi} = \intinf nf(nx)\phi(x)\,dx = \intinf f(y)\phi\left(\tfrac{y}{n}\right)dy.\]
    As $n\to\infty$, $\phi(y/n)\to\phi(0)$ for every $y$, and $|f(y)\phi(y/n)|\leq \max|\phi|\,f(y)$, which is integrable. By dominated convergence, the limit passes inside: $\ip{F_{f_n}}{\phi}\to\phi(0)\intinf f(y)\,dy = \phi(0)$.
\end{proof}
\begin{remark}
    Hats, spikes and the arctangent sequence are all of the form $nf(nx)$ (from their $n=1$ member). The Gaussians are the same with $\sqrt{n}$ in place of $n$, and the same proof works.
\end{remark}
\begin{remark}
    Difference quotients of $\delta_0$ converge to $\delta_0'$: for any test function $\phi$,
    \[\ip{\frac{\delta_0-\delta_h}{h}}{\phi} = \frac{\phi(0)-\phi(h)}{h}\xrightarrow{h\to0}-\phi'(0) = \ip{\delta_0'}{\phi}.\]
    Two opposite point sources of size $\frac1h$, close together, converge to a dipole.
\end{remark}

\subsubsection{Convergence of Sinusoids}
\begin{theorem}[Oscillations converge to zero]
    $\sin(nx),\cos(nx) \to 0$ in the sense of distributions.
\end{theorem}
\begin{proof}
    For any test function $\phi$ vanishing outside $[-a,a]$, integrating by parts (the boundary terms vanish),
    \[\left|\intinf\sin(nx)\phi(x)\,dx\right| = \left|\frac{1}{n}\intinf\cos(nx)\phi'(x)\,dx\right| \leq \frac{1}{n}\int_{-a}^{a}|\phi'(x)|\,dx \to 0.\]
    The same argument works for $\cos(nx)$.
\end{proof}
\begin{remark}[Riemann--Lebesgue lemma]
    The result holds with $\phi$ replaced by any integrable function $f$: $\intinf f(x)\sin(nx)\,dx\to0$, and the same for $\cos(nx)$. This is why Fourier coefficients of an integrable function tend to zero.
\end{remark}
\begin{remark}
    By Proposition~\ref{prop:derivatives-converge}, $(\sin(nx))'=n\cos(nx)\to 0$ in the sense of distributions, even though its amplitude $n$ is unbounded.
\end{remark}
\begin{remark}
    If $f$ is periodic with period $l$ and integral $A$ over a period, then $f(nx) \to \frac{A}{l}$, the average of $f$, in the sense of distributions.
\end{remark}

\subsection{Properties of sinusoids}
\begin{proposition}[Orthogonality relations]
    Let $l>0$. For positive integers $n\neq m$,
    \[\int_0^l\sin(\frac{n\pi x}{l})\sin(\frac{m\pi x}{l})dx=0,\]
    and for every positive integer $n$,
    \[\int_0^l\sin^2(\frac{n\pi x}{l})dx = \frac{l}{2}.\]
\end{proposition}
\begin{lemma}
    For an integer $k$,
    \[\int_{-\pi}^\pi e^{ik\theta}d\theta = \begin{cases}2\pi, k=0,\\0, k\neq0.\end{cases}\]
\end{lemma}

\subsection{Coefficients}
\begin{proposition}[Coefficients of sine series]
    For a function $\phi(x) = \sum_{m=1}^\infty b_m\sin(\frac{m\pi x}{l}), for x\in(0,l)$, its coefficient is
    \begin{equation}
        \label{eq:}
        b_n = \frac{2}{l}\int_0^l\phi(x)\sin(\frac{n\pi x}{l})dx.
    \end{equation}
\end{proposition}
\begin{proposition}[Coefficients of cosine series]
    \begin{equation}
        a_n = \frac{2}{l}\int_0^l \phi(x)\cos(\frac{n\pi x}{l})dx
    \end{equation}
\end{proposition}

\subsection{Fourier Series}
\begin{definition}[Fourier sine series]
    Let $\phi$ be integrable on $(0,l)$ and define $b_n$ by Equation.\eqref{eq:}. The series is then
    \begin{equation}
        \sum_{n=1}^\infty b_n\sin(\frac{n\pi x}{l}).
    \end{equation}
\end{definition}
\begin{definition}[Fourier cosine series]
    With $a_n$ given, the fourier cosine series of $\phi$ on $(0,l)$ is
    \begin{equation}
        \phi(x) = \frac{1}{2}a_0 + \sum_{n=1}^\infty a_n\cos(\frac{n\pi x}{l}).
    \end{equation}
\end{definition}
\begin{definition}[Full fourier series]
    With $a_n,b_n$ given, the full fourier series of $\phi(x)$ on $(-l,l)$ is
    \begin{equation}
        \frac{1}{2}a_0 + \sum_{n=1}^\infty [a_n\cos(\frac{n\pi x}{l}) + b_n\sin(\frac{n\pi x}{l})].
    \end{equation}
\end{definition}

\subsection{Rotational Invariance}
\begin{definition}[Rotational Invariance]
    A PDE is rotationally invariant if rotating any of its solutions still gives a solution.
\end{definition}
\begin{proposition}[Rotational Invariance]
    Let $U \in C^2(\R^n)$, $x \in \R^n$, and $R\in\R^{n\times n}$ be an orthogonal matrix ($R^TR = I$). If $u(x) := U(Rx)$ (the rotated version of $U$), then
    \begin{equation}
        \Delta u(x) = (\Delta U)(Rx).
    \end{equation}
    If $u(x) := U(x+a)$ with $a\in\R^n$ (the translated version of $U$), then
    \begin{equation}
        \Delta u(x) = (\Delta U)(x+a).
    \end{equation}
\end{proposition}
\begin{remark}
    A rotated or translated harmonic function stays harmonic.
\end{remark}
\begin{example}
    Let $\Omega$ be the region between two spheres centered at the origin, with $u = A$ on the inner sphere and $u = B$ on the outer one ($A, B$ constants). Rotating $u$ keeps it harmonic and does not change its boundary values, since a rotation maps each sphere to itself. So $u(Rx)$ solves the same problem, and $u$ depends only on the radius $r$.
\end{example}

\subsection{Polar Coordinates in the Plane}
\begin{proposition}
    Let $u \in C^2(\Omega)$ with $\Omega \subseteq \R^2 \setminus \{0\}$, and $U$ be $u$ written in polar coordinates, then write $x=r\cos\theta$, $y=r\sin\theta$, and $u(x,y) = U(r,\theta)$, then
    \begin{equation}
        \Delta u = U_{rr}+\frac{1}{r}U_r + \frac{1}{r^2}U_{\theta\theta}.
    \end{equation}
\end{proposition}
\begin{example}
    Let $U = r^n\cos(n\theta)$ with $n \in \mathbb{N}\setminus\{0\}$. Then $U$ is harmonic:
    \begin{equation*}
    \begin{split}
        U_{rr}+\frac{1}{r}U_r + \frac{1}{r^2}U_{\theta\theta} &= n(n-1)r^{n-2}\cos(n\theta)+nr^{n-2}\cos(n\theta)-n^2r^{n-2}\cos(n\theta)\\ &= (n^2-n^2+n-n)r^{n-2}\cos(n\theta)=0.
    \end{split}
    \end{equation*}
\end{example}
