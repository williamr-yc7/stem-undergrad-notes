\subsubsection{Scalar \& Vector Fields}
\begin{definition}[Scalar Field]
    A scalar field assigns a number to each point of space.
\end{definition}
\begin{definition}[Vector Field]
    A vector field assigns a vector (magnitude and direction) to each point of space.
\end{definition}

\subsubsection{The Gradient}
\begin{definition}[Gradient]
    The gradient collects the partial derivatives into a vector
    \begin{equation}
        \label{eq:gradient}
        \vec{\nabla} f(\vec{x}) = \left(\partial_1 f,\partial_2 f,\dots,\partial_n f\right)(\vec{x}).
    \end{equation}
\end{definition}
The gradient is the vector that converts a small step $\vec{h}$ into the resulting small change in $f$, by a dot product
\[f(\vec{x}+\vec{h}) = f(\vec{x}) + \vec{\nabla} f(\vec{x})\cdot \vec{h} + o(\|\vec{h}\|).\]
\begin{definition}[Directional Derivative]
    The directional derivative of $f$ along a unit vector $\hat{v}$ is
    \begin{equation}
        \label{eq:directional-derivative}
        D_{\hat{v}}f(\vec{x}) = \lim_{t\to0}\frac{f(\vec{x}+t\hat{v})-f(\vec{x})}{t}.
    \end{equation}
\end{definition}
\begin{proposition}[Directional Derivative via the Gradient]
    \begin{equation}
        \label{eq:directional-gradient}
        D_{\hat{v}}f(\vec{x}) = \vec{\nabla} f(\vec{x})\cdot \hat{v} = \|\vec{\nabla} f(\vec{x})\|\cos\theta,
    \end{equation}
    where $\theta$ is the angle between $\hat{v}$ and $\vec{\nabla} f(\vec{x})$.
\end{proposition}

\subsubsection{Level Sets}
\begin{definition}[Level Set]
    The level set of $f$ at height $c$ is
    \begin{equation}
        \label{eq:level-set}
        \{\vec{x}:f(\vec{x})=c\}.
    \end{equation}
\end{definition}
\begin{definition}[Equipotential]
    Level sets of the potential are called equipotentials.
\end{definition}
\begin{proposition}[Gradient Is Normal to Level Sets]
    The gradient is perpendicular to the level set.
\end{proposition}

\subsubsection{Critical Points}
\begin{definition}[Critical Point]
    A critical point is a point where $\vec{\nabla} f(\vec{x}) = \vec{0}$: no direction of increase, the linear approximation is flat.
\end{definition}
\begin{remark}
    Local maxima and minima of a differentiable function occur only at critical points, but the converse fails, and the failure has a name: a critical point that is neither a maximum nor a minimum is a saddle point.
\end{remark}
\begin{definition}[Hessian]
    The Hessian is the symmetric matrix of second partials
    \begin{equation}
        \label{eq:hessian}
        H_{ij} = \partial_i\partial_j f.
    \end{equation}
\end{definition}
\begin{proposition}[Second Derivative Test]
    Near a critical point $\vec{x}$, the behavior is governed by the signs of the eigenvalues of $H$:
    \begin{itemize}
        \item all eigenvalues $>0$: local minimum (a bowl);
        \item all eigenvalues $<0$: local maximum (a dome);
        \item eigenvalues of both signs: a saddle, increasing in some directions and decreasing in others, like the point at the center of a mountain pass.
    \end{itemize}
    If some eigenvalue is $0$ and the others share a sign, the test is inconclusive.
\end{proposition}

\subsubsection{Line Integrals, Work \& Conservative Fields}
\begin{definition}[Line Integral]
    The line integral of a vector field $\vec{F}$ along a curve $C$ parametrized by $\vec{\gamma}(t)$, $a\le t\le b$, is
    \begin{equation}
        \label{eq:line-integral}
        \int_C \vec{F}\cdot d\vec{l} = \int_a^b \vec{F}(\vec{\gamma}(t))\cdot\vec{\gamma}'(t)\,dt.
    \end{equation}
\end{definition}
\begin{definition}[Work]
    The work done by a force $\vec{F}$ along a path $C$ is
    \begin{equation}
        \label{eq:work}
        W = \int_C \vec{F}\cdot d\vec{l}.
    \end{equation}
\end{definition}
\begin{definition}[Conservative Field]
    A vector field is conservative if its line integral between two points depends only on the endpoints, not on the path. Equivalently, $\oint_C \vec{F}\cdot d\vec{l} = 0$ for every closed curve $C$.
\end{definition}

\subsubsection{Divergence}
\begin{definition}[Divergence]
    The divergence of a vector field is the scalar field
    \begin{equation}
        \label{eq:divergence}
        \vec{\nabla}\cdot\vec{F} = \partial_1 F_1 + \partial_2 F_2 + \partial_3 F_3.
    \end{equation}
\end{definition}
\begin{theorem}[Divergence Theorem]
    For a volume $V$ bounded by a closed surface $S$ with outward normal,
    \begin{equation}
        \label{eq:divergence-theorem}
        \oiint_S \vec{F}\cdot d\vec{a} = \iiint_V(\vec{\nabla}\cdot\vec{F})\,d\tau.
    \end{equation}
\end{theorem}

\subsubsection{The Laplacian}
\begin{definition}[Laplacian]
    The Laplacian composes the two first-order operators, the divergence of the gradient
    \begin{equation}
        \label{eq:laplacian}
        \Delta f = \nabla^2 f = \vec{\nabla}\cdot(\vec{\nabla} f) = \partial_1^2 f + \partial_2^2 f + \partial_3^2 f = \operatorname{tr} H,
    \end{equation}
    where $\operatorname{tr} H$ is the trace of the Hessian.
\end{definition}
\begin{definition}[Harmonic Function]
    A function $f\in C^2(\Omega)$ on an open set $\Omega$ is harmonic in $\Omega$ if
    \begin{equation}
        \label{eq:harmonic}
        \Delta f = 0
    \end{equation}
    at every point of $\Omega$.
\end{definition}

\subsection{Partial Differential Equations}
\begin{definition}[Partial Differential Equation]
    A partial differential equation (PDE) is a differential equation relating an unknown function $u$ of several variables, its partial derivatives, and the independent variables.
    \[F(\text{independent variables}, u, \text{partial derivatives of }u) = 0.\]
\end{definition}
\begin{remark}
    Physics rarely hands us a function $u$ directly. It gives a law for how $u$ changes. Example: heat flows from hot to cold, so the temperature $u(x,t)$ in a room changes in time wherever it varies in space. Recovering $u$ from such a relation is the whole subject.
\end{remark}

\subsubsection{Notation}
\begin{remark}
    For the rest of this notes, we will use these terms for a general function $u(x,y,t)$
    \[\nabla u = (u_x,u_y,u_z) = \left(\frac{\partial u}{\partial x}, \frac{\partial u}{\partial y}, \frac{\partial u}{\partial t}\right),\]
    and
    \[\Delta u = u_{xx}+u_{yy}+u_{tt} = \frac{\partial^2 u}{\partial x^2} + \frac{\partial^2 u}{\partial y^2} + \frac{\partial^2 u}{\partial t^2}.\]
\end{remark}

\subsection{Classification}
\subsubsection{Order}
\begin{definition}[Order]
    The order of a PDE is the order of the highest derivative appearing in it.
\end{definition}
\begin{example}
    $u_{xx}+u_{t} = 0$ is second order, and $u_t + u u_x = 0$ is first order.
\end{example}

\subsubsection{Linearity}
\begin{definition}[Linear PDE]
    A PDE is linear if it can be written as $L(u) = f(x)$, where $f$ does not involve $u$ and $L$ satisfies linearity (Def.\ref{def:linearity}).
\end{definition}
\begin{remark}[Practical test]
    Best way to determine if a PDE is linear is to check the following rules:
    \begin{itemize}
        \item $u$ and its derivatives appear only to the first power (no $u^2$, $(u_x)^2$),
        \item they are never multiplied together (no $u\,u_x$, $u_x u_y$),
        \item they never sit inside a function (no $\sin u$, $e^u$, $\ln u$).
    \end{itemize}
\end{remark}
\begin{example}
    $u_{xx} + xy\,u_y + e^x u = \cos x$ is linear. Nonlinear: $u\,u_x + u_y = 0$ (product), $u_{xx} + (u_y)^2 = 0$ (power), $u_{xx} + \sin u = 0$ (inside a function).
\end{example}
\begin{definition}[Degrees of nonlinearity]\leavevmode
    \begin{itemize}
        \item \textbf{Semilinear:} highest derivatives appear linearly, with coefficients depending only on the independent variables.\\Example: $u_{xx} + \sin u = 0$.
        \item \textbf{Quasilinear:} highest derivatives appear linearly, but coefficients may depend on $u$ and its lower derivatives.\\Example: $u_t + u\,u_x = 0$.
        \item \textbf{Fully nonlinear:} highest derivatives appear nonlinearly.\\Example: $(u_x)^2 + (u_y)^2 = 1$.
    \end{itemize}
\end{definition}

\subsubsection{Homogeneity}
\begin{definition}[Homogeneous]
    A linear PDE $L(u) = f(x)$ is homogeneous if $f \equiv 0$.
\end{definition}
\begin{example}
    $u_t + a u_x = x^3$ is linear with $L(u) = u_t + a u_x$ and $f(x) = x^3$, so it is inhomogeneous.
\end{example}

\subsection{Classical Solutions to a PDE}
\begin{definition}[Classical solution]
    A classical solution to a PDE, with order $k$ on a domain $\Omega \subset\R^N$, is a function $u \in C^k(\Omega)$ satisfying the PDE at every point of $\Omega$.
\end{definition}
\begin{example}
    For any function $\phi \in C^1$ of one variable, $u(x,t) = \phi(x-ct)$ solves $u_t+cu_x=0$, we can check this by the chain rule, $u_t = -c\phi'(x-ct)$, $u_x = \phi'(x-ct)$, then $u_t+cu_x = -c\phi'+c\phi' = 0$.
\end{example}

\subsection{General Solutions to a PDE}
Recall from the \href{https://github.com/williamr-yc7/stem-undergrad-notes/blob/86199275ed081fd521621a48faec61bef0cb1069/Winter\%202026/Math325.pdf}{Math 325 notes} that the general solution of an ODE contains arbitrary constants.
\begin{example}
    The general solution of $y' = y$ is $y(t) = Ce^t$, where $C$ is an arbitrary constant.
\end{example}
\begin{remark}
    The general solution of a PDE contains arbitrary functions.
\end{remark}
\begin{example}
    Find all $u : \R^2 \to \R$ with $u_{xx} = 0$. Since $(u_x)_x = 0$, $u_x$ does not change with $x$, so it can only depend on $y$. Then $u_x = f(y)$. Integrating in $x$, the general solution is $u(x,y) = f(y)\,x + g(y)$, where $f, g$ are arbitrary functions.
\end{example}

\subsection{Auxiliary Conditions}
\begin{definition}[Auxiliary Conditions]
    Extra conditions imposed on a PDE to pick out a single solution from the general solution.
\end{definition}
Two types of auxiliary conditions will be studied in this course.
\begin{definition}[Initial Value Problems]
    A PDE together with conditions on $u$ at a starting time $t = t_0$, such as $u(x,t_0) = \phi(x)$. A PDE of order $n$ in $t$ needs $n$ initial conditions, on $u, u_t, \dots, \partial_t^{\,n-1}u$.
\end{definition}
\begin{definition}[Boundary Value Problems]
    A PDE on a domain $D$ together with conditions on $u$ or its derivatives on the boundary $\partial D$.
\end{definition}

\subsection{Well-Posedness}
\begin{definition}[Well-Posedness]
    A PDE with auxiliary conditions is well-posed when:
    \begin{itemize}
        \item \textbf{Existence:} a solution exists for the given data.
        \item \textbf{Uniqueness:} there is only one solution.
        \item \textbf{Stability:} small changes in the data produce small changes in the solution.
    \end{itemize}
\end{definition}

\subsection{Transport Equation}
\begin{definition}[Solution Operator (ODE)]
    For times $t_0,t_1$, the solution operator is the map
    \[S_{t_0\to t_1}(x_0):=x(t_1),\]
    where $x(\cdot)$ solves the ODE with $x(t_0)=x_0$.
\end{definition}
\begin{proposition}[Properties of the Solution Operator]
    For $x'(s) = v(x(s),s)$ with $v$ smooth, uniqueness of trajectories gives
    \[S_{t\to t} = \text{id}, \qquad S_{t_1\to t_2}\circ S_{t_0\to t_1} = S_{t_0\to t_2}, \qquad S_{t_1\to t_0} = \left(S_{t_0\to t_1}\right)^{-1}.\]
\end{proposition}
\begin{example}
    If $v = c$, then $S_{0\to t}(x_0) = x_0 + ct$ and $S_{t\to 0}(x) = x - ct$.\\
    If $v = -x$, then $S_{0\to t}(x_0) = x_0 e^{-t}$ and $S_{t\to 0}(x) = x e^{t}$.
\end{example}
\begin{definition}[Transport Label]
    To find which particle is at position $x$ at time $t$, we run the ODE backwards, it is the particle that started at $x_0 = S_{t\to0}(x)$, hence
    \[u(x,t) = g\big(S_{t\to 0}(x)\big).\]
\end{definition}
\begin{example}
    $v = c$ gives $u(x,t) = g(x-ct)$.\\
    $v = -x$ gives $u(x,t) = g(xe^t)$.
\end{example}
\begin{definition}[Linear Transport Equation]
    The linear transport equation is defined to be
    \begin{equation}
        \label{eq:transport}
        u_t+v(x,t)u_x=0,\quad u(x,0)=g(x).
    \end{equation}
\end{definition}
\begin{remark}
    For every linear transport equation in this chapter, the solution recipe is flow backwards, read the label.
    \[\text{ODE flow}\to\text{transported labels}\to\text{PDE}.\]
\end{remark}

\subsection{Rotational Invariance}
\begin{definition}[Rotational Invariance]
    A PDE is rotationally invariant if rotating any of its solutions still gives a solution.
\end{definition}
\begin{proposition}[Rotational Invariance]
    Let $U \in C^2(\R^n)$, $x \in \R^n$, and $R\in\R^{n\times n}$ be an orthogonal matrix ($R^TR = I$). If $u(x) := U(Rx)$ (the rotated version of $U$), then
    \begin{equation}
        \Delta u(x) = (\Delta U)(Rx).
    \end{equation}
    If $u(x) := U(x+a)$ with $a\in\R^n$ (the translated version of $U$), then
    \begin{equation}
        \Delta u(x) = (\Delta U)(x+a).
    \end{equation}
\end{proposition}
\begin{remark}
    A rotated or translated harmonic function stays harmonic.
\end{remark}
\begin{example}
    Let $\Omega$ be the region between two spheres centered at the origin, with $u = A$ on the inner sphere and $u = B$ on the outer one ($A, B$ constants). Rotating $u$ keeps it harmonic and does not change its boundary values, since a rotation maps each sphere to itself. So $u(Rx)$ solves the same problem, and $u$ depends only on the radius $r$.
\end{example}

\subsection{Polar Coordinates in the Plane}
\begin{proposition}
    Let $u \in C^2(\Omega)$ with $\Omega \subseteq \R^2 \setminus \{0\}$, and $U$ be $u$ written in polar coordinates, then write $x=r\cos\theta$, $y=r\sin\theta$, and $u(x,y) = U(r,\theta)$, then
    \begin{equation}
        \Delta u = U_{rr}+\frac{1}{r}U_r + \frac{1}{r^2}U_{\theta\theta}.
    \end{equation}
\end{proposition}
\begin{example}
    Let $U = r^n\cos(n\theta)$ with $n \in \mathbb{N}\setminus\{0\}$. Then $U$ is harmonic:
    \begin{equation*}
    \begin{split}
        U_{rr}+\frac{1}{r}U_r + \frac{1}{r^2}U_{\theta\theta} &= n(n-1)r^{n-2}\cos(n\theta)+nr^{n-2}\cos(n\theta)-n^2r^{n-2}\cos(n\theta)\\ &= (n^2-n^2+n-n)r^{n-2}\cos(n\theta)=0.
    \end{split}
    \end{equation*}
\end{example}
